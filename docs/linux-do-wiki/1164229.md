# [笔记] React NextJS 动态添加卸载组件

佬友们看代码, 仅限提供个简单思路, 例如可以用来做某些前端功能模块的动态配置等, 很多地方不严谨
```
'use client';

import dynamic from 'next/dynamic';
import { useState } from 'react';

const WindowMouseTest = dynamic(() => import('./WindowMouseTest'), {
  ssr: false,
});

interface Instance {
  id: string;
}

export default function Page() {
  const [instances, setInstances] = useState<Instance[]>([
    { id: crypto.randomUUID() },
    { id: crypto.randomUUID() },
    { id: crypto.randomUUID() },
  ]);

  const handleAdd = (index: number) => {
    const newItem = { id: crypto.randomUUID() };
    const newInstances = [...instances];
    newInstances.splice(index + 1, 0, newItem);
    setInstances(newInstances);
  };

  const handleRemove = (id: string) => {
    setInstances((prev) => prev.filter((item) => item.id !== id));
  };

  return (
    <div className="space-y-4 p-6">
      {instances.map((item, index) => (
        <div key={item.id} className="space-y-2">
          <WindowMouseTest />
          <div className="flex space-x-2">
            <button
              className="px-3 py-1 bg-blue-500 text-white rounded"
              onClick={() => handleAdd(index)}
            >
              添加
            </button>
            <button
              className="px-3 py-1 bg-red-500 text-white rounded"
              onClick={() => handleRemove(item.id)}
            >
              删除
            </button>
          </div>
        </div>
      ))}
    </div>
  );
}
```
