# 前置需求
1. 一个海外vps,带宽不限(10M以上就够了,3M不会吃满一天吧 ![:distorted_face:](https://linux.do/images/emoji/twemoji/distorted_face.png?v=15),其实也只有第一次全量的慢,后面差量)
2. 一个网盘,如OneDrive,GoogleDrive
3. TG账号用于通知(可选)
# 配置ghorg
1. [下载地址](https://github.com/gabrie30/ghorg/releases/tag/v1.11.6)
2. 在github获取可读仓库的key,类似ghp_xxx
3. 尝试执行命令`ghorg clone "你的用户名" --clone-type=user --token="ghp_xxx"`
4. 如果需要拉mirror格式,加上`--backup`参数
> 到这里已经可以拉到本地`~/ghorg/`下了
# 配置rclone
各个网盘的配置不同,大家自行配置 [文档](https://rclone.org/docs/)

完成后`rclone config`可以看到列表就可以
[image976×404 20.1 KB](https://cdn3.ldstatic.com/original/4X/1/e/5/1e57669fbd5468b0b0e56264d800f2fa1052de36.png)

## 单网盘备份版
```
#!/bin/bash
set -euo pipefail

#####################
# 配置区域
#####################
# Telegram 机器人信息（需自行替换）
BOT_TOKEN="YOUR_TELEGRAM_BOT_TOKEN"
CHAT_ID="YOUR_TELEGRAM_CHAT_ID"

# 二进制路径
RCLONE_BIN="/usr/bin/rclone"
GHORG_BIN="/usr/local/bin/ghorg"   # 按实际路径修改

# ghorg 配置
GHORG_USER="YOUR_GITHUB_USERNAME"
GHORG_TOKEN="YOUR_GITHUB_TOKEN"    # TODO: 替换为你的真实 token
GHORG_DIR="${HOME}/ghorg"

# rclone 远端（单云盘）
CLOUD_REMOTE_PRIMARY="remoteA:/github-backup"

# 日志（本地日志文件路径）
LOG_FILE="/var/log/github-backup-to-remoteA.log"

#####################
# 函数
#####################
send_tg() {
    local text="$1"
    curl -s "https://api.telegram.org/bot${BOT_TOKEN}/sendMessage" \
        -d "chat_id=${CHAT_ID}" \
        --data-urlencode "text=${text}" >/dev/null 2>&1 || true
}

#####################
# 主流程
#####################
START_TIME="$(date '+%F %T')"

# 启动通知（不提什么单盘）
START_MSG=$(printf '🚀 备份任务开始：GitHub → 云盘
开始时间：%s' "$START_TIME")
send_tg "$START_MSG"

# 用于最终退出码
EXIT_CODE=0

mkdir -p "${GHORG_DIR}"
cd "${GHORG_DIR}"

#####################
# 1. ghorg clone（带 --backup，最多重试 3 次）
#####################
GHORG_BK_STATUS="未执行"
GHORG_BK_ATTEMPTS=0
GHORG_BK_RET=0

for i in 1 2 3; do
    GHORG_BK_ATTEMPTS="$i"
    GHORG_BK_STATUS="进行中"

    if "${GHORG_BIN}" clone "${GHORG_USER}" --backup --clone-type=user --token="${GHORG_TOKEN}"; then
        GHORG_BK_STATUS="成功"
        GHORG_BK_RET=0
        break
    else
        GHORG_BK_RET=$?
        if [ "$i" -lt 3 ]; then
            sleep 5
        else
            GHORG_BK_STATUS="失败"
            if [ "$EXIT_CODE" -eq 0 ]; then
                EXIT_CODE=$GHORG_BK_RET
            fi
        fi
    fi
done

#####################
# 2. ghorg clone（不带 --backup，最多重试 3 次）
#####################
GHORG_NB_STATUS="未执行"
GHORG_NB_ATTEMPTS=0
GHORG_NB_RET=0

for i in 1 2 3; do
    GHORG_NB_ATTEMPTS="$i"
    GHORG_NB_STATUS="进行中"

    if "${GHORG_BIN}" clone "${GHORG_USER}" --clone-type=user --token="${GHORG_TOKEN}"; then
        GHORG_NB_STATUS="成功"
        GHORG_NB_RET=0
        break
    else
        GHORG_NB_RET=$?
        if [ "$i" -lt 3 ]; then
            sleep 5
        else
            GHORG_NB_STATUS="失败"
            if [ "$EXIT_CODE" -eq 0 ]; then
                EXIT_CODE=$GHORG_NB_RET
            fi
        fi
    fi
done

#####################
# 3. rclone sync 本地 ghorg → 云盘（单远端）
#####################
GITHUB_SYNC_STATUS="未执行"
GITHUB_SYNC_RET=0

# 理论上只要不带 --backup 的 clone 成功，就可以 sync
if [ "$GHORG_NB_STATUS" = "成功" ]; then
    if "${RCLONE_BIN}" sync "${GHORG_DIR}" "${CLOUD_REMOTE_PRIMARY}" -P \
        --log-file="${LOG_FILE}" --log-level=INFO; then
        GITHUB_SYNC_STATUS="成功"
        GITHUB_SYNC_RET=0
    else
        GITHUB_SYNC_RET=$?
        GITHUB_SYNC_STATUS="失败"
        if [ "$EXIT_CODE" -eq 0 ]; then
            EXIT_CODE=$GITHUB_SYNC_RET
        fi
    fi
else
    GITHUB_SYNC_STATUS="跳过（ghorg clone 失败）"
fi

#####################
# 最终通知（只发这一条总结消息）
#####################
END_TIME="$(date '+%F %T')"

if [ "$EXIT_CODE" -eq 0 ]; then
    FINAL_ICON="✅"
    FINAL_TITLE="备份任务完成"
else
    FINAL_ICON="❌"
    FINAL_TITLE="备份任务存在失败"
fi

FINAL_MSG=$(printf '%s %s
开始：%s
结束：%s

[1] 仓库备份（ghorg clone --backup）
  状态：%s
  尝试次数：%s
  最后退出码：%s

[2] 仓库同步（ghorg clone）
  状态：%s
  尝试次数：%s
  最后退出码：%s

[3] 本地 → 云盘（rclone sync）
  状态：%s
  最后退出码：%s
  日志：%s' \
"$FINAL_ICON" "$FINAL_TITLE" \
"$START_TIME" "$END_TIME" \
"$GHORG_BK_STATUS" "$GHORG_BK_ATTEMPTS" "$GHORG_BK_RET" \
"$GHORG_NB_STATUS" "$GHORG_NB_ATTEMPTS" "$GHORG_NB_RET" \
"$GITHUB_SYNC_STATUS" "$GITHUB_SYNC_RET" "$LOG_FILE")

send_tg "$FINAL_MSG"

exit "$EXIT_CODE"
```
