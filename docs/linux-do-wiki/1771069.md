```
def compute_normals_o3d(x, y, z, knn=30):
    valid_mask = np.isfinite(x) & np.isfinite(y) & np.isfinite(z)
    if not np.all(valid_mask):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
        z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)

    points = np.ascontiguousarray(np.vstack((x, y, z)).T, dtype=np.float64)

    num_points = points.shape[0]
    if num_points < 3:
        return np.zeros_like(x), np.zeros_like(y), np.zeros_like(z)

    actual_knn = min(knn, num_points - 1)

    pcd = o3d.geometry.PointCloud()
    pcd.points = o3d.utility.Vector3dVector(points)
    pcd.estimate_normals(search_param=o3d.geometry.KDTreeSearchParamKNN(knn=actual_knn))

    normals = np.asarray(pcd.normals)

    if normals.ndim != 2 or normals.shape[1] != 3:
        return np.zeros_like(x), np.zeros_like(y), np.zeros_like(z)

    return normals[:, 0], normals[:, 1], normals[:, 2]

def collect_plot_data(las_path, out_filename):

    # 1. 提取
    # cls = np.array(las.label, dtype=np.int32)

    # 2. 提取高精度坐标
    x = np.array(las.x, dtype=np.float64)
    y = np.array(las.y, dtype=np.float64)
    z = np.array(las.z, dtype=np.float64) # 此时 Z 是 HAG

    # 3. 平移到原点，变成样地相对坐标
    x -= np.min(x)
    y -= np.min(y)

    # 安全转回 float32
    x = x.astype(np.float32)
    y = y.astype(np.float32)
    z = z.astype(np.float32)

    # 4. 实时计算法向量
    nx, ny, nz = compute_normals_o3d(x, y, z, knn=30)

    # 5. 提取并缩放强度 (Intensity)
    intensity = np.array(las.intensity, dtype=np.float32)
    i_min, i_max = np.min(intensity), np.max(intensity)
    if i_max > i_min:
        intensity = (intensity - i_min) / (i_max - i_min)
    else:
        intensity = np.zeros_like(intensity)

    # 6. 安全提取 Original Z 维度
    try:
        orig_z = np.array(las.orginalZ, dtype=np.float32)
    except AttributeError:
        try:
            orig_z = np.array(las["OriginalZ"], dtype=np.float32)
        except Exception as e:
            print(f"警告: 在 {os.path.basename(las_path)} 中未找到 OriginalZ 属性，暂时用 z 填充。")
            orig_z = np.copy(z)

    # 7. 拼装成终极 (N, 9) 矩阵
    # 顺序: [X_relative, Y_relative, Z_HAG, nx, ny, nz, intensity, cls, orig_z]
    # 使用 ascontiguousarray 防止后续 PyTorch 报 non-contiguous tensor 警告
    data = np.ascontiguousarray(
        np.vstack([x, y, z, nx, ny, nz, intensity, cls, orig_z]).T,
        dtype=np.float32
    )

    # 8. 保存一整个样地数据
    np.save(out_filename, data)
    return True
```

```
def process_split(split_name, input_dir, output_dir):
    """
    处理单个数据集划分 (train 或 test)
    """
    # 查找该文件夹下所有的 .las 和 .LAS 文件
    las_files = glob.glob(os.path.join(input_dir, '*.las'))

    if len(las_files) == 0:
        print(f"警告: 在 {input_dir} 中没有找到任何 .las 文件！")
        return []

    print(f"\n开始处理 [{split_name}] 集，共发现 {len(las_files)} 个文件...")
    os.makedirs(output_dir, exist_ok=True)

    failed_files = []

    # tqdm 提供美观的进度条
    for las_path in tqdm(las_files, desc=f"处理 {split_name} 数据"):
        file_name = os.path.basename(las_path)
        base_name = os.path.splitext(file_name)[0]

        # 拼接输出的 NPY 路径
        out_filename = os.path.join(output_dir, f"{base_name}.npy")

        try:
            # 【核心变化】：不再需要传 num_point, block_size 等参数了！
            # 只有极简的输入和输出路径
            success = collect_plot_data(las_path=las_path, out_filename=out_filename)
            if not success:
                failed_files.append(file_name)
        except Exception as e:
            print(f"\n处理 {file_name} 时发生错误: {e}")
            failed_files.append(file_name)

    return failed_files

def main():
    # ---------------------------------------------------------
    # 1. 路径配置 (请确保与你的实际物理路径一致)
    # ---------------------------------------------------------
    DATA_ROOT = r"xxxxxxXxXxxxxxxX"

    las_base_dir = os.path.join(DATA_ROOT)

    # 输出处理后 NPY 数据的根目录
    output_base_dir = os.path.join(DATA_ROOT, 'processed_npy')

    print("=" * 50)
    print("🚀 开始 FORinstance 数据集预处理 (Online Cropping 模式)")
    print("=" * 50)

    # ---------------------------------------------------------
    # 2. 分别处理 train 和 test 文件夹
    # ---------------------------------------------------------
    splits = ['train', 'test']
    all_failed_files = []

    for split in splits:
        input_dir = os.path.join(las_base_dir, split)

    if len(all_failed_files) > 0:
        print(f"\n⚠️ 共有 {len(all_failed_files)} 个文件处理失败或被跳过:")
        for ff in all_failed_files:
            print(f" - {ff}")
    else:
        print("\n✅ 所有文件均已完美处理，没有错误！")

if __name__ == '__main__':
```

```
import os
import pickle
import logging
import numpy as np
from tqdm import tqdm
import torch
from torch.utils.data import Dataset
from ..data_util import crop_pc, voxelize
from ..build import DATASETS

@DATASETS.register_module()
class MYDataset(Dataset):
    """
    输入格式: (N, 9)
    [PlotX, PlotY, Z_HAG, nx, ny, nz, intensity, Label, OriginalZ]
    """
    gravity_dim = 2

    def __init__(self,
                 data_root: str,
                 split: str = 'train',
                 voxel_size: float = 0.1,
                 voxel_max=None,
                 transform=None,
                 loop: int = 1,
                 presample: bool = False,
                 variable: bool = False,
                 shuffle: bool = True,
                 label_mapping: dict = None,
                 ignore_index: int = -1,
                 class_names: list = None,
                 class_colors: list = None):

        super().__init__()
        self.split = split
        self.voxel_size = voxel_size
        self.transform = transform
        self.voxel_max = voxel_max
        self.loop = loop
        self.presample = presample
        self.variable = variable
        self.shuffle = shuffle
        self.label_mapping = label_mapping
        self.ignore_index = ignore_index

# ---------------------------
        # dataset meta information
        # ---------------------------
        if class_names is not None:
            self.classes = class_names
        elif self.label_mapping is not None:
            # 根据映射后的类别编号数量自动生成占位类名
            mapped_ids = sorted(set(self.label_mapping.values()))
            self.classes = [f'class_{i}' for i in mapped_ids]
        else:
            self.classes = None

        if self.classes is not None:
            self.num_classes = len(self.classes)
        elif self.label_mapping is not None:
            self.num_classes = len(set(self.label_mapping.values()))
        else:
            self.num_classes = None

        if class_colors is not None:
            self.cmap = class_colors
        elif self.num_classes is not None:
            # 给一个默认颜色表，占位即可
            default_colors = [
                [150, 100, 50],   # brown
                [170, 85, 0],     # trunk brown
                [0, 180, 0],      # green
                [180, 180, 60],   # yellow-green
                [0, 120, 255],
                [255, 0, 255],
                [100, 100, 100],
                [255, 100, 100],
            ]
            self.cmap = default_colors[:self.num_classes]
        else:
            self.cmap = None

        self.data_dir = os.path.join(data_root, split)
        self.data_list = sorted([f[:-4] for f in os.listdir(self.data_dir) if f.endswith('.npy')])

        if self.num_classes is not None:
                    self.num_per_class = self._compute_num_per_class()
        else:
            self.num_per_class = None

        processed_root = os.path.join(data_root, 'processed')
        filename = os.path.join(
            processed_root, f'forest_{split}_{voxel_size:.3f}_{str(voxel_max)}.pkl'
        )

        if presample and not os.path.exists(filename):
            np.random.seed(0)
            self.data = []
            for item in tqdm(self.data_list, desc=f'Loading ForestDataset {split} split'):
                data_path = os.path.join(self.data_dir, item + '.npy')
                cdata = np.load(data_path).astype(np.float32)

                coord = cdata[:, 0:3]   # [X, Y, Z_HAG]
                feat = cdata[:, 3:7]    # [nx, ny, nz, intensity]
                label = cdata[:, 7:8]   # [Label]

                if self.label_mapping is not None:
                    mapped_label = np.full_like(label, self.ignore_index)
                    for old_lbl, new_lbl in self.label_mapping.items():
                        mapped_label[label == old_lbl] = new_lbl
                    label = mapped_label

                cdata = np.hstack((coord, feat, label))
                self.data.append(cdata)

            npoints = np.array([len(data) for data in self.data])
            logging.info('split: %s, median npoints %.1f, avg num points %.1f, std %.1f' % (
                self.split, np.median(npoints), np.average(npoints), np.std(npoints)))
            os.makedirs(processed_root, exist_ok=True)
            with open(filename, 'wb') as f:
                pickle.dump(self.data, f)
                print(f"{filename} saved successfully")

        elif presample:
            with open(filename, 'rb') as f:
                self.data = pickle.load(f)
                print(f"{filename} load successfully")

        self.data_idx = np.arange(len(self.data_list))
        assert len(self.data_idx) > 0
        logging.info(f"\n🌲 Totally {len(self.data_idx)} samples in {split} set")

    def _compute_num_per_class(self):
        counts = np.zeros(self.num_classes, dtype=np.int64)

        for item in self.data_list:
            data_path = os.path.join(self.data_dir, item + '.npy')
            cdata = np.load(data_path).astype(np.float32)
            label = cdata[:, 7:8].astype(np.int64)

            if self.label_mapping is not None:
                mapped_label = np.full_like(label, self.ignore_index)
                for old_lbl, new_lbl in self.label_mapping.items():
                    mapped_label[label == old_lbl] = new_lbl
                label = mapped_label

            label = label.squeeze(-1)
            valid = label != self.ignore_index
            if np.any(valid):
                counts += np.bincount(label[valid], minlength=self.num_classes)

        logging.info(f"{self.split} num_per_class: {counts.tolist()}")
        return counts

    def __getitem__(self, idx):
        data_idx = self.data_idx[idx % len(self.data_idx)]

        if self.presample:
            coord, feat, label = np.split(self.data[data_idx], [3, 7], axis=1)
        else:
            data_path = os.path.join(self.data_dir, self.data_list[data_idx] + '.npy')
            cdata = np.load(data_path).astype(np.float32)

            coord = cdata[:, 0:3]   # [X, Y, Z_HAG]
            feat = cdata[:, 3:7]    # [nx, ny, nz, intensity]
            label = cdata[:, 7:8]   # [Label]

            if self.label_mapping is not None:
                mapped_label = np.full_like(label, self.ignore_index)
                for old_lbl, new_lbl in self.label_mapping.items():
                    mapped_label[label == old_lbl] = new_lbl
                label = mapped_label

        coord, feat, label = crop_pc(
            coord, feat, label, self.split, self.voxel_size, self.voxel_max,
            downsample=True,  variable=self.variable, shuffle=self.shuffle
        )

        label = label.squeeze(-1).astype(np.int64)
        data = {'pos': coord, 'x': feat, 'y': label}

        if self.transform is not None:
            data = self.transform(data)

        # if 'heights' not in data.keys():
        #     data['heights'] =  torch.from_numpy(coord[:, self.gravity_dim:self.gravity_dim+1].astype(np.float32))
        # return data
        if 'heights' not in data.keys():
            if isinstance(data['pos'], torch.Tensor):
                data['heights'] = data['pos'][:, self.gravity_dim:self.gravity_dim+1].float()
            else:
                data['heights'] = torch.from_numpy(
                    data['pos'][:, self.gravity_dim:self.gravity_dim+1].astype(np.float32)
                )
        return data

    def __len__(self):
        return len(self.data_idx) * self.loop
```
