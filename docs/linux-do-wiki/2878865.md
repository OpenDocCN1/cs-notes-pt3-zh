# AI手搓了一个在Agent客户端的消息通知MCP

功能如下：
1. **屏幕右下角原生 Toast 通知 (notify_user)**

调用 WinRT 原生 `ToastNotificationManager`，仅在屏幕右下角滑出小卡片，几秒后自动淡出并收纳至系统通知中心。
2. **Win32 内存直写剪贴板 (copy_to_clipboard)**

放弃不稳定的命令行注入，采用 `ctypes` 直接调用 Windows 底层 Win32 API（`GlobalAlloc` + `CF_UNICODETEXT` UTF-16LE）。
3. **零依赖纯净后台语音朗读 (voice_speak)**

原生调用 Windows 内置的 `System.Speech` 语音合成引擎。
4. **安全只读剪贴板资源 (system://clipboard)**

提供标准的 MCP Resource，可供大模型在需要时安全读取当前剪贴板文本。
---
### 二、完整代码实现
以下为单文件实现的完整源码（仅依赖官方 `mcp` 库）：
```
"""
Windows Desktop Companion MCP Server (Windows 原生桌面伴侣 MCP 服务 - 终极优雅版)
=============================================================================
专为桌面非侵入式体验设计：
1. notify_user: 仅在屏幕右下角滑出 Windows 10/11 原生 Toast 小通知卡片，不打扰当前工作
2. voice_speak: 调用 Windows SAPI 执行纯粹的后台语音朗读
3. copy_to_clipboard: 使用 Win32 原生 UTF-16LE API 直写内存，杜绝任何中文乱码
4. get_current_clipboard: 纯只读获取剪贴板当前内容
"""

import ctypes
import subprocess
from mcp.server.mcpserver import MCPServer

mcp = MCPServer(
    name="windows-desktop-companion",
    instructions="提供 Windows 原生桌面系统级交互能力，包括屏幕右下角 Toast 通知、语音朗读和无乱码剪贴板操作。"
)

# ----------------- 【核心能力 1：屏幕右下角优雅 Toast 通知】 -----------------

@mcp.tool()
def notify_user(title: str, message: str) -> str:
    """在 Windows 桌面右下角滑出优雅的原生 Toast 小通知卡片（非侵入式，几秒后自动滑回通知中心）。

    参数:
        title: 通知标题（如：分析完成、知识库已同步）
        message: 通知正文
    """
    safe_title = title.replace('"', '`"').replace("'", "''")
    safe_msg = message.replace('"', '`"').replace("'", "''")

    ps_toast = f"""
    $app = '{{1AC14E77-02E7-4E5D-B744-2EB1AE5198B7}}\\WindowsPowerShell\\v1.0\\powershell.exe'
    [Windows.UI.Notifications.ToastNotificationManager, Windows.UI.Notifications, ContentType = WindowsRuntime] | Out-Null
    [Windows.Data.Xml.Dom.XmlDocument, Windows.Data.Xml.Dom.XmlDocument, ContentType = WindowsRuntime] | Out-Null

    $template = @"
<toast>
    <visual>
        <binding template="ToastGeneric">
            <text>{safe_title}</text>
            <text>{safe_msg}</text>
        </binding>
    </visual>
</toast>
"@

    $xml = New-Object Windows.Data.Xml.Dom.XmlDocument
    $xml.LoadXml($template)
    $toast = New-Object Windows.UI.Notifications.ToastNotification $xml
    [Windows.UI.Notifications.ToastNotificationManager]::CreateToastNotifier($app).Show($toast)
    """
    subprocess.run(["powershell", "-NoProfile", "-Command", ps_toast], capture_output=True)
    return f"已在屏幕右下角派发通知卡片：【{title}】"

# ----------------- 【核心能力 2：原生语音朗读】 -----------------

@mcp.tool()
def voice_speak(text: str) -> str:
    """调用 Windows 原生语音引擎朗读文本，无多余噪音。

    参数:
        text: 朗读内容
    """
    safe_text = text.replace("'", "''").replace('"', '`"')
    ps_speak = f"""
    Add-Type -AssemblyName System.Speech
    $synth = New-Object System.Speech.Synthesis.SpeechSynthesizer
    $synth.Volume = 100
    $synth.Rate = 0
    $synth.Speak('{safe_text}')
    """
    subprocess.Popen(["powershell", "-NoProfile", "-Command", ps_speak])
    return f"语音正在后台播报: '{text}'"

# ----------------- 【核心能力 3：Win32 原生 UTF-16 内存剪贴板（100% 杜绝乱码）】 -----------------

@mcp.tool()
def copy_to_clipboard(text: str) -> str:
    """将文本无乱码注入 Windows 全局剪贴板，供用户随时 Ctrl+V 使用。"""
    user32 = ctypes.windll.user32
    kernel32 = ctypes.windll.kernel32

    kernel32.GlobalAlloc.restype = ctypes.c_void_p
    kernel32.GlobalAlloc.argtypes = [ctypes.c_uint, ctypes.c_size_t]
    kernel32.GlobalLock.restype = ctypes.c_void_p
    kernel32.GlobalLock.argtypes = [ctypes.c_void_p]
    kernel32.GlobalUnlock.argtypes = [ctypes.c_void_p]
    user32.SetClipboardData.argtypes = [ctypes.c_uint, ctypes.c_void_p]
    user32.SetClipboardData.restype = ctypes.c_void_p
    user32.OpenClipboard.argtypes = [ctypes.c_void_p]
    user32.OpenClipboard.restype = ctypes.c_bool

    data = (text + "\0").encode("utf-16le")

    for _ in range(10):
        if user32.OpenClipboard(0):
            break
        import time; time.sleep(0.05)
    else:
        return "写入失败：无法打开系统剪贴板"

    try:
        user32.EmptyClipboard()
        h_mem = kernel32.GlobalAlloc(0x0002, len(data))
        p_mem = kernel32.GlobalLock(h_mem)
        ctypes.memmove(p_mem, data, len(data))
        kernel32.GlobalUnlock(h_mem)
        user32.SetClipboardData(13, h_mem)  # 13 = CF_UNICODETEXT
        return "【写入成功】文本已无缝注入剪贴板！可以直接按 Ctrl+V 粘贴。"
    finally:
        user32.CloseClipboard()

# ----------------- 【资源原语：只读读取剪贴板】 -----------------

@mcp.resource("system://clipboard")
def get_current_clipboard() -> str:
    """读取用户当前剪贴板中的文本。"""
    try:
        res = subprocess.run(
            ["powershell", "-NoProfile", "-Command", "[Console]::OutputEncoding=[System.Text.Encoding]::UTF8; Get-Clipboard"],
            capture_output=True
        )
        return res.stdout.decode("utf-8", errors="replace").strip() or "【当前剪贴板为空】"
    except Exception as e:
        return f"读取剪贴板异常: {str(e)}"

if __name__ == "__main__":
    mcp.run(transport="stdio")
```

### 三、安装与 MCP 配置示例
#### 1. 安装依赖
```
pip install mcp
```
#### 2. 在客户端中配置（以 Claude Desktop 或任意 MCP Client 为例）
在你的 MCP 配置文件（如 `claude_desktop_config.json`）中加入：
```
{
  "mcpServers": {
    "windows-companion": {
      "command": "python",
      "args": [
        "C:/Users/asus/Desktop/windows_companion_mcp.py"
      ]
    }
  }
}
```
