# 脚本编写

有了所有准备后，就可以开始写脚本了，不断迭代中完成了本地尝试

迭代 1：准备一个入参.txt 文件，将语料库整理到里面，一行就是一句话，作为入参不断调用

![232×399 24.1 KB](https://cdn3.ldstatic.com/original/4X/8/6/4/864e2ac6be8362d166bffa2e9c10d2e582e50b85.png)

共计 20w 行，意味着需要调用 20w 次接口

尝试后发现，很多入参返回数据都是空。

迭代2：有了一个很好的想法，将 20w 行的入参，统计出所有单个字，排除标点符号和英文字母，再进行去重，一个字一行，只剩下 5k 行，这时候就已经能很好爬虫了

![197×389 14 KB](https://cdn3.ldstatic.com/original/4X/d/3/5/d350d502ceb9b2943e4e33d113433e7d5e02de65.png)

迭代 3: 五千行入参，其中还是会有一些生僻字，查询接口后返回也是没有数据的，在接口调用过程中，可以将有返回数据的入参复制到新文件中

迭代 4：爬到一半后想起，是不是应该开个代理呢，就跟抢银行银行带个头套一样，于是将代理加了进行，继续爬

import json
import logging
import os
import sys
import threading
import time
import warnings
from concurrent.futures import ThreadPoolExecutor, as_completed
from datetime import datetime
from queue import Queue, Empty
from threading import Lock, Event
from typing import List, Dict, Tuple

import pandas as pd
import requests

# 禁用所有警告
warnings.filterwarnings('ignore')

# 配置日志格式和处理器
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(levelname)s] [%(threadName)s] %(message)s',
    handlers=[
        logging.StreamHandler(sys.stdout)
    ]
)

# 配置参数
BASE_URL = 'https://www.xxx.com/api/search'
# MAX_WORKERS = 10  # 并发线程数
MAX_WORKERS = 2  # 并发线程数
BATCH_SIZE = 100  # 批量保存大小
MAX_RETRIES = 3   # 最大重试次数
REQUEST_TIMEOUT = 10  # 请求超时时间（秒）

# 全局变量
query_counter = 0
counter_lock = Lock()
excel_lock = Lock()
data_queue = Queue()  # 用于存储待保存的数据
existing_pks = set()  # 用于存储已存在的pk
stop_event = Event()  # 用于通知所有线程停止
save_thread = None   # 保存数据的线程

def terminate_program(error_msg: str, line_number: int = None):
    """
    终止程序的辅助函数
    Args:
        error_msg: 错误信息
        line_number: 发生错误的行号
    """
    if line_number is not None:
        logging.error(f"程序终止于第 {line_number} 行: {error_msg}")
        # 保存错误行号到文件
        with open('last_error_line.txt', 'w') as f:
            f.write(str(line_number))
    else:
        logging.error(f"程序终止: {error_msg}")

    # 发送退出信号给保存线程，让它处理缓冲区数据
    data_queue.put(None)

    # 等待数据保存线程完成
    try:
        # 等待最多30秒
        if 'save_thread' in globals() and save_thread.is_alive():
            save_thread.join(timeout=30)
            if save_thread.is_alive():
                logging.error("数据保存线程未能在30秒内完成")
    except Exception as e:
        logging.error(f"等待数据保存时出错: {e}")

    # 设置停止标志
    stop_event.set()

    sys.exit(1)  # 强制退出程序

def read_input_file(file_path='chinese_only.txt', start_line=1) -> List[Tuple[int, str]]:
    """
    从文本文件中读取单字输入，每行一个字
    Args:
        file_path: 文本文件路径
        start_line: 开始读取的行号（从1开始计数）
    Returns:
        List of tuples containing (line_number, character)
    """
    chars = []
    total_lines = sum(1 for _ in open(file_path, 'r', encoding='utf-8'))

    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            # 跳过前start_line-1行
            for _ in range(start_line - 1):
                next(f, None)

            # 从start_line开始读取
            for i, line in enumerate(f, start_line):
                if i % 1000 == 0:
                    logging.info(f'读取进度: {i}/{total_lines} ({(i/total_lines*100):.2f}%)')

                character = line.strip()
                if not character:
                    continue

                chars.append((i, character))

        logging.info(f'总共读取了 {len(chars)} 个字符（从第 {start_line} 行开始）')
        return chars
    except Exception as e:
        terminate_program(f'读取文件时出错: {str(e)}')
        return []

def fetch_data_with_retry(keyword: str, query_type: str, total_queries: int, line_number: int) -> None:
    """
    带重试机制的数据获取
    """
    global query_counter

    # 如果收到停止信号，直接返回
    if stop_event.is_set():
        return

    current_thread = threading.current_thread()

    for attempt in range(MAX_RETRIES):
        try:
            # 更新计数器
            with counter_lock:
                query_counter += 1
                current_count = query_counter

            logging.info(f'线程 {current_thread.name} 正在处理第 {line_number} 行: {current_count}/{total_queries} - {query_type}: {keyword}')

            # 发送请求
            params = {'kw': keyword}
            response = requests.get(
                BASE_URL,
                params=params,
                timeout=REQUEST_TIMEOUT,
                verify=False
            )
            response.raise_for_status()
            data = response.json()

            # 添加原始关键词和行号
            for item in data:
                item['original_keyword'] = keyword
                item['line_number'] = line_number
                if 'pk' in item:  # 使用API返回的pk字段
                    item['pk'] = int(item['pk'])  # 确保pk是整数类型

            # 将数据放入队列
            data_queue.put(data)
            return

        except Exception as e:
            if attempt == MAX_RETRIES - 1:
                terminate_program(f"线程 {current_thread.name} 获取数据失败，已达到最大重试次数: {e}", line_number)
            else:
                logging.warning(f"线程 {current_thread.name} 获取第 {line_number} 行 {keyword} 数据失败 (尝试 {attempt + 1}/{MAX_RETRIES}): {e}")
                time.sleep(1)  # 等待1秒后重试

def save_data_worker():
    """
    后台数据保存线程
    """
    global existing_pks

    # 设置线程名
    threading.current_thread().name = 'SaveWorker'

    output_dir = 'output'
    os.makedirs(output_dir, exist_ok=True)
    filepath = os.path.join(output_dir, 'source.xlsx')

    # 只读取现有的pk到内存中
    if os.path.exists(filepath):
        try:
            df = pd.read_excel(filepath)
            existing_pks = set(df['pk'].tolist() if 'pk' in df.columns else [])
            logging.info(f"已从文件中读取 {len(existing_pks)} 个唯一pk")
        except Exception as e:
            terminate_program(f"读取现有pk时出错: {e}")

    data_buffer = []  # 新数据缓冲区

    while not stop_event.is_set():
        try:
            # 获取数据，设置超时以便检查退出条件
            try:
                data = data_queue.get(timeout=5)
            except Empty:
                continue

            if data is None:  # 退出信号
                # 处理剩余数据
                if data_buffer:
                    try:
                        df = pd.DataFrame(data_buffer)
                        if 'pk' in df.columns:
                            # 对剩余数据进行去重
                            df = df.drop_duplicates(subset=['pk'], keep='first')
                            df = df[~df['pk'].isin(existing_pks)]
                            unique_data = df.to_dict('records')

                            if len(unique_data) > 0:
                                save_batch_to_excel(unique_data, filepath)
                                logging.info(f"最终保存 {len(unique_data)} 条新数据")
                    except Exception as e:
                        logging.error(f"保存剩余数据时出错: {e}")
                break

            # 将新数据添加到缓冲区
            data_buffer.extend(data)

            # 当缓冲区达到1000条时进行去重
            if len(data_buffer) >= 1000:
                try:
                    logging.info(f"缓冲区达到1000条，进行去重处理")
                    # 对缓冲区数据进行去重
                    df = pd.DataFrame(data_buffer)
                    if 'pk' in df.columns:
                        # 先进行批次内去重
                        df = df.drop_duplicates(subset=['pk'], keep='first')
                        print("批次内去重完成，剩余数据量:", len(df))
                        # 再与已存在的pk去重
                        df = df[~df['pk'].isin(existing_pks)]
                        print("已存在的pk去重完成，剩余数据量:", len(df))

                        # 获取去重后的数据
                        unique_data = df.to_dict('records')

                        # 如果去重后的数据达到100条或以上，保存到文件
                        if len(unique_data) >= 10:
                            save_batch_to_excel(unique_data, filepath)
                            # 更新已存在的pk集合
                            existing_pks.update(df['pk'].tolist())
                            logging.info(f"已保存 {len(unique_data)} 条新数据，当前共有 {len(existing_pks)} 个唯一pk")
                            # 清空已保存的数据
                            data_buffer = []
                        else:
                            # 如果不足100条，保留在缓冲区继续累积
                            data_buffer = unique_data

                except Exception as e:
                    logging.error(f"处理数据批次时出错: {e}")
                    # 尝试保存当前缓冲区数据
                    if data_buffer:
                        try:
                            save_batch_to_excel(data_buffer, filepath)
                            logging.info(f"保存了 {len(data_buffer)} 条未处理的数据")
                            data_buffer = []
                        except Exception as save_e:
                            logging.error(f"保存未处理数据时出错: {save_e}")

        except Exception as e:
            logging.error(f"保存数据时出错: {e}")
            # 尝试保存当前缓冲区数据
            if data_buffer:
                try:
                    save_batch_to_excel(data_buffer, filepath)
                    logging.info(f"保存了 {len(data_buffer)} 条未处理的数据")
                except Exception as save_e:
                    logging.error(f"保存未处理数据时出错: {save_e}")
            break

def save_batch_to_excel(data_batch: List[Dict], filepath: str):
    """
    批量保存数据到Excel
    """
    global existing_pks

    if not data_batch:
        return

    try:
        with excel_lock:
            new_df = pd.DataFrame(data_batch)

            if os.path.exists(filepath):
                existing_df = pd.read_excel(filepath)

                if 'pk' in new_df.columns:
                    # 更新pk集合
                    existing_pks.update(new_df['pk'].tolist())

                    # 追加新数据
                    combined_df = pd.concat([existing_df, new_df], ignore_index=True)
                else:
                    combined_df = pd.concat([existing_df, new_df], ignore_index=True)
            else:
                combined_df = new_df
                if 'pk' in new_df.columns:
                    existing_pks.update(new_df['pk'].tolist())

            # 保存到Excel
            combined_df.to_excel(filepath, index=False, engine='openpyxl')
            logging.info(f"批量数据已追加到 {filepath}, 当前共有 {len(combined_df)} 条记录")

    except Exception as e:
        # 保存为JSON作为备份
        backup_file = filepath.replace('.xlsx', f'_backup_{datetime.now().strftime("%Y%m%d_%H%M%S")}.json')
        try:
            with open(backup_file, 'w', encoding='utf-8') as f:
                json.dump(data_batch, f, ensure_ascii=False, indent=2)
            logging.info(f"数据已备份为JSON: {backup_file}")
        except Exception as backup_error:
            terminate_program(f"保存数据和备份都失败了: 原始错误: {e}, 备份错误: {backup_error}")
        terminate_program(f"保存到Excel失败: {e}")

def process_qa_pair(args: Tuple[Tuple[int, str], int, int]) -> None:
    """
    处理单个入参
    """
    qa_pair, total_pairs, thread_id = args
    line_number, question = qa_pair

    # 设置有意义的线程名
    current_thread = threading.current_thread()
    current_thread.name = f'Worker-{thread_id}'

    fetch_data_with_retry(question, "问题", total_pairs, line_number)

def main():
    """
    主函数
    """
    # 检查是否有上次的错误行号
    start_line = 164
    if os.path.exists('last_error_line.txt'):
        try:
            with open('last_error_line.txt', 'r') as f:
                start_line = int(f.read().strip())
                logging.info(f"从上次错误的第 {start_line} 行继续执行")
        except Exception as e:
            logging.warning(f"读取上次错误行号失败: {e}")

    start_time = time.time()

    # 读取输入文件
    qa_pairs = read_input_file(start_line=start_line)
    if not qa_pairs:
        terminate_program("没有读入参")
        return

    total = len(qa_pairs)
    logging.info(f'开始处理 {total} 个入参（总计 {total} 次查询）')

    # 启动保存数据的后台线程
    global save_thread
    save_thread = threading.Thread(target=save_data_worker)
    save_thread.start()

    # 准备任务参数
    tasks = [(pair, total, i % MAX_WORKERS) for i, pair in enumerate(qa_pairs)]

    # 使用线程池处理任务
    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = [executor.submit(process_qa_pair, task) for task in tasks]

        # 等待所有任务完成
        for future in as_completed(futures):
            try:
                future.result()
            except Exception as e:
                terminate_program(f"任务执行出错: {e}")

    # 发送退出信号给保存线程
    data_queue.put(None)
    save_thread.join()

    # 统计信息
    end_time = time.time()
    total_time = end_time - start_time
    avg_time = total_time / total if total > 0 else 0

    logging.info(f'''处理完成！
总耗时: {total_time:.2f}秒
平均每次查询耗时: {avg_time:.2f}秒
总处理入参数量: {total}
总API调用次数: {total}''')

if __name__ == "__main__":
    main()
