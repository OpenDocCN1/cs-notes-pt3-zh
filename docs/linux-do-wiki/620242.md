# 函数：解析域名获取IP地址（IPv4）
function Resolve-DomainsToIPs {
    param($domains)
    $resolvedIPs = @()
    foreach ($domain in $domains) {
        try {
            $results = Resolve-DnsName $domain -Type A -ErrorAction Stop
            foreach ($result in $results) {
                if ($result.IPAddress -match '^\d+\.\d+\.\d+\.\d+$') {
                    $resolvedIPs += $result.IPAddress
                }
            }
        } catch {
            Write-Host "域名解析失败: $domain"
        }
    }
    return $resolvedIPs | Select-Object -Unique
}

# 合并预定义IP和解析到的IP
$resolvedIPs = Resolve-DomainsToIPs $Domains
$AllIPs = $IPs + $resolvedIPs | Select-Object -Unique

# 创建防火墙规则（出站阻止）
foreach ($ip in $AllIPs) {
    $ruleName = "BlockBloatware_IP_$($ip.Replace('.','_'))"
    if (-not (Get-NetFirewallRule -DisplayName $ruleName -ErrorAction SilentlyContinue)) {
        New-NetFirewallRule -DisplayName $ruleName `
            -Direction Outbound `
            -Action Block `
            -RemoteAddress $ip `
            -Protocol Any `
            -Enabled True
        Write-Host "已屏蔽IP: $ip"
    } else {
        Write-Host "规则已存在: $ip"
    }
}

# 可选：屏蔽常见全家桶软件路径（需根据实际安装路径修改）
$SoftwarePaths = @(
)

foreach ($path in $SoftwarePaths) {
    if (Test-Path $path) {
        $ruleName = "BlockBloatware_App_$((Get-Item $path).Name)"
        New-NetFirewallRule -DisplayName $ruleName `
            -Direction Outbound `
            -Program $path `
            -Action Block `
            -Enabled True
        Write-Host "已屏蔽程序: $path"
    }
}

Write-Host "`n完成！已阻止全家桶域名/IP和常见软件的网络访问。"
Write-Host "提示：需定期更新IP列表以确保有效性。"
