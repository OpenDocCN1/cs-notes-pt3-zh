# C++ 高精度算法 CSP 模板

# 高精度算法
1. *高精度算法是什么*

· ![:white_check_mark:](https://linux.do/images/emoji/twemoji/white_check_mark.png?v=15) **一句话解释： **当数字大到 long long 都装不下时，我们就用“数组”来模拟“手算竖式”，这就是高精度算法。**

2.*模板* **包含所有高精度算法**
```
#include <bits/stdc++.h>
using namespace std;

struct BigInt {
    vector<int> d;            // d[0] 是个位，d[1] 是十位……
    /* 1. 构造函数 *****************************************************/
    BigInt(string s = "0") {  // 从字符串初始化
        d.clear();
        for (int i = (int)s.size() - 1; i >= 0; --i) d.push_back(s[i] - '0');
    }

    /* 2. 去掉前导 0 ***************************************************/
    void trim() {
        while (d.size() > 1 && d.back() == 0) d.pop_back();
    }

    /* 3. 转字符串（为了输出） ******************************************/
    string toString() const {
        string res;
        for (int i = (int)d.size() - 1; i >= 0; --i) res += char(d[i] + '0');
        return res.empty() ? "0" : res;
    }

    /* 4. 输入输出 ******************************************************/
    friend istream& operator>>(istream& in, BigInt& x) {
        string s; in >> s; x = BigInt(s); return in;
    }
    friend ostream& operator<<(ostream& out, const BigInt& x) {
        return out << x.toString();
    }

    /* 5. 比较大小 ******************************************************/
    bool operator<(const BigInt& rhs) const {
        if (d.size() != rhs.d.size()) return d.size() < rhs.d.size();
        for (int i = (int)d.size() - 1; i >= 0; --i)
            if (d[i] != rhs.d[i]) return d[i] < rhs.d[i];
        return false;
    }
    bool operator<=(const BigInt& rhs) const { return !(rhs < *this); }
    bool operator>(const BigInt& rhs)  const { return rhs < *this; }
    bool operator>=(const BigInt& rhs) const { return !(*this < rhs); }
    bool operator==(const BigInt& rhs) const { return !(*this < rhs) && !(rhs < *this); }
    bool operator!=(const BigInt& rhs) const { return !(*this == rhs); }

    /* 6. 高精度加法 ****************************************************/
    BigInt operator+(const BigInt& rhs) const {
        BigInt res;
        int carry = 0, i = 0;
        while (i < (int)d.size() || i < (int)rhs.d.size() || carry) {
            int sum = carry;
            if (i < (int)d.size()) sum += d[i];
            if (i < (int)rhs.d.size()) sum += rhs.d[i];
            res.d.push_back(sum % 10);
            carry = sum / 10;
            ++i;
        }
        return res;
    }

    /* 7. 高精度减法 (保证 *this >= rhs) ********************************/
    BigInt operator-(const BigInt& rhs) const {
        BigInt res = *this;
        int borrow = 0, i = 0;
        while (i < (int)rhs.d.size() || borrow) {
            if (i < (int)rhs.d.size()) res.d[i] -= rhs.d[i] + borrow;
            else res.d[i] -= borrow;
            if (res.d[i] < 0) { res.d[i] += 10; borrow = 1; }
            else borrow = 0;
            ++i;
        }
        res.trim();
        return res;
    }

    /* 8. 高精度乘法 ****************************************************/
    BigInt operator*(const BigInt& rhs) const {
        BigInt res;
        res.d.assign(d.size() + rhs.d.size(), 0);
        for (int i = 0; i < (int)d.size(); ++i)
            for (int j = 0; j < (int)rhs.d.size(); ++j)
                res.d[i + j] += d[i] * rhs.d[j];
        int carry = 0;
        for (int i = 0; i < (int)res.d.size(); ++i) {
            int tmp = res.d[i] + carry;
            res.d[i] = tmp % 10;
            carry = tmp / 10;
        }
        res.trim();
        return res;
    }

    /* 9. 高精度除法 (*this / rhs) 与取模 (*this % rhs) *****************/
    /* 同时返回商和余数，避免重复计算 */
    pair<BigInt, BigInt> divmod(const BigInt& rhs) const {
        BigInt quotient, remainder = *this;
        quotient.d.assign(d.size(), 0);
        for (int i = (int)d.size() - 1; i >= 0; --i) {
            remainder.d.insert(remainder.d.begin(), 0);   // 高位补 0
            remainder.trim();
            int cnt = 0;
            while (remainder >= rhs) {
                remainder = remainder - rhs;
                ++cnt;
            }
            quotient.d[i] = cnt;
        }
        quotient.trim();
        return {quotient, remainder};
    }

    BigInt operator/(const BigInt& rhs) const { return divmod(rhs).first; }
    BigInt operator%(const BigInt& rhs) const { return divmod(rhs).second; }
};
```
主函数示例
```
int main() {
    BigInt a, b;
    cin >> a >> b;
    cout << "a + b = " << a + b << '\n';
    if (a >= b) cout << "a - b = " << a - b << '\n';
    else        cout << "a - b = -" << b - a << '\n';
    cout << "a * b = " << a * b << '\n';
    cout << "a / b = " << a / b << '\n';
    cout << "a % b = " << a % b << '\n';
    return 0;
}
```
