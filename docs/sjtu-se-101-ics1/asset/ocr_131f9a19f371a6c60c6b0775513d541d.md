    movl    -12(%ebp), %eax    # i
    sall    $2, %eax           # 4*i
    addl    8(%ebp), %eax      # a+4*i
    movl    (%eax), %eax       # a[i]
    movl    %eax, 4(%esp)      # pushl a[i]
    movl    $LC0, (%esp)       # pushl "%d\n"
    call    _printf
    addl    $1, -12(%ebp)      # i = i + 1
