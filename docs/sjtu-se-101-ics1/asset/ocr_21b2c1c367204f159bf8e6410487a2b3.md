# Example

```
xorl    %edx,          %edx          # i = 0
xorl    %eax,          %eax          # val = 0
movl    8(%ebp),        %ecx          # get x
L2:
    leal    (%eax,%eax,4), %eax
    addl    %eax,          %eax          # val *= 10
    addl    (%ecx,%edx,4), %eax          # val += x[i]
    addl    $1,            %edx          # i++
    cmpl    $5,            %edx          # i ? 5
    jne     L2
```
