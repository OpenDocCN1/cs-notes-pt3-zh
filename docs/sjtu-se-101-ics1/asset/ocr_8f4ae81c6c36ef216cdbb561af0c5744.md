# EXAMPLE

```
CC := gcc
OPT := -O3
CFLAGS := -Wall -I. $(OPT)

BINS := gen_log gen_sort

all: $(BINS)

%.o: %.c
	$(CC) -c -o $@ $^ $(CFLAGS)

clean:
	rm $(BINS)

Makefile
```

```
>ls
gen_log.c  gen_sort.c
>make
gcc -Wall -I. -O3   gen_log.c  -o gen_log
gcc -Wall -I. -O3   gen_sort.c -o gen_sort
>make clean
rm gen_log gen_sort
>make gen_sort
gcc -Wall -I. -O3   gen_sort.c -o gen_sort
```
