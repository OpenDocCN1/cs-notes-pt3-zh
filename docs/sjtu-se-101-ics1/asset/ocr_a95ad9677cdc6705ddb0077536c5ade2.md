# Jump Table Implementation

```
code jt[7] = {loc_a, loc_def, loc_b, loc_c, loc_d, loc_def, loc_d};
```

```
int switch_eg_goto ( int x, int n)
{
    unsigned ni = n-100;
    int result = x ;
    if ( ni >6 )
        goto loc_def ; //default
    goto jt[ni];

loc_a: //100
    result *= 13 ;
    goto done ;
loc_b: //102
    result += 10 ;  /* fall through */
```

```
loc_c: //103
    result +=11;
    goto done ;
loc_d: //104, 106
    result *= result ;
    goto done ;
loc_def: //default
    result = 0 ;
done:
    return result ;
}
```
