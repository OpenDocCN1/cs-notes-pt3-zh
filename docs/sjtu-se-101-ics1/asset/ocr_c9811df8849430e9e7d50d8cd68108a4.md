# While Loop Translation

```
int fib_w(int n)
{
    int i=1;
    int val=1;
    int nval=1;

    while ( i<n ) {
        int t=val+nval ;
        val = nval ;
        nval = t ;
        i++;
    }
    return val ;
}
```

```
int fib_w_goto(int n)
{
    int val=1;
    int nval=1;
    int nmi, t ;

    if ( val >= n )
        goto done ;
    nmi = n-1;

loop:
    t=val+nval ;
    val = nval ;
    nval = t ;
    nmi--;
    if ( nmi )
        goto loop
 done:
    return val
}
```
