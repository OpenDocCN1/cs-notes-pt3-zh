# Boolean Algebra

# And

A&B = 1 when both A=1 and B=1

| & | 0 | 1 |
|---|---|---|
| 0 | 0 | 0 |
| 1 | 0 | 1 |

# Or

A|B = 1 when either A=1 or B=1

| | | 0 | 1 |
|---|---|---|---|
| 0 | | 0 | 1 |
| 1 | | 1 | 1 |

# Not

~A = 1 when A=0

| ~ |  |
|---|---|
| 0 | 1 |
| 1 | 0 |

# Exclusive-Or (Xor)

A^B = 1 when either A=1 or B=1, but not both

| ^ | 0 | 1 |
|---|---|---|
| 0 | 0 | 1 |
| 1 | 1 | 0 |
