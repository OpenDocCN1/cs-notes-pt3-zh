```asm

2.     movl  12(%ebp), %eax                        get n
3.     subl  $100, %eax                            compute index = n - 100
4.     cmpl  $6, %eax                              compare index: 6
5.     ja    .L2                                   If >, goto default
6.     jmp   *.L7(, %eax, 4)
7. .L2:                                             default:
8.     mov   $0, %eax                              result = 0
9.     jmp   .L8                                   goto done
10. .L5:                          loc_c: // 103
11.   movl  %edx, %eax            result = x
12.   jmp   .L9                   goto rest
13. .L3:                          loc_a: // 100
```

## Switch Statements

```c
int switch_eg(int x, int n)
{
    int result = x ;
    switch ( n ) {
    case 100:
        result *= 13 ;
        break ;
    case 102:
        result += 10 ;
        /* fall through */
    case 103:
        result += 11;
        break ;
    case 104: case 106:
        result *= result ;
        break ;
    default:
        result = 0 ;
    }
    return result ;
}
```

```c
int switch_eg_goto ( int x, int n)
{
    unsigned ni = n - 100;
    int result = x ;
    if ( ni > 6 )
        goto loc_def; // default
    goto jt[ni];

loc_a: // 100
    result *= 13 ;
    goto done ;
loc_b: // 102
    result += 10 ;  /* fall through */

loc_c: // 103
    result += 11;
    goto done ;
loc_d: // 104, 106

    goto done ;
loc_def: // default
    result = 0 ;
done:
    return result ;
}
```
