## Procedures - Register

- X86-64/Linux Register Usage
  - Arguments passed via regs
    - %rcx %rdx %rsi %rdi %r8 %r9
    - If more than 6 integer parameters, then pass rest on stack
  - Return value by %rax
  - No frame pointer
  - Special
    - %rsp stack pointer
    - %r11 used for linking

| %rax ret | %r8 arg#5 |
| %rbx | %r9 arg#6 |
| %rcx arg#4 | %r10 |
| %rdx arg#3 | %r11 link |
| %rsi arg#2 | %r12 |
| %rdi arg#1 | %r13 |
| %rsp stack | %r14 |
| %rbp | %r15 |

## Procedures - Stack

- x86-64/Linux Stack Frame
  - Caller Stack Frame
    - Arguments passed via registers
    - Return Address (pushed by "call")
  - Callee Stack Frame
    - Saved registers
    - Local variables

## X86-64 Swap

```c
void swap(long *xp, long *yp)
{
    long t0 = *xp;
    long t1 = *yp;
    *xp = t1;
    *yp = t0;
}
```

```asm
swap:
    movq (%rdi), %rdx
    movq (%rsi), %rax
    movq %rax, (%rdi)
    movq %rdx, (%rsi)
    ret
```

- Operands passed in registers
  - First (xp) in %rdi, second (yp) in %rsi
- No stack operations required (except ret)
- Avoid stack
  - Can hold all local information in registers

## Local Variables in Stack

```c
void swap_a(long *xp, long *yp)
{
    volatile long loc[2];
    loc[0] = *xp;
    loc[1] = *yp;
    *xp = loc[1];
    *yp = loc[0];
}
```

```asm
swap_a:
    movq (%rdi), %rax
    movq %rax, -24(%rsp)
    movq (%rsi), %rax
    movq %rax, -16(%rsp)
    movq -16(%rsp), %rax
    movq %rax, (%rdi)
    movq -24(%rsp), %rax
    movq %rax, (%rsi)
    ret
```

- Avoid Stack Pointer change
  - Can hold all information within small windows beyond stack pointer

## Without Stack Frame

```c
long s_count = 0

void swap_b(long a[], int i)
{
    swap(&a[i], &a[i+1]);
    s_count++
}
```

- No value held while swap being invoked
- No callee save registers needed

```asm
swap_b:
movslq %esi,%rsi          # sign extend
leaq (%rdi,%rsi,8), %rdi  # &a[i]
leaq 8(%rdi,%rsi,8),%rsi  # &a[i+1]
call swap                 # swap()
incq s_count(%rip)        # s_count++;
ret
```

## Call using Jump

```c
long scount = 0

void swap_c(long a[], int i)
{
    swap(&a[i], &a[i+1]);
}
```

- Directly return from swap
- Possible since swap is a “tail call”

```asm
swap_c:
    movslq %esi,%rsi        # Sign extend
    leaq (%rdi,%rsi,8), %rdi # &a[i]
    leaq 8(%rdi, rsi,8), %rsi # &a[i+1]
    jmp swap # swap()
```

## Understanding x86-64 Stack Frame

```asm
swap_d:
    movq %rbx, -16(%rsp)    # save %rbx
    movslq %esi,%rbx
    movq %r12, -8(%rsp)    # save %r12
    movq %rdi, %r12
    leaq (%rdi,%rbx,8), %rdi
    subq $16, %rsp
    leaq 8(%rdi), %rsi
    ...
    addq %rax, sum(%rip)
    movq (%rsp), %rbx
    movq 8(%rsp), %r12
    addq $16, %rsp
    ret
```

## Understanding x86-64 Stack Frame

```asm
swap_d:
    movq %rbx, -16(%rsp)    # save %rbx
    movslq %esi,%rbx
    movq %r12, -8(%rsp)    # save %r12
    movq %rdi, %r12
    leaq (%rdi,%rbx,8), %rdi
    subq $16, %rsp          # move stack frame
    leaq 8(%rdi), %rsi
    ...
    addq %rax, sum(%rip)
    movq (%rsp), %rbx       # restore %rbx
    movq 8(%rsp), %r12      # restore %r12
    addq $16, %rsp          # move stack frame
    ret
```
