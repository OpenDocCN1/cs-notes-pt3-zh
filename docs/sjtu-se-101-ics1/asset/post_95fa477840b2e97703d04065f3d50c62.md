
## While Loop Translation

```
int fib_w(int n)
{
    int i=1;
    int val=1;
    int nval=1;

    while ( i<n ) {
        int t=val+nval ;
        val = nval ;
        nval = t ;
        i++;
    }
    return val ;
}
```

```
int fib_w_goto(int n)
{
    int val=1;
    int nval=1;
    int nmi, t ;

    if ( val >= n )
        goto done ;
    nmi = n-1;

loop:
    t=val+nval ;
    val = nval ;
    nval = t ;
    nmi--;
    if ( nmi )
        goto loop
 done:
    return val
}
```

## Switch Statements

```
int switch_eg(int x, int n)
{
    int result = x ;
    switch ( n ) {
    case 100:
        result *= 13 ;
        break ;
    case 102:
        result += 10 ;
        /* fall through */
```

```
case 103
    result += 11;
    break ;

    result *= result ;
    break ;
default:
    result = 0 ;
}
```
