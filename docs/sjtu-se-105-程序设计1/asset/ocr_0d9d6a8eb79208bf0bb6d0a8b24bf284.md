#include <vector>
#include "std_lib_facilities.h"

int n;
cout << "Please input the upper bound:";
cin >> n;
vector<int> primeNumbers;
for (int i=2;i<=n;i++){
    bool pn=true;
    for (int j=0;j<primeNumbers.size();j++)
        if (i%primeNumbers[j] == 0){
            pn=false;
            break;
        }
    if (pn==true)
        primeNumbers.push_back(i);
}
