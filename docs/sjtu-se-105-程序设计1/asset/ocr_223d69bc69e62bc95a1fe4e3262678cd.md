bool Book::is_valid(string isbnin){
    istringstream ss(isbnin);
    int temp = 0;
    char c = '';
    if(ss>>temp){
        ss >> c;
        if(c=='-')
            if(ss>>temp){
                ss >> c;
                if(c=='-')
                    if(ss>>temp) {
                        ss >> c;
                        if(c=='-') {
                            ss >> c;
                            if(isalnum(c))
                                return true;
                        }
                    }
            }
    }
    return false;
};
