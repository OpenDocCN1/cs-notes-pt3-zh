```
struct Transaction{
    Book* book;
    Patron* patron;
    Date date;
    Transaction(Book* bb, Patron* pp, Date dd): book(bb), patron(pp), date(dd){};
};
```

```
ostream& operator<<(ostream& os, Transaction& t){
    cout <<"Transaction:" <<endl;
    cout << (*t.book) << " is borrowed by" << (*t.patron);
    return os;
};
```
