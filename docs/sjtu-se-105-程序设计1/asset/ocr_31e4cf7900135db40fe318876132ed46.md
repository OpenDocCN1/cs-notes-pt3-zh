// 简单的算术运算

int main()
{
    cout << "please enter a floating-point number: ";
    double n;                         // 浮点数变量
    cin >> n;
    cout << "n == " << n
         << "\nn+1 == " << n+1          // '\n' 表示“换行"
         << "\nthree times n == " << 3*n
         << "\ntwice n == " << n+n
         << "\nn squared == " << n*n
         << "\nhalf of n == " << n/2
         << "\nsquare root of n ==" << sqrt(n)   // 库函数
         << endl;                        // 换行的另一个名字
}
