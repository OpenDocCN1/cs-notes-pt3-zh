+   - 如果一个成员是protected的，那就意味着：
+     - 它可以被这个类以及所有的导出类的成员访问

```
const int MAXELTS = 100;
class IntSet {
    // a mutable set of integers, |set| <= MAXELTS
    protected:
    /* all of the data members plus indexOf */
    int elts[MAXELTS];
    int numElts;
    public:
    /* the public interface */
};
```
