class Date {
public:
    // ...
    int day() const { return d; }      // const member: can't modify

Date d(2000, Date::jan, 20);
const Date cd(2001, Date::feb, 21);

cout << d.day() << " - " << cd.day() << endl;      // ok
d.add_day(1);    // ok
cd.add_day(1);   // error: cd is a const
