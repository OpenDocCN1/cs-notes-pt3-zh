```
int main()
{
    string previous = " ";    // 初值
    string current;
    while (cin>>current) {    // 读入单词流
        if (previous == current) // 检查新输入单词是否与最后一个输入单词相同
            cout << "repeated word: " << current << '\n';
        previous = current;
    }
}
```
