```
int main()
{
    cout << "Please enter expression (we can handle + and -): ";
    int lval = 0;
    int rval;
    char op;
    int res;
    cin>>lval>>op>>rval;    // 读入表达式，例如1 + 3
    if(op=='+')
        res = lval + rval;  // 加法
    else if(op=='-')
        res = lval - rval;  // 减法
    cout << "Result: " << res << "\n";
    keep_window_open();
    return 0;
}
```
