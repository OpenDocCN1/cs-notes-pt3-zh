const int notFound = - 1;

int binSearch( const vector<int> &vec, int val ) {
    int low = 0;
    int high = vec.size() - 1;

    while ( low <= high ){
        int mid = ( low + high ) / 2;
        if ( val == vec[ mid ] ) return mid;
        if ( val < vec[ mid ] ) high = mid - 1;
        else low = mid + 1;
    }

    return notFound;
}
