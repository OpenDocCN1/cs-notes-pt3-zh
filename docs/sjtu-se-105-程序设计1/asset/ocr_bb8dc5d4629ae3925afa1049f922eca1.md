for (int i = 1; i <= max; i++) {
    vector<string> temps;
    for (int j = 0; j < inputs.size(); j++) {
        if (inputs[j].size() == i)
            temps.push_back(inputs[j]);
    }
    sort(temps.begin(), temps.end());

    for (int j = 0; j < temps.size(); j++)
        cout << temps[j] << endl;
}
