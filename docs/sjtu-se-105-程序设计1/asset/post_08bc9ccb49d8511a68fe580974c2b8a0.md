## 冒泡排序

```
void BubbleSort(int* r,int length){
    int i,j,tmp;
    for(i = 0; i < length; i++){
        for(j = i+1; j < length; j++)
            if(r[i] > r[j]){
                tmp = r[i];
                r[i] = r[j];
                r[j] = tmp;
            }
    }
}

int main(){
    int a[10] = {2,4,6,8,10,1,3,5,7,9};
    BubbleSort(a,10);
}
```

## 选择排序

```
void SelectionSort(int* r,int length){
    int i,j,k,tmp;
    for(i = 0; i < length-1; i++){
        k = i;
        for(j = i+1; j < length; j++)
            if(r[k] > r[j])
                k = j;
        tmp = r[i];
        r[i] = r[k];
        r[k] = tmp;
    }
}
```

## 插入排序

```
void InsertSort(int* r, int length){
    int i, j, tmp;
    for(i = 1; i < length; i++){
        tmp = r[i];
        for(j = i - 1; j >= 0; j--)
            if(tmp < r[j])
                r[j+1] = r[j];
            else
                break;
        r[j+1] = tmp;
    }
}
```

```
void qs(int* a, int begin,int end) {
    int compare=a[begin], left = begin, right = end;
    if (left > right)
        return;
    while (left < right){
        while ((left < right) && a[right] >= compare)
            right--;
        swap(a[left],a[right]);
        while ((left < right) && a[left] < compare)
            left++;
        swap(a[right],a[left]);
    }
    qs(a,begin,right-1);
    qs(a,right+1,end);
}

void QuickSort (int* r, int length){
    qs(r,0,length-1);
}
```

int LinearSearch(int *arr,int length,int num)
{
    for(int i=0;i<length;i++)
        if(arr[i] == num)
            return i;
    return -1;
}

## 二分搜索

```
int BinarySearch(int *arr,int length,int num){
    int begin = 0;
    int end = length - 1;
    int mid;
    while(begin <= end){
        mid = begin + (end - begin)/2;
        if (arr[mid] == num)
            return mid;
        if (arr[mid] < num)
            begin = mid + 1;
        else
            end = mid - 1;
    }
    return -1;
}
```

## 二分搜索 - 递归

```cpp
int RBS(int *arr,int begin,int end, int num){
    int mid = begin + (end - begin)/2;
    if (begin <= end){
        if (arr[mid] == num)
            return mid;
        if (arr[mid] < num)
            begin = mid + 1;
        else
            end = mid - 1;
        RBS(arr,begin, end, num);
    }else
        return -1;
}
int RecursiveBinarySearch(int *arr,int length,int num){
    return RBS(arr, 0, length-1, num);
}
```

## 重载函数声明
- 函数重载 function overloading
  - 允许多个函数共享同一个函数名
  - 但是针对不同参数类型提供共同的操作
  - 1 + 3
  - 1.0 + 3.0
  - "1.0" + "3.0"

## 为什么要重载一个函数名
- 我们希望定义一组函数
  - 它们执行同样的一般性动作
  - 但是应用在不同的参数类型上
  - int i_max( int, int );
  - int vi_max( const vector<int> & );
  - int matrix_max( const matrix & );
  - int max( int, int );
  - int max( const vector<int> & );
  - int max( const matrix & );

- 编译器按如下步骤解释多次出现的函数名
- 如果两个函数的参数表中参数的个数或类型不同，则认为这两个函数是重载的
  ```
  void print( const string & );
  void print( vector<int> & );
  ```
- 如果两个函数的返回类型和参数表精确匹配，则第二个声明被视为第一个的重复声明（编译不报错）
  ```
  void print( const string &str );
  void print( const string & );
  ```
