```cpp
#include <math.h>
#include <vector>
#include "std_lib_facilities.h"

vector<int> primeNumbers;
for (int i=2;i<=100;i++){
    bool pn=true;
    for (int j=2;j<sqrt((double)i);j++)
        if (i%j == 0){
            pn=false;
            break;
        }
    if (pn==true)
        primeNumbers.push_back(i);
}

for (int i=0; i<primeNumbers.size();i++)
    cout << "Prime Number:" << primeNumbers[i] << endl;
```

## 指针类型

- C++可以在运行时获取变量或者对象的地址，并且通过地址操纵数据
- 指针定义
  int *ip1, *ip2;
  string *pstring;
  double *dp;
  long *lp, lp2;
  
  int *ip1;
  int i = 255;
  ip1 = &i;

![](img/d2d3f908f0e1c0f8ca98d9dfbe006a9f_191_0.png)

## Example 1: 指针
```cpp
int i=255;
int *pi = &i;
*pi = *pi + 2;
// i 加2 (i = i + 2)
pi = pi + 2;
// 加到pi 包含的地址上
```

## 指针赋值

- 指针不能被初始化或赋值为其他类型对象的地址值（&obj）
  int *pi = 0;
  double dval;
  pi = &dval;  //error,无效的类型赋值: int* ← double*

- 空（void*）类型指针表明这个是个指针，但是指向的数据类型不明。因此可以被任何数据指针类型的地址值赋值。
  void *pv = pi;
  pv = pd;

## Example 1: 指针
```cpp
int i, j, k;
int *pi = &i;
*pi = *pi + 2;
// i 加2 (i = i + 2)
pi = pi + 2;
// 加到pi 包含的地址上
```

## Example 2: 指针
```cpp
int ia[10];
int *iter = &ia[0];
int *iter_end = &ia[10];

while ( iter != iter_end ) {
    do_something_with_value(*iter);
    // 现在iter 指向下一个元素
    ++iter;
}
```
