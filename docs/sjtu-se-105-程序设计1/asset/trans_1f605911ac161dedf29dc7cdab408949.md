bool Book::is_valid(string isbnin){
    istringstream ss(isbnin);
    int temp = 0;
    char c = '';
    if(ss>>temp){
        ss >> c;
        if(c=='-')
            if(ss>>temp){
                ss >> c;
                if(c=='-')
                    if(ss>>temp) {
                        ss >> c;
                        if(c=='-') {
                            ss >> c;
                            if(isalnum(c))
                                return true;
                        }
                    }
            }
    }
    return false;
};

class Patron{
    string name;
    string number;
    int owing;
public:
    Patron(string namein, string numberin, int owingin){
        name=namein;
        number=numberin;
        owing=owingin;
    }
    string get_name(){return name;}
    string get_number(){return number;}
    int get_owing(){return owing;}
    void set_owing(int owingin){ owing= owingin;}
    bool is_owed(){
        if (owing < 0)
            return true;
        return false;
    }
};
