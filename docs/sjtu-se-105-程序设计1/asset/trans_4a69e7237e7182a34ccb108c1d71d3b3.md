```cpp
string GetMaxNumber(string str){
    string max_number=str.substr(0,3);
    str=str.substr(3);

    while(str.size()!=0)
    {
        string current_number=str.substr(0,3);
        if(max_number.compare(current_number)<0){
            max_number=current_number;
        }
        str=str.substr(3);
    }

    return max_number;
}
```

- 质数
  - 也是称为素数 Prime Number
  - 只能被1和它自身整除的数字
- 问题1:
  - 请找出1-100之内的所有质数

- 第一个版本
`#include "std_lib_facilities.h"`
```
int main()
{
  for (int i=2;i<=100;i++){
    bool pn=true;
    for (int j=2;j<i;j++)
      if (i%j == 0)
        pn=false;
    if (pn==true)
      cout << "Prime Number:" << i << endl;
  }
  keep_window_open();
}
```

## 实例-计算质数

- 第五版本

```
#include <math.h>
#include "std_lib_facilities.h"

int main()
{
    int primeNumbers[50];
    int sizeOfPn = 0;
    for (int i=2;i<=100;i++){
        bool pn=true;
        for (int j=2; j<=sqrt((double)i); j++)
            if (i%j == 0) {
                pn=false;
                break;
            }
        if (pn==true){
            primeNumbers[sizeOfPn] = i;
            sizeOfPn++;
        }
    }
    keep_window_open();
}
```
