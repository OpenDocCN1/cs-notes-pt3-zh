# The exp.h Interface

```
/*
 * Class: ConstantExp
 * ----------------
 * This subclass represents a constant integer expression.
 */

class ConstantExp: public Expression {
public:
    ConstantExp(int val);

    virtual int eval(EvalState & state);
    virtual std::string toString();
    virtual ExpressionType type();

    int getValue();

private:
    int value;
};
```
