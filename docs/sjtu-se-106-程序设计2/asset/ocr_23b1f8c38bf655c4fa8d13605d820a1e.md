# The hashmap.h Interface

```
/*
 * Method: size
 * Usage: int nEntries = map.size();
 * --------------------------------
 * Returns the number of entries in this map.
 */

int size();

/*
 * Method: isEmpty
 * Usage: if (map.isEmpty()) ...
 * --------------------------------
 * Returns true if this map contains no entries.
 */

bool isEmpty();

/*
 * Method: clear
 * Usage: map.clear();
 * ----------------
 * Removes all entries from this map.
 */

void clear();
```
