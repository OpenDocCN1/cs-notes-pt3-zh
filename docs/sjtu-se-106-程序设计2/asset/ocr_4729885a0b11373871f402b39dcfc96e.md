# The PostalLookup Program

```
int main() {
    Map<string,string> stateMap;
    initStateMap(stateMap);
    while (true) {
        cout << "Enter two-letter state abbreviation: ";
        string code = getLine();
        if (code == "") break;
        if (stateMap.containsKey(code)) {
            cout << code << " = " << stateMap.get(code) << endl;
        } else {
            cout << code << " = ???" << endl;
        }
    }
}
```
