bool isBalanced(Node* root, int& depth) {
    int left = 0, right = 0;
    if (!root) return true;
    if (!isBalanced(root->left, left)) return false;
    if (!isBalanced(root->right, right)) return false;
    depth = max(left, right) + 1;
    return abs(left - right) <= 1;
}
