void Vector<T>::resize(size_t n, T val) {
    if (n <= Size) Size = n;
    else{
        T *tmp=new T [Size];
        for (int i=0; i < Size; i++)
            tmp[i]=vec[i];
        delete [] vec;
        vec=new T [2*n+1]; csize=2*n+1;
        for (int i = 0; i < Size; i++)
            vec[i] = tmp[i];
        for (int i = Size+1; i < n; i++)
            vec[i]=val;
        Size=n;
        delete [] tmp;
    }
}
