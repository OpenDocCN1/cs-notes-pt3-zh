Vector<T>& Vector<T>::operator=(const Vector<T>& x) {
    if (Size != x.Size) {
        if (x.Size < 1) {
            Size = x.Size; csize = Size; vec = NULL;
        } else {
            Size = x.Size; csize = Size;
            delete []vec; vec=new T[x.Size];
        }
    }
    for (int i = 0; i < Size; i++)
        vec[i] = x.vec[i];
    return *this;
}
