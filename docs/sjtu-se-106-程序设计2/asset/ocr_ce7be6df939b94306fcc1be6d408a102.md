# The class-based graph.h Interface

```
/*
 * Method: size
 * Usage: int size = g.size();
 * -------------------------
 * Returns the number of nodes in the graph.
 */

int size();

/*
 * Method: isEmpty
 * Usage: if (g.isEmpty()) ...
 * -------------------------
 * Returns true if the graph is empty.
 */

bool isEmpty();

/*
 * Method: clear
 * Usage: g.clear();
 * -------------------------
 * Reinitializes the graph to be empty, freeing any heap storage.
 */

void clear();
```
