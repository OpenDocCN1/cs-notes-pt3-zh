```
Vector<T>& Vector<T>::operator=(const Vector<T>& x) {
    if (Size != x.Size) {
        if (x.Size < 1) {
            Size = x.Size; csize = Size; vec = NULL;
        } else {
            Size = x.Size; csize = Size;
            delete []vec; vec=new T[x.Size];
        }
    }
    for (int i = 0; i < Size; i++)
        vec[i] = x.vec[i];
    return *this;
}
```

```
void Vector<T>::resize(size_t n, T val) {
    if (n <= Size) Size = n;
    else{
        T *tmp=new T [Size];
        for (int i=0; i < Size; i++)
            tmp[i]=vec[i];
        delete [] vec;
        vec=new T [2*n+1]; csize=2*n+1;
        for (int i = 0; i < Size; i++)
            vec[i] = tmp[i];
        for (int i = Size+1; i < n; i++)
            vec[i]=val;
        Size=n;
        delete [] tmp;
    }
}
```

## Random Writer

- Vector & Map
- Order-k Markov model
  - Count occurrences of k-character sequences
  - Choose the most commonly occurring sequence as seed
  - Next character is chosen based on the probability that it follows the seed
  - Repeats until you have 2000 characters
    - stop writing early if no characters to choose

- 以词作为单元，以汉字作为单元
- 推断某文章最可能的作者
- 中文分词
