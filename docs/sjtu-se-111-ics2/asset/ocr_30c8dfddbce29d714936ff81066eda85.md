J-type  J name        PC₃₆..₆₃ ← name

JAL name        Regs[R31] ← PC+8; PC₃₆..₆₃ ← name
((PC+4)−2²⁷) <= name <= ((PC+4)+ 2²⁷)

JR R3        PC ← Regs[R3]

JALR R2        Regs[R31] ← PC+8; PC ← Regs[R3]

((PC+4)−2¹⁷) <= name <= ((PC+4)+ 2¹⁷)

R-type  MOVZ R1,R2,R3 if (Regs[R3]==0) Regs[R1] ← Regs[R2]
