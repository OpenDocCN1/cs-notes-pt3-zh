```
#demo-j.ys
0x000:    xorl %eax, %eax
0x002:    jne t          # Not taken
0x007:    irmovl $1, %eax # Fall through
0x00d:    nop
0x00e:    nop
0x00f:    nop
0x010:    halt
0x011: t:  irmovl $3, %edx # Target (Should not execute)
```
