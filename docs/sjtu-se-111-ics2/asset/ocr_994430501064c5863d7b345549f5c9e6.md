/* if lst arg is a builtin command,
   run it and return true */
int builtin_command(char **argv)
{
    if (!strcmp(argv[0], "quit"))  /* quit command */
        exit(0);
    if (!strcmp(argv[0], "&"))  /* ignore singleton & */
        return 1;
    return 0;  /* not a builtin command */
}
