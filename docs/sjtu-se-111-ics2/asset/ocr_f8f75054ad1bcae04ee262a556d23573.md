while(1) {
    Sigemptyset(&mask);
    Sigaddset(&mask, SIGCHLD);
    Sigprocmask(SIG_BLOCK, &mask, NULL);

    /* child process */
    if ((pid = Fork()) == 0) {
        Sigprocmask(SIG_UNBLOCK, &mask, NULL);
        Execve("/bin/ls", argv, NULL);
    }

    /* parent process */
    addjob(pid) ; /* Add the child to the job list */
    Sigprocmask(SIG_UNBLOCK, &mask, NULL);
}
