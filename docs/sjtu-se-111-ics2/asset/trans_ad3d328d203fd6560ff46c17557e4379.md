{
    Sem_init(&mutex, 0, 1);

## Prethreading
```
void echo_cnt(int connfd)
{
    int n; char buf[MAXLINE]; rio_t rio;
    static pthread_once_t once = PTHREAD_ONCE_INIT;

    Pthread_once(&once, init_echo_cnt);
    Rio_readinitb(&rio, connfd);
    while((n = Rio_readlineb(&rio, buf, MAXLINE)) != 0) {
        P(&mutex);
        byte_cnt += n;
        printf("thread %d rec. %d(%d) byte on fd %d\n",
               (int)pthread_self(),n,byte_cnt, connfd);
        V(&mutex);
        Rio_writen(connfd, buf, n);
    }
}
```

## 线程安全函数
- 从线程中调用的函数必须是线程安全的
- 定义：函数是线程安全的当且仅当它在被多个并发线程重复调用时始终能产生正确的结果。

## 线程安全函数
- 我们识别出四类（非不相交的）线程不安全函数：
  - 类 1：未能保护共享变量。
  - 类 2：依赖于跨调用的持久状态。
  - 类 3：返回指向静态变量的指针。
  - 类 4：调用线程不安全函数。

## 线程不安全函数（情况 2）
- 类 2：依赖于跨调用的持久状态
```
unsigned int next = 1;
/* rand - return pseudo-random int on 0..32767 */
int rand(void)
{
    next = next * 110351524 + 12345 ;
    return (unsigned int)((next/65536) % 32768);
}
/* srand - set seed for rand() */
void srand(unsigned int seed)
{
    next = seed;
}
```

## 线程不安全函数（情况 2）
- 将状态作为参数的一部分传递
  - 从而消除静态状态
```
/* rand_r - return pseudo-random int on 0..32767 */
int rand_r(int *nextp)
{
    *nextp = *nextp * 110351524 + 12345 ;
    return (unsigned int)((*nextp/65536) % 32768);
}
```

## 线程安全函数
- 类 3：返回指向静态变量的指针
```
struct hostent *gethostbyname(char name)
{
    static struct hostent host;
    <contact DNS and fill in h>
    return &host;
}
```

## 线程安全函数
```
hostp = Malloc(...);
gethostbyname_r(name, hostp);
```
- 类 3：（修复方法）
  - 重写以使调用者传递指向结构的指针
    - 问题：需要在调用者和被调用者中进行更改
  - “加锁并复制”
    - 问题：调用者必须释放内存
```
struct hostent *gethostbyname_ts(char *name)
{
    struct hostent *p,*q = Malloc(...);
    P(&mutex);
    p = gethostbyname(name);
    *q = *p;
    V(&mutex);
    return q;
}
```

## 线程安全函数
- 类 4：调用线程不安全函数。
  - 调用一个线程不安全函数会使整个调用它的函数变为线程不安全
  - 修复：修改函数使其仅调用线程安全函数 ☺

## 可重入函数
- 函数是可重入的当且仅当它在被多个线程调用时不访问任何共享变量。
  - 可重入函数是线程安全函数的重要子集。
  - 不需要同步操作。
