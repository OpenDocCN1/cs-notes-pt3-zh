public List<User> get() throws SQLException {

    rowset.beforeFirst();
    while (rowset.next()) {
        User record = new User();
        record.setId(rowset.getLong(1));
        record.setUsername(rowset.getString(2));
        record.setPassword(rowset.getString(3));
        record.setEmail(rowset.getString(4));
        records.add(record);
    }
    return records;
}
