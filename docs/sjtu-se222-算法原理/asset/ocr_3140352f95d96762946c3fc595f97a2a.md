for(int i=0;i<n;i++)

2、开始建立哈夫曼树，
int x,y;
for (int i=n;i<2*n-1;i++) {
    selectMin(x,y,0,i);//从1到i选择两个权值最小的
    HTree[x].parent=HTree[y].parent=i;
    HTree[i].weight=HTree[x].weight+HTree[y].weight;
    HTree[i].LChild=x;
    HTree[i].RChild=y;
    HTree[i].parent=-1;
}
