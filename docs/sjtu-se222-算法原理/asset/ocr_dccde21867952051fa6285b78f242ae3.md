# Strassen’s algorithm

D₁ = (A₁₁ + A₂₂)(B₁₁ + B₂₂)
D₂ = (A₂₁ + A₂₂)B₁₁
D₃ = A₁₁(B₁₂ − B₂₂)
D₄ = A₂₂(B₂₁ − B₁₁)
D₅ = (A₁₁ + A₂₂)B₂₂
D₆ = (A₂₁ − A₁₁)(B₁₁ + B₁₂)
D₇ = (A₁₂ − A₂₂)(B₂₁ + B₂₂)

⎛ C₁₁ C₁₂ ⎞   ⎛ D₁ + D₄ − D₅ + D₇   D₃ + D₅ ⎞
⎝ C₂₁ C₂₂ ⎠ = ⎝ D₂ + D₄           D₁ + D₃ − D₂ + D₆ ⎠

        7T(n/2) + 18(n/2)² a   if n ≥ 2 }
