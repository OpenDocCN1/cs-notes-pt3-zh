# Sys_unlink

struct inode *dp =
nameiparent(path, name)

```
begin_trans();
ilock(dp);
// Cannot unlink "." or ".."
if(namecmp(name, ".") == 0 || namecmp(name, "..") == 0)
    goto bad;
if((ip = dirlookup(dp, name, &off)) == 0)
    goto bad;
ilock(ip);
if(ip->nlink < 1)
    panic("unlink: nlink < 1");
if(ip->type == T_DIR && !isdirempty(ip)){
    iunlockput(ip);
    goto bad;
}
memset(&de, 0, sizeof(de));
if(writei(dp, (char*)&de, off, sizeof(de)) != sizeof(de))
    panic("unlink: writei");
if(ip->type == T_DIR){
    dp->nlink--;
    iupdate(dp);
}
iunlockput(dp);
ip->nlink--;
iupdate(ip);
iunlockput(ip);
commit_trans();
```
