void
computc(int c)
{
    if(panicked){
        cli();
        for(;;)
            ;
    }

    if(c == BACKSPACE){
        uartputc('\b'); uartputc(' '); uartputc('\b');
    } else
        uartputc(c);
    cgaput(c);
}

    uint *argp;
    char *s;

    if(locking)
        acquire(&cons.lock);

        panic("null fmt");

    for(i = 0; (c = fmt[i] & 0xff) != 0; i++){

static void
cgaput(int c)
{
    int pos;

    // Cursor position: col + 80*row
    outb(CRTPORT, 14);
    pos = inb(CRTPORT+1) << 8;
    outb(CRTPORT, 15);
    pos |= inb(CRTPORT+1);

    if(c == '\n')
        pos += 80 - pos%80;
    else if(c == BACKSPACE){
        if(pos > 0) --pos;
    } else
        crt[pos++] = (c&0xff) | 0x0700;  // black on white

    if((pos/80) >= 24){  // Scroll up.
        memmove(crt, crt+80, sizeof(crt[0])*23*80);
        pos -= 80;
        memset(crt+pos, 0, sizeof(crt[0])*(24*80 - pos));
    }

    outb(CRTPORT, 14);
    outb(CRTPORT+1, pos>>8);
    outb(CRTPORT, 15);
    outb(CRTPORT+1, pos);
    crt[pos] = ' ' | 0x0700;
}
