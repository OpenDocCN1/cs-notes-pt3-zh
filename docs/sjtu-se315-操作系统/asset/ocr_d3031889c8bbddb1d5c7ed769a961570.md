# Polling Mechanism

```
32 // Wait for IDE disk to become ready.
33 static int
34 idewait(int checkerr)
35 {
36     int r;
37
38     while(((r = inb(0x1f7)) & (IDE_BSY|IDE_DRDY)) != IDE_DRDY)
39     ;
40     if(checkerr && (r & (IDE_DF|IDE_ERR)) != 0)
41         return -1;
42     return 0;
43 }
```
