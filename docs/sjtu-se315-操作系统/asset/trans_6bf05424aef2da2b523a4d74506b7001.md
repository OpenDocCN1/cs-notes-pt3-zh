```c
// Interrupt handler.
void
ideintr(void)
{
    struct buf *b;

    // First queued buffer is the active request.
    acquire(&idelock);
    if((b = idequeue) == 0){
        release(&idelock);
        // printf("spurious IDE interrupt\n");
        return;
    }
    idequeue = b->qnext;

    // Read data if needed.
    if((b->flags & B_DIRTY) && idewait(1) >= 0)
        insl(0x1f0, b->data, 512/4);

    // Wake process waiting for this buf.
    b->flags |= B_VALID;
    b->flags &= ~B_DIRTY;
    wakeup(b);

    // Start disk on next buf in queue.
    if(idequeue != 0)
        idestart(idequeue);

    release(&idelock);
}
```

## IDE.c

- 驱动程序一次只能处理 1 个操作
  - 仅发送队列前端的缓冲区
  - 其他的只是等待轮到自己
- iderw() 维护一个请求队列
  - 将缓冲区 b 添加到队列末尾
  - 如果 b 是第一个，则调用 idestart()，如果不是，则只是等待
  - 然后只是睡眠，而不是轮询
- ideintr() 将处理第一个请求
  - 然后如果有的话启动其他请求

## I/O Device Notifying the OS

- 操作系统需要知道何时：
  - I/O 设备已完成一个操作
  - I/O 操作遇到错误
- 两种方法
  - 轮询
  - 中断驱动
- 轮询：
  - I/O 设备将完成/错误信息放在设备特定的状态寄存器中
  - 操作系统定期检查状态寄存器
  - 优点：开销低
  - 缺点：如果 I/O 操作不频繁或不可预测，可能会浪费大量周期在轮询上

## Interrupt Driven

- CPU 在 I/O 之前设置中断处理程序向量
- CPU 发出 I/O 请求并继续其他任务
- I/O 设备处理 I/O 请求
- I/O 设备触发 CPU 中断请求线
- 中断处理程序接收中断并分发到正确的处理程序

## Interrupt Driven

- 优点：处理不可预测事件效果好
- 缺点：中断相对开销较高
- 一些设备可能会同时结合轮询和中断驱动
  - 高带宽网络设备示例：
    - 第一个入站数据包中断
    - 直到硬件空为止轮询后续数据包

```c
void
consoleintr(int (*getc)(void))
{
    int c;

    acquire(&input.lock);
    while((c = getc()) >= 0){
        switch(c){
        case C('P'):  // Process listing.
            procdump();
            break;
        case C('U'):  // Kill line.
            while(input.e != input.w &&
                input.buf[(input.e-1) % INPUT_BUF] != '\n'){
                input.e--;
                consputc(BACKSPACE);
            }
            break;
        case C('H'): case '\x7f':  // Backspace
            if(input.e != input.w){
                input.e--;
                consputc(BACKSPACE);
            }
            break;
        default:
            if(c != 0 && input.e-input.r < INPUT_BUF){
                c = (c == '\r') ? '\n' : c;
                input.buf[input.e++ % INPUT_BUF] = c;
                consputc(c);
                if(c == '\n' || c == C('D') || input.e == input.r+INPUT_BUF){
                    input.w = input.e;
                    wakeup(&input.r);
                }
            }
            break;
        }
    }
    release(&input.lock);
}
```
