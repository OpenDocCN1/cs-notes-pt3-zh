template <class T>
void adjust (T *a, const int root, const int n)
{ // No node index is > n
    T e=a[root];
    // find proper place for e
    for (int j=2*root; j<=n; j*=2) {
        if (j<n && a[j]<a[j+1]) j++;  // j is the lager child of its parent
        if (e>=a[j]) break;
        a[j/2]=a[j];  //move jth record up the tree
    }
    a[j/2]=e;
}
