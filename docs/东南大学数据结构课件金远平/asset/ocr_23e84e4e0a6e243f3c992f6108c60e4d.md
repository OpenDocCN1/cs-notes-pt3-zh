class LinkedGraph
{
public:
    LinkedGraph (const int vertices): e(0)
    {
        if (vertices < 1) throw "Number of vertices must be > 0";
        n = vertices;
        adjLists = new Chain<int>[n];
        Chain<int> c; // set c.first to 0
        fill(adjLists, adjLists+n,c);
    };
private:
    Chain<int>* adjLists;
    int n;
    int e;
};
