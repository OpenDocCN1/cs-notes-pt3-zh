# # Reverse

```
template <class T>
void Chain<T>::Reverse()
{
    // make (a₁,..., aₙ) becomes (aₙ,..., a₁).
    ChainNode<T> *current = first, *previous = 0;
    while (current) {
        ChainNode<T> *r = previous; // r trails previous
        previous = current;
        current = current→link;
        previous→link = r;  //          r       previous  current
    }
    first = previous;
}
```
