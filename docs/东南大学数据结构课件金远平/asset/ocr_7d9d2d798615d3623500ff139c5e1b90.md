int Sets::CollapsingFind (int i )
{ // Find the root of tree containing element i. Use the
  // collapsing rule to collapse all nodes from i to the root.
  for (int r = i; parent[r] >= 0; r = parent[r]); // find the root
  while ( i != r) {
      int s = parent[i];
      parent[i] = r;
      i = s;
  }
  return r;
}
