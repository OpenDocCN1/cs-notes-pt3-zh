class Sets {
public:
    // Set operations
    ...
private:
    int *parent;
    int n; // number of set elements
};

Sets::Sets (int numberOfElements)
{
    if (numberOfElements < 2) throw "Must have at least 2 elements.";
    n=numberOfElements;
    parent=new int[n];
    fill(parent, parent+n, -1);
}
