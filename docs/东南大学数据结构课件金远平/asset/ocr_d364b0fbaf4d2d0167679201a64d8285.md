```
template <class T>
void Bag<T>::Pop() {
    if (isEmpty()) throw "Bag is empty, cannot delete";
    int deletePos = top/2; // note we don’t care which to delete
    copy(array+deletePos+1, array+top+1, array+deletePos);
    array[top--].~T(); // destructor for T
}
```
