//Workhorse– assume to be a friend of GenListNode
template <class T>
bool Equal(GenListNode<T>* s, GenListNode<T>* t)
{
    if ((!s) && (!t)) return true;
    if ( s && t && (s->tag == t->tag) )
        if (s->tag)
            return Equal(s->down, t->down)
                && Equal(s->next, t->next);
        else return (s->data == t->data)
                && Equal(s->next, t->next);
    return false;
}
