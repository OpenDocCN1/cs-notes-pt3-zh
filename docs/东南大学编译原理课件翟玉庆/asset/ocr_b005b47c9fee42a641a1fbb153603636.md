%token INTEGER
%%

S : E    {printf("%d\n", $1);}

E : T    {$$ = $1;}
| E A T  {switch ($2) {
    case '+': $$ = $1 + $3; break;
    case '-': $$ = $1 - $3; break; }
};

T : F    {$$ = $1;}
| T M F  {switch ($2) {
    case '*': $$ = $1 * $3; break;
    case '/': $$ = $1 / $3; break; }
};

F : '(' E ')'    {$$ = $2;}
| INTEGER    {$$ = $1;}
| '-' INTEGER    {$$ = -$2;}
| '-' '(' E ')'  {$$ = -$3;};
```

```
A : '+'    {$$ = '+';}
| '-'    {$$ = '-';};

M : '*'    {$$ = '*';}
| '/'    {$$ = '/';};

%%

main()
{
    if (!yyparse())
        printf("completed parse\n");
    else

}
