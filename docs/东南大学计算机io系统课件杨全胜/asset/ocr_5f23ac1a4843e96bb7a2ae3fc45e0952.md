# 6.5 Application

; initial 8255, Port A mode 0, output, Port C upper is input, lower is output

```
MOV AL, 88H
OUT PORTCN, AL
MOV AL, 01H      ; set PC0, STB invalid
OUT PORTCN, AL
WAIT: IN AL, PORTC ; read status of printer
TEST AL, 80H      ; if "busy", wait
JNZ WAIT
MOV AL, [SI]
CMP AL, '$'
JZ PRINT_OVER
```
