# 事务(Transaction)

```
#include "lrv_custom_body.h"

vuser_init()
{
    lr_start_transaction("homepage");

    web_url("MercuryWebTours",
        "URL=http://localhost/MercuryWebTours/",
        "Resource=0",
        "RecContentType=text/html",
        "Referer=",
        "Snapshot=t2.inf",
        "Mode=HTML",
        LAST);

    lr_end_transaction("homepage", LR_AUTO);

    return 0;
}
```
