若用if 语句实现上述功能，程序为：

```
main()
{ int s;
float p, w, d , f ;
scanf("%%f , %%f , %%d", &p, &w, &s);
if(s>=3000) d=15;
else if (s>=2000&&s<3000) d=10;
else if (s>=1000&&s<2000) d=8;
else if (s>=500&&s<1000) d=5;
else if (s>=250&&s<500) d=2;
else d=0;
f=p*w*s*(1-d/100.0) ;
printf("freight=%%15.4f", f) ;
}
```
