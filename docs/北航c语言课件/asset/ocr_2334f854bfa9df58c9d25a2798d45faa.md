# 程序举例：

```
#include<math.h>
#define PR printf
#define NL "\n"
#define D "%d%d%d"
#define F "%f"
main()
{
    int a,b,c;
    float s,area;
    scanf(D,&a,&b,&c);// scanf("%d%d%d",&a,&b,&c);
    s=0.5*(a+b+c);
    area=sqrt(s*(s-a)*(s-b)*(s-c));
    PR(D NL,a,b,c);// printf("%d%d%d\n", a,b,c);
    PR(F NL,s); // printf("%f\n", s);
    PR(F NL,area); // printf("%f\n", area);
}
```
