main()
{
    int a[7], i , j , t ;
    printf("Input 6 numbers : \n");
    for(i=1 ; i<7 ; i++)
        scanf("%d",&a[i]);
    printf("\n");
    for(j=1 ; j<=5 ; j++) /* 控制5轮排序 */
        for(i=1 ; i<=6-j ; i++) /* 每轮排序比较次数 */
            if(a[i]>a[i+1])
                { t=a[i]; a[i]=a[i+1]; a[i+1]=t; }
    for(i=1 ; i<=6 ; i++)
        printf("%d",a[i]);
}
