# 例：编写能返回结果串地址的串拷贝函数。

```
#include "stdio.h"
main()
{ char *strcpy1(char* str1,char*str2);
    char *ps, s1[80]="Tom is a boy";
    ps=strcpy1(s1,"Jane is a girl");
    puts(ps);
}
char *strcpy1(char *str1,char *str2)
{  char *s=str1;
    while(*str2) *str1++=*str2++;
    *str1='\0';
    return s;
}
```
