main()
{
    int a[6], i , j , t;
    printf("Input 6 numbers : \n");
    for(i=0 ; i<6 ; i++)
        scanf("%d",&a[i]);
    printf("\n");
    for(j=0 ; j<=4 ; j++) /* 控制5轮排序 */
        for(i=0 ; i<5-j ; i++) /* 每轮排序比较次数 */
            if(a[i]>a[i+1])
                { t=a[i]; a[i]=a[i+1]; a[i+1]=t; }
    for(i=0 ; i<=5 ; i++)
        printf("%d",a[i]);
}
