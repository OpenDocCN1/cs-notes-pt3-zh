void sort(char *p[],int n);
main()
{
    char *s[10]={"  DDD "," AAA ","878787"},*ps[10],i;
    for(i=0;i<10;i++) ps[i]=s[i];
    sort(ps,3);//
    for(i=0;i<3;i++) puts(ps[i]);
}
void sort(char *pt[],int n)
{
    int i, j;
    char *s;
    for(i=0;i<n-1;i++)
        for(j=i+1;j<n;j++)
            if(strcmp(pt[i],pt[j])>0)
                s=pt[i], pt[i]=pt[j], pt[j]=s;
}
