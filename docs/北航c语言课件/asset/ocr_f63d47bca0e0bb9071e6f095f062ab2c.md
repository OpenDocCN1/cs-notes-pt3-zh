# 例：通过函数指针调用两个不同函数。

```
#include<math.h>
#define PI 3.1415926
double y(double);
double cos(double);
double table(double (*f)(), double, double, double);

void main()
{
    printf("Table of y(x)=2*x*x-x+1\n");
    table(y,0.0,2.0,0.5);
    printf("Table of cos(x)\n");
    table(cos,0.0,PI,0.5);
}
```
