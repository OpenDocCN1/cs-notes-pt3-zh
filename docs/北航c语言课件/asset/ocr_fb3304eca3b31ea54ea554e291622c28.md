# 例1 把数组a写入文件fa.txt;再从fa读入数组b。

```
#include "stdio.h"
main()
{
    FILE* fp;
    int a[100]={1,5,6,78,21,34,67,87,23},b[100], i ;
    fp=fopen("fa.txt","wb");
    fwrite(a,sizeof(a),1,fp);
    /* fwrite(a,sizeof(int),100,fp);*/
    fclose(fp);
    fp=fopen("fa.txt","rb");
    fread(b,sizeof(a),1,fp);
    for(puts(""),i=0;i<10;i++) printf("%6d",b[i]);
    fclose(fp);
}
```
