# 有关的矩阵运算

# 1. 数乘矩阵

tA = [ t a₁₁  t a₁₂  …  t a₁ₙ ]
      [ t a₂₁  t a₂₂  …  t a₂ₙ ]
      [ …     …     …   …   ]
      [ t aₘ₁  t aₘ₂  …  t aₘₙ ]

# 2. 矩阵的加法运算

A + B = [ a₁₁ + b₁₁  a₁₂ + b₁₂  …  a₁ₙ + b₁ₙ ]
         [ a₂₁ + b₂₁  a₂₂ + b₂₂  …  a₂ₙ + b₂ₙ ]
         [ …        …        …   …        ]
         [ aₘ₁ + bₘ₁  aₘ₂ + bₘ₂  …  aₘₙ + bₘₙ ]
