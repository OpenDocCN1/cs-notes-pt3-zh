# 采用管程解决生产者-消费者问题

```
Monitor PC {
    condition full, empty;
    int buffer[N];
    int pnum = N;
    int cnum = 0;

    public producer {
        if( pnum > 0 )
            pnum--;
        else
            full.wait;
        one >> buffer;
        cnum++;
        if (cnum <= 0)
            empty.signal;
    }

    public consumer {
        if( cnum > 0 )
            cnum--;
        else
            empty.wait;
        one >> buffer;
        pnum++;
        if (pnum <= 0)
            full.signal;
    }
}
```
