# 算法

```
procedure INSERTS( S, T, i )
    if (T=nil) then
        return
    if (S=nil) then
        [  S←T
           return  ]
    if (i<0) then
        [  call ERROR(“i error!”)
           return  ]
    p←S
    for j←1 to i−1 do
        p←link(p)
    end
    q←link(p)
```

// p 指向 S 的第i个字符 //
// q 指向 S 的第i+1个字符 //
