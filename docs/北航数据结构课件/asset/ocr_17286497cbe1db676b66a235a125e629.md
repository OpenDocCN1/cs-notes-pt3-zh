# 算法

```
procedure BUBBLESORT( K, n )
    flag←1
    i←n-1
    while (i>0 and flag=1) do
        flag←0
        for j←1 to i do
            if (K[j]>K[j+1]) then
            [  temp←K[j]
            K[j]←K[j+1]
            K[j+1]←temp      // 交换两个元素的位置 //
            flag←1  ]
        end
        i←i-1
    end
end
```
