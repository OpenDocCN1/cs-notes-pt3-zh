```
procedure JOSEPHU(n, m, k)
    list←nil
    for i←1 to n do
        call GETNODE(p)
        data(p)←i
        if (list=nil) then
            list←p
        else
            link(r)←p
            r←p
        end
    link(p)←list    // 建立循环链表 //
    p←list
    for i←1 to k-1 do
        r←p
        p←link(p)
    end    // 找到第一个出发点 //
```
