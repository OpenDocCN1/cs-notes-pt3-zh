procedure MBSEARCH(T, k)
  ptr←T
  qtr←nil
  A[0]←-max
  while (ptr≠nil) do
    n ←keynum(ptr)
    A[n+1]←+max
    for j←1 to n do
      A[j]←key(ptr,j)
    end
    for j←0 to n do
      B[j]←pointer(ptr, j)
    end
    i←SEQSEARCH(A, k)
    if (key(ptr,i)=k) then
      return(ptr, i, 1)
    qtr←ptr
    ptr←B[i]
  end
  return(qtr, i, 0)
end
```

```
function SEQSEARCH(A,k)
  j←1
  loop
    if (k<A[j]) then
      return(j-1)
    j←j+1
  forever
end
