```
procedure CHANGE( V,A,n,G )
    for i ← 1 to n do
        vertex[i] ← V[i]
        link[i] ← nil
        for j ← 1 to n do
            if ( A[i,j] ≠ 0 ) then
                [   call GETNODE(p)
                    adjvex(p) ← j
                    next(p) ← nil
                    if ( link[i] = nil ) then
                        link[i] ← p
                    else
                        next(r) ← p
                    r ← p   ]
            end
        end
    end
```
