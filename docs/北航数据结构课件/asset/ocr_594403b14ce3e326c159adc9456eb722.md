procedure DEL( list )
    p←link(list)
    r←list
    while (p≠nil) do
        q←list
        while (q≠p) do
            if (data(q)=data(p)) then
                [ link(r)←link(p)
                  call RET(p)
                  p←link(r)
                  q←list ]
            else
                q←link(q)
        end
        r←p
        p←link(p)
    end
end
