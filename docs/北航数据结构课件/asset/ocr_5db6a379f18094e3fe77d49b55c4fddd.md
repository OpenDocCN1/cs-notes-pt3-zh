```
prodecure INSERTLIST( A, n, i, item )
    if (n=M) then
        call ERROR('The space is full!')
    else
        if (i<1) or (i>n+1) then
            call ERROR('Position is not exist ! ')
        else
            [ for j←n downto i do
                A[j+1]←A[j]
            end
            // 元素依次后移一个位置 //
            A[i]←item
            // 将item插入表的第i个位置 //
            n←n+1
            // 线性表的长度加1 //
            ]
end
```
