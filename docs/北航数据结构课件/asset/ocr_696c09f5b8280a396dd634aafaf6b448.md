procedure DEL(T, item)
if (T=nil) then
  return
top←0
p←T
repeat
  while (p≠nil) do
    call VISIT(p)
    if (top=M) then
      call ERROR('overflow!')
      return
    top←top+1
    STACK[top]←p
    q←p
    p←lchild(p)
  end
  p←STACK[top]
  top←top-1
  q←p
  p←rchild(p)
until (p=nil and top=0)
end
