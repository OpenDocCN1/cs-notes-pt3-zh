```
JOSEPHU( int n;  int k;  int m )
{
    struct node {
        int data;
        struct node  *link;
    } *list, *p, *r;
    int  i;
    list=NULL;
    for( i=1;  i<=n;  i++ )
    {
        p=( struct node*)malloc(sizeof(struct node));
        p->data=i;
        if ( list==NULL )
            list=p;
        else
            r->link=p;
        r=p;
    }
    p->link=list;       // 到此，建立一个循环链表 //
```
