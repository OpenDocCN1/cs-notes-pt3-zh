procedure QUICKSORT( K, s, t )
    if (s<t) then
        [ j←s
          temp← K[s]
          for i←s+1 to t do
              if ( K[i]<temp ) then
                  [ j←j+1
                    SWAP( K[i], K[j] ) ]
              end
          SWAP( K[s], K[j] )
          call QUICKSORT( K, s, j-1 )
          call QUICKSORT( K, j+1, t ) ]
    end
