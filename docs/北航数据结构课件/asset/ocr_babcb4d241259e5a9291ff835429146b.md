# 链式存储结构的堆栈？！

```
procedure CHANGE( num )
    top←nil
    while (num ≠ 0) do
        call GETNODE(p)
        data(p)← num MOD 8
        link(p)←top
        top←p
        num←⌊num/8⌋
    end
    while (top ≠ nil) do
        print( data(top) )
        p←top
        top←link(top)
        call RET(p)
    end
end
```
