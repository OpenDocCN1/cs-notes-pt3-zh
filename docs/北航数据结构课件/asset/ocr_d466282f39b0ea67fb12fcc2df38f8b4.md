procedure DEGREE( VERTEX, LMTA, n, e )
    for k←1 to n do                //统计第k个顶点的度//
        d←0                         //置第i个顶点的度的初值为0//
        for i←2 to e+1 do
            if (LMTA[i,1]=k) then
                d←d+1               //统计第i行非零元素的个数//
            if (LMTA[i,2]=k) then
                d←d+1               //统计第i列非零元素的个数//
        end
        print(VERTEX[k],’DEGREE=’,d)
    end
end
