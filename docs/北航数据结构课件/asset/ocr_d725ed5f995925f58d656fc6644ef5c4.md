```
procedure DELETELINK2(list, q)
if (q=list) then        // 当删除链表第一个结点 //
    list←link(q)
else
[   r←list
    while ( link(r)≠q and link(r)≠nil ) do
        r←link(r)                // 移向下一个链结点 //
    end                        // 寻找q结点的直接前驱r //
    if (link(r)=nil) then
    [   print(' There is no this node !')
        return  ]
    else
        link(r)←link(q)  ]
call RET(q)                // 释放被删除结点的空间 //
end
```
