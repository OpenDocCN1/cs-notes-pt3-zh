# 递归算法

```
function SORTSEARCH1( T, key )
    if (T≠nil) then
    [ if (data(T)=key) then
        return(T)                // 查找成功 //
      if (data(T)<key) then       // 查找T的右子树 //
        return( SORTSEARCH1( rchild(T), key) )
      else
        return( SORTSEARCH1( lchild(T), key) ) ]
                              // 查找T的左子树 //
    else
      return(nil)                // 查找失败 //
end
```
