
## 8. 从非空线性链表中删除q所指的链结点, 设q的直接前驱结点由r指出

```
r←link(r)

r←list
while link(r)≠q do
r←link(r)
end
```

list r q

list r

### 算法

```
procedure DELETELINK2(list, q)
if (q=list) then        // 当删除链表第一个结点 //
    list←link(q)
else
[   r←list
    while ( link(r)≠q and link(r)≠nil ) do
        r←link(r)                // 移向下一个链结点 //
    end                        // 寻找q结点的直接前驱r //
    if (link(r)=nil) then
    [   print(' There is no this node !')
        return  ]
    else
        link(r)←link(q)  ]
call RET(q)                // 释放被删除结点的空间 //
end
```

### 建立一个无头链表

```
procedure CREATE(A, n, list)
    list←nil                         //创建一个空链表//
    for i←1 to n do
        read(a)                       //取一个数据元素//
        call GETNODE(p)               //申请一个新结点//
        data(p)←a
        link(p)←nil
        if (list=nil) then
            list←p
        else
            link(r)←p                  //将新结点连接在链表尾部//
        r←p
    end
end
```

### 算法

```
procedure JOSEPHU(n, m, k)
    list←nil
    for i←1 to n do
        call GETNODE(p)
        data(p)←i
        if (list=nil) then
            list←p
        else
            link(r)←p
            r←p
        end
    link(p)←list    // 建立循环链表 //
    p←list
    for i←1 to k-1 do
        r←p
        p←link(p)
    end    // 找到第一个出发点 //
    while link(p)≠p do
        for i←1 to m-1 do
            r←p
            p←link(p)
        end
        link(r)←link(p)
        print(data(p))
        call RET(p)
        p←link(r)
    end
    print( data(p) )
end
```

| SPARKS | Pascal 语言 | C 语言 |
|---|---|---|
| `list←nil` | `list:=NIL;` | `list=NULL;` |
| `link(r)←link(p)` | `r↑link:=p↑link;` | `r->link=p->link;` |
| `for i←1 to n do`<br>`......`<br>`end` | `for i:=1 to n do`<br>&nbsp;&nbsp;&nbsp;&nbsp;`begin`<br>&nbsp;&nbsp;&nbsp;&nbsp;`......`<br>&nbsp;&nbsp;&nbsp;&nbsp;`end;` | `for ( i=1; i<=n; i++ )`<br>`{`<br>&nbsp;&nbsp;&nbsp;&nbsp;`......`<br>`}` |
| `print(data(p))` | `write(p↑data);` | `printf("%d", p->data);` |

| `call RET(p)` | `dispose(p);` | `free(p);` |

```c
JOSEPHU( int n, int k, int m )
{
    struct node {
        int data;
        struct node  *link;
    } *list, *p, *r;
    int  i;
    list=NULL;
    for( i=1;  i<=n;  i++ )
    {
        p=( struct node*)malloc(sizeof(struct node));
        p->data=i;
        if ( list==NULL )
            list=p;
        else
            r->link=p;
        r=p;
    }
    p->link=list;       // 到此，建立一个循环链表 //
```
