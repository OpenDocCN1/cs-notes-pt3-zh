```
function HASHSEARCH(HT,M,key)
i←H(key)
D←i
while (HT[D]≠空 and HT[D]≠key) do
    D←(D+1) MOD M
    if (D=i) then        // 转了一圈回到开始点（表满）//
        return (−D)
    end
    if (HT[D]=key) then
        return(D)        // 查找成功 //
    else
        return(−D)       // 查找失败 //
    end
```

```
procedure INSERTSORT( K, n )
    for i←2 to n do
        temp←K[i]
        j←i−1
        while (j>0 and temp<K[j]) do
            K[j+1]←K[j]
            j←j−1
        end
        K[j+1]←temp
    end
end
```

O(n²)

```
procedure BININSERTSORT( K, n )
  for i←2 to n do
    temp←K[i]
    low←1
    high←i-1
    while (low≤high) do
      mid←⌊(low+high)/2⌋
      if (temp<K[mid]) then
        high←mid-1
      else
        low←mid+1
    end
    for j←i-1 downto low do
      K[j+1]←K[j]
    end
    K[low]←temp
  end
end
```
