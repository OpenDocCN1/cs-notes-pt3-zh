# Compiler

<语句>:: = <变量>:=<表达式>
|IF<表达式>THEN<语句>[ELSE <语句>]

<变量>:: = i['<表达式>']

```
void statement()
{
    if (sym == " IF ") {
        getsym();
        expr();
        if (sym != " THEN")
            error();
        else { getsym();
            statement();
            if (sym == "ELSE") {
                getsym();
                statment();
            }
        }
    }
    else { var();
        if (sym != ":=")
            error();
        else { getsym();
            expr();
        }
    }
}
```

```
void var ()
{
    if (sym != "i") error();
    else { getsym();
        if (sym == "[") {
            getsym();
            expr();
            if (sym != "]")
                error();
            else getsym();
        }
    }
}
```

<表达式>:: = <项>{+<项>}

```
void expr ()
{
    term();
    while (sym == "+") {
        getsym();
        term();
    }
}
```
