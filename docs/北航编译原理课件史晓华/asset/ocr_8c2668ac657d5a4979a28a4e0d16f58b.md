# 例：

```
(1) prod := 0
(2) i := 1
```

```
(3) t1 := 4 * i
(4) t2 := a [ t1 ]
(5) t3 := 4 * i
(6) t4 := b [ t3 ]
(7) t5 := t2 * t4
(8) t6 := prod + t5
(9) prod := t6
(10) t7 := i + 1
(11) i := t7
(12) if i <= 20 goto B1
```

+   - RISC: ARM
+     - prod = R5
+     - t5 = [SP+8]
+     - t6 = R2

```
ldr R3, [SP, #8]  ; R3 = t5
add R5, R2, R3   ; prod = t6 + R3
```

+   - CISC: X86
+     - prod = EBX
+     - t5 = [ESP+8]
+     - t6 = ECX

```
mov ECX, EBX   ; t6 = prod
add ECX, [ESP+8] ; t6 = prod + t5
mov EBX, ECX    ; prod = t6
```
