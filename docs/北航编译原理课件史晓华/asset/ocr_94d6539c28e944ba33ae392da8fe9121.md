array B ( N, -2: 1 ) char ;

# 以前面所举的二维数组B为例，若N = 3

P(2) = 1

P(1) = [U(2) - L(2) +1]
    = 1 - (-2) +1
    = 4

RC = - ∑_{i=1}^{2} L(i) P(i)
    = -[ 1 × 4 + (-2) × 1 ]
    = -2
