# 12.2.1 程序地址空间的实例分析

```
// C12P1.cpp
#include "stdafx.h"
int global_val = 0;
int foo(int n){
    static int static_val = 0;
    int i;
    for(i=0; i<100; i++){
        n = n * i + global_val + static_val;
    }
    static_val = n / 2;
    printf("static_val is %x\n", static_val);
    return n;
}
int main(int argc, char* argv[]){
    global_val = 99;
    int n = foo(100);
    printf("foo(100) is %x\n",n);
    return 0;
}
```
