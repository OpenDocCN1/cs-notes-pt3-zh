```
procedure chkty​pe(t, i, m, z);
  string t; integer m, i, z;
  if z < 1
  then begin
    error('实参数大于形参数', symtbl [i].name, statno);
    return (z);
  end
  m := m+1;        /* 实参计数 */
  if t ≠ symtbl [i+m].type
  then error('实参和形参类型不匹配', symtbl [i+m].name, statno);
  z := z-1;        /* 减去已匹配的形参数 */
  return (z);      /* 剩下待匹配的形参数 */
end;
```
