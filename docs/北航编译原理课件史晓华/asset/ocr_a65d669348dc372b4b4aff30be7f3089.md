# 3、词法分析程序算法

```
START: TOKEN := '';  /*置TOKEN为空串*/
        GETCHAR; GETNBC;
CASE CHAR OF
'A'..'Z': BEGIN
    WHILE ISLETTER OR ISDIGIT DO
        BEGIN CAT; GETCHAR END;
    UNGETCH;
    C:= RESERVE;   /* 返回0，为标识符 */
    IF C=0 THEN RETURN('IDSY':TOKEN)
    ELSE RETURN (C,-)   /*C为保留字编码*/
END;
'0'..'9': BEGIN
    WHILE DIGIT DO
        BEGIN CAT; GETCHAR END;
    UNGETCH;
    RETURN ('INTSY',ATOI)
END;
'+': RETURN('PLUSS Y',-);
```
