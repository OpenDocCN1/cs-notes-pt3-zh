main()
{
    FILE *fpr,*fpw;
    int j;
    clrscr();
    fpr=fopen("d:\\creatr.txt","r");
    fpw=fopen("d:\\creatw.txt","w");
    if(fpr==NULL || fpw==NULL)
    {printf("can not open file\n");
      exit(0);
    }

    for(j=0;j<number;j++)
        fscanf(fpr,"%ld,%s,%d",&test[j].num, test[j].name,&test[j].age);
    for(j=0;j<number;j++)
        fprintf(fpw,"%ld,%s,%d\n",test[j].num,test[j].name,test[j].age);
}
