# 算法描述

```
Procedure pnarc(r,xc,yc: integer);
  var x,y,f: integer;
  begin
    x:=xc; y:=yc+r; f:=0;
    while y>yc do {        //画1/4圆弧
      plot(x,y);
      if f>0 then { f:=f-2*(y-yc)+1;
                    y:=y-1 }    //向圆内走
      else { f:=f+2*(x-xc)+1;
             x:=x+1 }    //向圆外走
    }
    if (y=yc) plot(x,y)
  end
```
