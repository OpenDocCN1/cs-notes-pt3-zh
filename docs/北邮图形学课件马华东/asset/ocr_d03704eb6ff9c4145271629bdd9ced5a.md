Procedure intersection (ray, scene)
begin
    if (ray与scene的包围盒有交点) then
    begin
        if (scene是终结点) then
        begin
            ray与scene求交;
            若有交,则交点置入交点表中
        end
        else for (scene的所有孩子child_of_scene) do
            intersection (ray, child_of_scene)
    end
end
