# The Readers-Writers Problem(Cont.)

+   - Shared data
+     semaphore mutex, wrt;

+   - Initially
+     mutex = 1
+     wrt = 1
+     readcount = 0

+   - Writer process
+     wait(wrt);
+     ...
+     writing is performed
+     ...
+     signal(wrt);

+   - Reader process
+     wait(mutex);
+     readcount++;
+     if (readcount == 1)
+       wait(wrt);
+     signal(mutex);
+     ...
+     reading is performed
+     ...
+     wait(mutex);
+     readcount--;
+     if (readcount == 0)
+       signal(wrt);
+     signal(mutex);
