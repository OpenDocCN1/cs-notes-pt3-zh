int index(char const *t, char const *p) //蛮力搜索
{
    int i=1; j=1; //假设字符数组的下标从1算起
    int n=strlen(t); m=strlen(p);
    while ((i<=n) && (j<=m))
    {
        if (t[i]==p[j])
        {
            i=i+1;
            j=j+1;
        }
        else
        {
            i=i-j+2;
            j=1;
        }
    }
    if (j==m+1)
        return (i-m);
    else
        return(0);
}

> 最坏情况字符比较次数为：(n-m+1)*m

> 算法复杂度：O(m*n)
