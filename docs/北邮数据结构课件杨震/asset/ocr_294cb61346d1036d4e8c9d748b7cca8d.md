# 顺序表的存储管理

```
template<typename T>
Struct Vector {
    void reserve(int n)
    {
        if (n>capacity)
        {
            elem=(T*)realloc(elem,n*sizeof(T));
            uninitialized_fill(elem+size,elem+n,T());
            capacity=n;
        }
    }
    protected:
    bool full(void) const
    {      return size==capacity; }
    int next_capacity(void) const
    {
        return capacity<= 0 ?1:2*capacity;
    }
}
```
