# 败者树的程序实现

```
typedef int LoserTree[k];

typedef struct { KeyType key;
                 } Exnode, External[k+1];
```

```
Void K_Merge ( LoserTree &ls,  External & b)
{
  for ( i = 0; i < k; ++i) input(b[i].key);

  CreateLoserTree(s);

  { while ( b[ls[0]].key != MAXKEY)
      { q = ls[0]; output(q); input(b[q].key); Adjust( ls, q ); }
      output( ls[0]);
  }
} // K_Merge
```
