    InitQueue (Q); EnQueue(Q,v); //先置空队列，然后第一个顶点v入队
    while (!QueueEmpty(Q))
    {
        v=DeQueue (Q); p=G[v].firstarc; //g是全局变量，v入队列。
        while (p!=null)
        {
            if(visited[p->adjvex]==false)
            {
                printf("%3d",p->adjvex); visited[p->adjvex]=true;
                EnQueue(Q,p->adjvex);}
            p=p->next;
        }
    }
}
