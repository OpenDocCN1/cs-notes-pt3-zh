template <class Type>
void Push( const int i, const Type & item ) {
    if ( t[i] == b[i+1] ) StackFull(i);
    else V[++t[i]] = item;      //第i 个栈进栈
}

template <class Type> Type *Pop( const i ) {
    if ( t[i] == b[i] )
        { StackEmpty(i); return 0; }
    else return & V[t[i]--];    //第i 个栈出栈
}
