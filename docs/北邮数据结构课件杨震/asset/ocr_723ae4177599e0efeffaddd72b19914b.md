```
void fact(int n, int & f)
{
    init_stack(s);

    { 1: top=top+1;
        s[top].rd=2;
        s[top].n=n;
        s[top].f=f;
        n=n-1; goto 0;
        2: f=n*f }
    if (top!=0)
    {  s[top].f=f;
        top=top-1;
        n=s[top+1].n;
        f=s[top+1].f;
        goto s[top+1].rd; }
}
```

```
void fact(int n,int &f)
{
    init_stack(s);
    while (n!=0)
    {  top=top+1;
        s[top].n=n;
        n=n-1;}
    f=1;
    while (top!=0)
    {  top=top-1;
        n=s[top+1].n;
        f=n*f; }
}
```

注：此时可确定栈中只存放n值即可。
