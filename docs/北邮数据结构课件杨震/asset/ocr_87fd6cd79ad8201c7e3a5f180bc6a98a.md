void mergesort ( SqList &r) {
    len = 1;
    While (len < n)
    {
        s1 = 1;    q = 1;
        While (s1 + len <= n)
        {
            s2 = s1 + len;    t1 = s2 - 1;
            t2 = s2 + len - 1;
            if (t2 > n) t2 = n;
            merge (r, t, s1, t1, s2, t2, q);
            s1 = q;
        }
        for (i = 1; i <= q - 1) r[i] = t[i];
        len = len*2;
    }
}
