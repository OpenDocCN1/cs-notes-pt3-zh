void SpnTree (AdjList g)
{
    typedef struct {int i,j,w}EDGE; //该顶点信息就是顶点编号，权是整数
    EDGE edge[];
    scanf("%d%d",&e,&n); //输入边数和顶点数。
    for (i=1;i<=e;i++)    //输入e条边：顶点，权值。
        scanf("%d%d%d%d", &edge[i].i, &edge[i].j, &edge[i].w);
    for (i=2;i<=e;i++)    //按边上的权值大小，对边进行逆序排序。
    {
        edge[0]=edge[i]; j=i-1;
        while (edge[j].w<edge[0].w)
            edge[j+1]=edge[j--];
        edge[j+1]=edge[0];
    }
    k=1; eg=e;
    while (eg>=n)    //破圈，直到边数e=n-1。
    {
        if (connect(k)) //删除第k条边若仍连通。
        {
            edge[k].w=0; eg--; //测试下一条边edge[k]，权值置0表示该边被删除
        }
        k++; //下条边
    }
}
