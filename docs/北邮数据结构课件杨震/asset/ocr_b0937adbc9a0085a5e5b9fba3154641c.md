{char data; struct node *lchild,*rchild;}*bitree;

{
    bitree p; p=bt; initstack(s);  /*初始化栈s为空栈*/
    while(p || !empty(s))         /*栈s不为空*/
        if(p)
        { push (s,p); (1)____; }   /*P入栈*/

        { p=pop(s); printf("%c",p->data); (2)____; } /*栈顶元素出栈*/
}
