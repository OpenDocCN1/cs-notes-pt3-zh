# [在二叉树中查找结点值为x的结点]

```
void pre_find(BiTree bt, ElemType x, BiTree &q)//用q返回
{ //F是全局bool型变量，初值为FALSE

    if (bt && F==FALSE)
    {
        if (bt->data==x)
        { q=bt; F=TRUE;}
        else
        { pre_find(bt->lchild, x, q);
          pre_find(bt->rchild, x, q)
        }
    }
}
```
