# 顺序表的其他成员函数

```
template<typename T>
void Vector<T>::resize(int n,T const &t=T())
{
    reverse(n);
    if (size<n)
        uninitialized_fill(elem+size,elem+n,T);
    size=n;
}
template<typename T>
void swap(Vector<T>& a, Vector<T>& b)
{
    std::swap(a.elem,b.elem);
    std::swap(a.size,b.size);
    std::swap(a.capacity,b.capacity);
}
```
