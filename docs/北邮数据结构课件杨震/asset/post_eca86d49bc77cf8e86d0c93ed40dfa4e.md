Link LinkListSort(Link list)
{
    p=list->next;  // p是工作指针，指向待排序的当前元素。
    list->next=NULL; // 假定第一个元素有序，即链表中现只有一个结点。
    while(p!=null)
    {
        r=p->next;   // r是p的后继。
        q=list;
        if(q->data>p->data) // 处理待排序结点p比第一个元素结点小的情况。
        {  p->next=list;
           list=p; // 链表指针指向最小元素。
        }
        else // 查找元素值最小的结点。
        {  while(q->next!=null && q->next->data<p->data)
               q=q->next;
           p->next=q->next; // 将当前排序结点链入有序链表中。
           q->next=p; }
        p=r; // p指向下一个待排序结点。
    }
}
