//合并算法
void Merge( Type c[], Type d[], int l, int m, int r)
{
    int i = l, j = m+1, k = l;
    while(( i <= m ) && ( j <= r ))
    {
        if ( c[i] <= c [j] ) d[k++] = c[i++];
        else d[k++] = c [j++];
    }
    if ( i > m )
    {
        for (int q = j; q <= r; q++)
            d[k++] = c[q];
    }
    else    for ( int q = i; q <= m; q++ )
        d[k++] = c[q];
}
