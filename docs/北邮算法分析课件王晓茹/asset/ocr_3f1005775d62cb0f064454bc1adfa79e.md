```
int BinarySearch ( const ElementType A[],
                    ElementType X, int N )
{
    int Low, Mid, High;
    /* 1 */ Low = 0; High = N - 1;
    /* 2 */ while ( Low <= High ) {
    /* 3 */     Mid = ( Low + High ) / 2;
    /* 4 */     if ( A[ Mid ] < X )
    /* 5 */         Low = Mid + 1;
    else
    /* 6 */         if ( A[ Mid ] > X )
    /* 7 */             High = Mid - 1;
    else
    /* 8 */         return Mid; /* Found */
    } /* end while */
    /* 9 */ return NotFound; /* NotFound is defined as -1 */
}
    T_worst( N ) = O( log N )
```

Very useful in case the data are static and is in sorted order (e.g. find words from a dictionary).
