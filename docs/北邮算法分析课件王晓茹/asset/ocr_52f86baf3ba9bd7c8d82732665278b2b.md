template < class Type >
void Loading< Type > :: Backtrack( int i )
{
    if (i > n)
    {
        if (cw > bestw)
        {
            for(j =1; j<=n;j++)
                bestx[j] =x[j];
            bestw = cw;
        }
        return;
    }
    //搜索子树
    r -= w[i];
    if ( cw + w[i] <= c)
    {//x[i] = 1; //二叉树的左孩子
        x[i] =1;
        cw+= w[i];
        Backtrack(i+1);
        cw -= w[i]; //回溯
    }
    if( cw+r > bestw)
    {
        x[i] = 0;
        Backtrack(i+1); //x[i] =0;//二叉树的右孩子
    }
    r += w[i];
}
