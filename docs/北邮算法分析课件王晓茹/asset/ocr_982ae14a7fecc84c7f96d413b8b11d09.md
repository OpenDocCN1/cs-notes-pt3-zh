//mergepass
void MergePass ( Type x[], Type y[], int s, int n )
{
    int i = 0;
    while ( i <= n - 2 * s )
    {//合并大小为s的相邻2段子数组
        Merge( x, y, i, i+s-1, i+2*s-1 );
        i = i + 2 * s;
    }
    //剩下的元素个数少于2s
    if ( i + s < n ) Merge ( x, y, i, i+s-1, n-1 );
    else for (int j = i; j <= n-1; j++)
        y[j] = x[j];
}
