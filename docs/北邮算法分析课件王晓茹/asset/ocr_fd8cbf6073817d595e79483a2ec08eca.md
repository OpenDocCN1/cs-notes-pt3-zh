bool Queen::Place(int k)
{
    for (int j=1;j<k;j++)
        if ( (abs(k-j)==abs(x[j]-x[k])) || (x[j]==x[k]) ) return false;
    return true;
}

void Queen::Backtrack(int t)
{
    if (t>n) sum++;
    else
        for (int i=1;i<=n;i++) {
            x[t]=i;
            if (Place(t)) Backtrack(t+1);
        }
}
