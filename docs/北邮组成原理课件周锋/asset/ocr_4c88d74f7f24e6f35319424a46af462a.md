# 文件开启+模式注意-写操作无效

```
/*测试读操作后能否直接跟写操作 2011.10.5*/
#include<stdio.h>
int main(void)
{
    int ch;
    FILE *fp;
    if((fp=fopen("test.txt","r+"))==NULL)
    {
        printf("can not open file\n");
        exit(0);
    }
    fseek(fp,1L,0); //将fp移动到距文件首1字节的位置
    ch=fgetc(fp); printf("%c\n",ch);
    //fseek(fp,1L,0); //注释这条读写切换的定位语句导致写操作无效
    fputs("test",fp);
    ch=fgetc(fp);  //写操作后面紧跟着读操作
    printf("%c\n",ch);
    fclose(fp);
}
```
