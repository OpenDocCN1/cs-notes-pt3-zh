# 读写一个数据块

```
```c
int fwrite(void *buffer, int size, int count, FILE *stream);

struct student stu[3], aStu;

//把数组stu中所有元素一次性写入文件中
fwrite(stu, sizeof(struct student), 3, stream);

//把变量aStu中内容写入文件中
fwrite(&aStu, sizeof(struct student), 1, stream);

//读取一个字节写入字符变量ch中
fwrite(&ch, sizeof(char), 1, stream);
