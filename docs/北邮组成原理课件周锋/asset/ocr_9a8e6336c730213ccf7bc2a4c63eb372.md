# 文件以文本/二进制打开区别2-测试程序

```
#include<stdio.h>
#include<stdlib.h>

int main(void)
{
    FILE *fp;
    int ch;
    if((fp=fopen("test","rb"))==NULL)
    {
        printf("can not open file\n");
        exit(0);
    }
    ch=fgetc(fp);
    while(feof(fp)==0)
    {
        printf("%02X\n",ch);
        ch=fgetc(fp);
    }
    fclose(fp);
    return 0;
}
```
