# 举例　基本块划分：

+   →
+   (1)　i:=n-1
+   (2)　j:=n
+   (3)　t₁:=4*n
+   (4)　v:=a[t₁]
+   
+   B₁

+   →
+   (5)　i:=i+1
+   (6)　t₂:=4*i
+   (7)　t₃:=a[t₂]
+   (8)　if t₃<v goto (5)
+   
+   B₂

+   →
+   (9)　j:=j-1
+   (10)　t₄:=4*j
+   (11)　t₅:=a[t₄]
+   (12)　if t₅>v goto (9)
+   
+   B₃

+   →
+   (13)　if i>=j goto (23)
+   
+   B₄

+   →
+   (14)　t₆:=4*i
+   (15)　x:=a[t₆]
+   (16)　t₇:=4*i
+   (17)　t₈:=4*j
+   (18)　t₉:=a[t₈]
+   (19)　a[t₇]:=t₉
+   (20)　t₁₀:=4*j
+   (21)　a[t₁₀]:=x
+   (22)　goto (5)
+   
+   B₅

+   →
+   (23)　t₁₁:=4*i
+   (24)　x:=a[t₁₁]
+   (25)　t₁₂:=4*i
+   (26)　t₁₃:=4*n
+   (27)　t₁₄:=a[t₁₃]
+   (28)　a[t₁₂]:=t₁₄
+   (29)　t₁₅:=4*n
+   (30)　a[t₁₅]:=x
+   
+   B₆
