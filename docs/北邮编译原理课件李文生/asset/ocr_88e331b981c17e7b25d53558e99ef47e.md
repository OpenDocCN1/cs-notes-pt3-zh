+   ■ **对应语法树的代码**
+   
+   `t₁ := -c`
+   
+   `t₂ := b*t₁`
+   
+   `t₃ := -c`
+   
+   `t₄ := b*t₃`
+   
+   `t₅ := t₂+t₄`
+   
+   `a := t₅`

+   ▪ **对应 DAG 的代码**
+   
+   `t₁ := -c`
+   
+   `t₂ := b*t₁`
+   
+   `t₅ := t₂+t₂`
+   
+   `a := t₅`
