（考虑类型）

```
S→id:=E { p=lookup(id.name);
    if (p!=nil) {
        t=gettype(p);
        if (t==E.type) {
            emit(p ':=' E.place);
            S.type=void;
        };
        else if (E.type 可转换为t) {
            u=newtemp;
            emit(u ':=' 类型转换符 E.place);
            emit(p ':=' u);
            S.type=void;
        }
        else S.type=type_error;
    };
    else error(); }
```
