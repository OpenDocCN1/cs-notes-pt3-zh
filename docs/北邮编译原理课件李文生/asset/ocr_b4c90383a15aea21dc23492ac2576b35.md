# 用C++语言描述的对n个数按递增顺序排序的多态函数

```
#include "iostream.h"
/* 定义函数模板 */
template <class T>
void sort (T a[], int n) {
int i, j, k;
T x;
for(i=0; i<n-1; i++) {
    k=i;
    for(j=i+1; j<n; j++)
        if (a[k]>a[j]) k=j;
    if (i!=k) { x=a[i]; a[i]=a[k];
    a[k]=x; }
}
}
```

```
void main()
{
int i, a[8]={4, 6, 1, 3, 8, 2, 7, 5};
double d[5]={5.5, 8.8, 2.2, 7.7, 3.3};
/* 对int型数组排序 */
sort(a, 8);
for (i=0; i<10; i++) cout << a[i] <<endl;
/* 对double型数组排序 */
sort(d, 5);
for (i=0; i<5; i++) cout << d[i] <<endl;
}
```
