| 0id5 | +id*id$ | reduce by F→id | goto[0,F]=3 |
| 0F3 | +id*id$ | reduce by T→F | goto[0,T]=2 |
| 0T2 | +id*id$ | reduce by E→T | goto[0,E]=1 |
| 0E1 | +id*id$ | shift 6 |  |
| 0E1+6 | id*id$ | shift 5 |  |
| 0E1+6id5 | *id$ | reduce by F→id | goto[6,F]=3 |
| 0E1+6F3 | *id$ | reduce by T→F | goto[6,T]=9 |

| 0E1+6T9*7 | id$ | shift 5 |  |

| 0E1+6T9*7F10 | $ | reduce by T→T*F | goto[6,T]=9 |
| 0E1+6T9 | $ | reduce by E→E+T | goto[0,E]=1 |
