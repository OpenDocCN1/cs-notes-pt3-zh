void swap(int *xx, int *yy){
    int temp;
    temp=*xx;
    *xx=*yy;
    *yy=temp;
}
void main(){
    int x,y;
    int *px,*py;
    scanf("%d %d", &x,&y);
    px=&x;
    py=&y;
    if(x<y)
        swap(px,py);
    printf("\n%d\t%d\n",x,y);
}
