int half_search (int n, int a [], int key) { /* key为检索关键字 */
int lower,upper,j;
lower = 0 ;
upper = n-1 ;
while ( upper-lower>=0 ) {
 j = ( lower+upper ) / 2 ; /* 两分 */
if ( key==a[j] )
 return j; /* 已经找到，位置为 j */
else
 if ( key > a[j] )
 lower=j+1 ; /* key在a[j+1]与a[upper]之间 */
else
 upper=j-1 ; /* key在a[lower]与a[j-1]之间 */
}
return -1 ;
}
