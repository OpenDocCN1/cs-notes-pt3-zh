void code(int n, int A[]){
    int i,j,k,t;
    bool flag;
    for(i=0;i<n;i++){
        t=1;
        for(j=0;j<n;j++){
            flag=false;
            for(k=0;k<j;k++)
                if(A[k]==A[j])
                    flag=true;
            if(A[i]>A[j]&&!flag)
                t++;
        }
    }
    c[i]=t;
}
