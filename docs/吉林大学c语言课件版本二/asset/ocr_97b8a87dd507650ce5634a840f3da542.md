#define L 100
void main(){
    char x[L];
    int i,j,k;
    int n=0;
    scanf("%s",x);
    while(x[n]!='\0')    n++;
    for(i=n-1;i>=0;i--){
        printf("%s\n",x);
        for(j=0;j<i;j++){
            if(x[j]==x[j+1])  x[j]='+';
            else  x[j]='-';
        }
        for(k=i;k<n;k++) x[k]=' ';
    }
}
