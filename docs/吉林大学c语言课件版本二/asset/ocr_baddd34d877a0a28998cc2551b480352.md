int gcd( int u , int v ) {      /* 7 */
    int r ;                       /* 8 */
    r=v;                          /* 9 */
    while ( r!=0 ) {              /* 10 */
        r = u % v ;               /* 11 */
        u = v ;                   /* 12 */
        v = r ;                   /* 13 */
    }                            /* 14 */
    return u ;                    /* 15 */
}                                
void reduce ( int *x , int *y ){  /* 17 */
        int g ;                   /* 18 */
        g = gcd( *x,*y ) ;        /* 19 */
        *x = (*x) / g ;            /* 20 */
        *y = (*y) / g ;            /* 21 */
}                                /* 22 */
