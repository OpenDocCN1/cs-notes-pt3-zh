```
char* reverse(char str[]){
    char *head,*tail,tmp;
    head=tail=str;
    while(*tail!='\0')
        tail++;
    tail--;
    while(tail>head){
        tmp=*tail;
        *tail=*head;
        *head=tmp;
        tail--; head++;
    }
    return str;
}
```

```
void main(){
    char sp[]="abcde";
    reverse(sp);
}

void main(){
    char sp[]="abcde";
    char *string=sp;
    reverse(string);
}

    char *sp="abcde";
```
