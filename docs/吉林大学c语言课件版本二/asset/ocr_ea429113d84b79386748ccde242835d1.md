void moveone ( char u , char v ) {
    printf ( "%c -> %c\n" , u , v );
}
void move ( int n, char x, char y, char z ) {
    if ( n>0 ) {
        move( n-1 , x,z,y );
        moveone( x,y );
        move( n-1 , z,y,x )
    }
}
int main(void) {
    int  n ;
    printf( "pleace input n:");
    scanf( "%d" , &n );
    move ( n ,'a' ,'b' ,'c' )
}
