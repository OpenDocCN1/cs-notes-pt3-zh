
## 递归方法

```
typedef struct tr * trpointer;
trpointer reverse(tr head, tr cur){//将cur插入到head前
    tr tmp;
    if(cur!=NULL){
        tmp=cur->next;
        cur->next=head;
        reverse(cur,tmp);
    }else{
        return head;
    }
}
```

```
struct item{
    int d;
    struct item *right, *left;
};
typedef struct item * itempointer;
bool same(itempointer p, itempointer q){//比较是否相同
    if(p==NULL&&q==NULL)
        return true;
    if(p!=NULL&&q!=NULL){
        if(p->d==q->d)
            return same(p->right,q->right)&&
                   same(p->left,q->left);
        else  return false;
    }else  return false;
}
```
