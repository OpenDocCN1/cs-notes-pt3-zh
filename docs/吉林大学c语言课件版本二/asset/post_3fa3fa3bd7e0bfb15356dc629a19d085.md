	equation(rx,ry,sx,sy,tx,ty,ux,uy ,
		&l1_a,&l1_b,&l2_a,&l2_b);

}

/*已知两个直线方程的斜率和截距，求它们交点*/
void inter( float ma,float mb,float na,float nb,
    float *wx,float *wy){
    *wx=(nb-mb)/(ma-na);
    *wy=ma*wx+mb;
}

```c
/*计算由两点确定直线方程的斜率(a)和截距(b)*/
void straightline(float ex,float ey,float fx,float fy,
float *a, float *b){
*a=(fy-ey)/((fx-ex)); /* 斜率 */
*b=ey-a*ex; /* 截距 */
}

## 数组作参数

- 数组名实际是一个指针
- 数组名做实参传送给
  形参信息实质上是一个指针值
- 相应形参应是指针类型

数组做函数参数的形式一般是：

- 形参用数组声明符说明

```
int f( float x [10] )
```

- 实参用数组名对应形参数组

```
float a[10];
f(a);
```

## 数组与指针

- 数组作参数，传递给形参值是实参数组首地址
- 即把实参数组名的值送入形参中
- 在函数内，形参数组实际是使用实参数组名字开始的那片存储区
- 事实上C是把数组参数当作指针来处理的。

```
void f(int a[10]){ ... }
void main(){
    int x[11];
    f(x);
}
```

## 数组作参数，形参是一指针类型变量

- 实参数组名实际也是一个指针值
- 参数结合时，把实参指针值送入形参
- 实际是把数组首地址送入指针变量中，然后形参指向实参数组第一个元素

```
void f(int * a){ ... }
void main(){
    int x[11];
    f(x); // f(&x[0]);
}
```

## ■ C参数都值参

- 数组参数传递给函数形参不是整个数组的值，
  而是数数组名的值，也就是实参数组首地址
- 在函数内不给形参开辟数组存储空间，只给它
  开辟一个指针空间
- 参数传递的是实参数组名字的指针值。

- ① 声明可省略数组形参最外层的尺寸

if ( a[k] < a[j] )

## 例8-5 给定数组中的n个整数反序存放

void inv(int x[],int n){
    int temp,i,j;
    for(i=0;
        i<=(n-1)/2; i++){
        j=n-1-i;
        temp=x[i];
        x[i]=x[j];
        x[j]=temp;
    }
}

## 例8-6 从n个整数中找出最大和最小

```
void max_min_value(
    int x[], int n,
    int *max, int *min){
    int *p;
    *max = *min = *x;
    for(p=x+1; p<x+n; p++){
        if (*p > *max)
            *max = *p;
        if (*p < *min)
            *min = *p;
    }
}
```

```
void main(){
    int a[10],
        max_value,min_value;
    max_min_value(a,10,
        &max_value,
        &min_value);
}
void main(){
    int a[10],
        max_value,min_value;
    int *ptr=a;
    max_min_value(ptr,10,
        &max_value,
        &min_value);
}
