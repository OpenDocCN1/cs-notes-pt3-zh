int f(int n){
    int r,s,t;
    r=0;
    while(n/10>0||n>0){
        t=n%10;
        s=1;
        while(t>0){
            s=s*t;
            t=t-1;
        }
        r=r+s;
        n=n/10;
    }
    return r;
}

## 习题8.29

H(int x,int n)

n==0

return 1

n==1

return 2*x

计算Hn(x)

for(i=2;i<=n;i++)

return h2

结束

h2=2*x*h1-2*(i-1)*h0

h0=h1

h1=h2

```
int H(int x,int n){
    int h0,h1,h2;
    if(n==0)
        return 0;
    if(n==1)
        return 2*x;
    h0=1;
    h1=2*x;
    for(int i=2;i<=n;i++)
        h2=2*x*h1-2*(i-1)*h0;
        h0=h1;
        h1=h2;
    }
    return h2;
}
```
