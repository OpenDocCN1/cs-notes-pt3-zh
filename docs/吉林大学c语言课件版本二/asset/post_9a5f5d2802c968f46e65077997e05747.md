
pt sort(pt base) {          /* 链表排序，base 为链首 */
    pt p, p0, r, r0, q;
    p0 = NULL;
    p = base;
    while (p!=NULL) {   /* 逐项处理，把p加入到子序列中，并保持 “base--p”递增 */
        r = base;       /* 寻找位置 */
        while ((r->key < p->key) && (r!=p)) {
            r0 = r;
            r = r->next;
        }
        if (r!=p) {     /* p插入到r0、r之间 */ /* 若 r==p，在链尾，则不用插入 */
            q = p;      /* 把 p 独立出来，令 q 指向它 */
            p0->next = p->next;
            p = p0;
            if (r == base) {    /* 插入 */    /* 插在链首 */
                q->next = base;
                base = q;
            } else {    /* 插在链中r0、r之间 */
                q->next = r;
                r0 -> next = q;
            }
        }
        p0 = p;     /* 前进一项 */
        p = p->next;
    }
    return base;
} /* sort */

## 例12-3 法雷序列

对任意给定的自然数 n，把所有如下形式的不约分数

\frac{j}{i} \ (0 < i \le n ; 0 \le j \le i )

按递增顺序排列起来，称该序列为 n 级法雷序列 Fn 。

例如F8 为：

```
0 1 1 1 1 1 2 1 3 2 3 1 4 3 5 2 5 3 4 5 6 7 1
1 8 7 6 5 4 7 3 8 5 7 2 7 5 8 3 7 4 5 6 7 8 1
```

编函数，对任意给定的正整数n，求n级法雷序列，并带回n级法雷序列链指针。

## 分析：

法雷序列的每项是个分数，不能用实数精确的表示，而且这些数的排列顺序是不规则的。用下图形式的链表来表示它。

![](img/e9111c49b068ca3c37fc762f47ecac61_1007_0.png)

## 显然法雷序列的各项均在区间[0/1,1/1]之内。生成法雷序列算法

- ● 先把一阶法雷序列：0/1,1/1放入链表中；
- ● 然后顺序分别以i=2,3, ...,n 作分母，对任意i以j=1,2, ..., i-1 作分子，作成分数j/i；
- ● 若j/i是不可约分数，则该j/i必然是法雷序列的一项；把该j/i插入到链表的合适位置上，并使链表仍保持按递增排序。

## 生成 n 阶法雷序列链表

![](img/e9111c49b068ca3c37fc762f47ecac61_1008_0.png)

/*构造法雷序列,并返回序列头指针*/
farleipointer farlei(int n) {
    int i,j;
    farleipointer fn , r , r0 , p ;
    if( n<1 )                //如果n<=0,则没有法雷序列
        return NULL;
    /* 0/1 , 1/1 送入序列， fn指向链首 */
    fn = (farleipointer)malloc(sizeof(struct farlei_item));
                            //构造0/1
    fn->numerator = 0 ;
    fn->denominator = 1 ;
    fn->next = (farleipointer)
                malloc(sizeof(struct farlei_item)); //构造1/1
    fn->next->numerator = 1 ;
    fn->next->denominator = 1 ;
    fn->next->next = NULL ;

/* 生成序列 */
for ( i=2; i<=n; i++ )
    for (j=1; j<i; j++)
        if ( gcd(i,j) == 1 ) { /* 查 j/i 位置 r0,r */
            r = fn ;
            while ( j*r->denominator > i*r->numerator ) {
                r0 = r ;
                r = r->next ;
            }
            /* j/i 插入到 r0,r 之间 */
            p = (farleipointer)malloc(sizeof(struct farlei_item));
            p->numerator = j ;
            p->denominator = i ;
            r0->next = p ;
            p->next = r ;
        }
    return fn ;
}

设有类型说明：

struct item {

解:在本程序的算法中，在链表上利用了一个哨兵项。所谓哨兵是在链表的链首多加一节，该节不存储有效的链表项的值，而保存一个边界值或空值，本程序的哨兵项是空值。由于利用了哨兵变量，所以处理统一了。多项式加法的算法如下图：

```
void adds( int exp0 , float coef0 , polynome r ) {
    /* 向s加入一项 */
    polynome r0;
    r0 = ( polynome)malloc( sizeof(struct item) );
    r0->exp = exp0;
    r0->coef = coef0;
    r0->next = NULL;
    r->next=r0;
}
```

```
polinome addpolynome(polinome p, polinome q) {
    polinome s, r
    // s为结果多项式链首；r 始终指向结果多项式s的最低次幕项。
    /* 申请一个哨兵变量，以便算法统一 */
    s = (polynome)malloc( sizeof(struct item) );
    r = s;
    while ( (p!=NULL)&&(q!=NULL) )      /* 相加 */
        if ( p->exp > q->exp ) {
            adds( p->exp , p->coef , r );
            p = p->next;
        }else if ( p->exp < q->exp ) {
            adds( q->exp , q->coef , r );
            q = q->next;
        }else{
            adds( p->exp , p->coef+q->coef , r );
            q = q->next;      p = p->next;
        }
    r=r->next;
```
