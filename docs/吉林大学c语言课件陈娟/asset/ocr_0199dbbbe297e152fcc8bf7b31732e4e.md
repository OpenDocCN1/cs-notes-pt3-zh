```c
#include "stdio.h"
#include "math.h"
#define length 50.0
#define width 30.0
#define area 500.0
void main(){
    float x1,x2,b,d;
    b=-2.0*(length+width);
    d=sqrt(b*b-4.0*4.0*area);
    x1=(-b+d)/(2*4);
    x2=(-b-d)/(2*4);
    printf("x1=%.2f x2=%.2f \n",
    x1,x2);
}
