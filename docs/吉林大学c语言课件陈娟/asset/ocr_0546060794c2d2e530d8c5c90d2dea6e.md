void sortofup ( int n, int a[ ] ) {
    int i,r ;
    bool flag ;
    flag = true ;
    while ( flag ) {
        flag = false ;
        for ( i=0 ; i<n-1 ; i++ )
            if ( a[i] > a[i+1] ){
                r = a[i] ;
                a[i] = a[i+1] ;
                a[i+1] = r ;
                flag=true;
            }
    }
}
