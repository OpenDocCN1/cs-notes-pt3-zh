int subMax(int a[m][n]){
    int max=a[0][0],sub=-eps;
    for(int i=0;i<m;i++)
        for(int j=0;j<n;j++){
            if(a[i][j]>max){
                sub=max;
                max=a[i][j];
            }else{
                if(a[i][j]<max&&a[i][j]>sub)
                    sub=a[i][j];
            }
        }
    return sub;
}
