bool check( int u) {
    int i,j,k;
    int b[256],v;
    v=0 ;
    for ( i=n; i<=u; i++ ) {
        k=trans(i-n+1 , i);        /* A[i-n+1]~A[i]翻译成10进制=>k */
        if test_b( k , b , v )        /* 检验B中有k */
            return false ;
        b[v] = k ;
        v++;
    }
    if ( u==nn ) {
        /* 以下处理A[i]~A[u]+后边n-(u-i+1)个0 */
        for ( i=u-n+2; i<=u; i++ ) {    /* 初始位置是u-n+2而不是u-n+1 */
            k= trans( i , u );          /* 翻译A[i]~A[u]部分 */
            for (j=1;j<=n-(u-i+1);j++)  /* 后边补n-(u-i+1)个0 */
                k = k*2;
            if test_b( k , b , v )      /* 检验B中有k */
                return false ;
            b[v] = k ;
            v++;
        }
    }
    return true;                    /* 检验结果值 */
}
