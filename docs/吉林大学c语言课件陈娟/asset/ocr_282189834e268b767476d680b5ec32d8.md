#define n 10
#define n1 11
float a[n][n1],x[n] ;        /* 注意数组从0开始编下标 */
void gaoss ( void ){
float r;
int i,j,k;
for ( i=0 ; i<n-1 ; i++ )        /* 列控制： 0到n-2列 */
    for ( j=i+1 ; j<n ; j++ ){ /* 行控制： i+1到n-1行(最后一行) */
        r = a[j][i] / a[i][i] ;
        for ( k=i ; k<=n ; k++ )        /* 第j行运算，直到第n列 */
            a[j][k] = a[j][k] - a[i][k] * r ;
    }
for ( i=n-1 ; i>=0 ; i-- ){    /* 从最后一个方程（第n-1行）开始 */
    r = 0 ;
    for ( j=i+1 ; j<=n-1 ; j++ )   /* 从第i+1列到第n-1列求和 */
        r = r + a[i][j] * x[j] ;
    x[i] = ( a[i][n] - r ) / a[i][i] ;
}
}
