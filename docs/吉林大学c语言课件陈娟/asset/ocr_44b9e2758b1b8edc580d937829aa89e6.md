void sort(int col,int row,int *p){
    int i,r ;
    bool flag ;
    flag = true ;
    while ( flag ) {
        flag = false ;
        for ( i=0 ; i<col*row-1 ; i++ )
            if (*(p+i) > *(p+i+1)){
                r = *(p+i) ;
                *(p+i) = *(p+i+1) ;
                *(p+i+1) = r ;
                flag=true;
            }
    }
}
