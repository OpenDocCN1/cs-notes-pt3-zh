```
#include "stdio.h"
#define SUM 100
#define B 3
#define M 2
#define S 0.5
void main(){
    int big,mid,sm;
    for(big=0;big<=SUM/B;big++)
        for(mid=0;mid<=SUM/M;mid++){
            sm=SUM-big-mid;
            if(big*B+mid*M+sm*S==SUM)
                printf("big=%d,mid=%d,small=%d\n", big,mid,sm);
        }
}
```
