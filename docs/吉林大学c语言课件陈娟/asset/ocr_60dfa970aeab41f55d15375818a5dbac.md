#include "stdio.h"
void main( ) {
    int u,v,w,k;
    u=0;
    v=1;
    printf ( "%5d\n%5d\n",u,v);
    k=3;
    do {
        if ( k%2 == 0 )
            w=v-u;
        else
            w=v+u;
        printf ( "%12d\n",w);
        u=v;
        v=w;
        k=k+1;
    }while ( k<=50 );
}
