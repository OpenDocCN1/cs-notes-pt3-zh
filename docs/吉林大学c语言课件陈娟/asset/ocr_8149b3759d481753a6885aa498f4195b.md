bool prime((int n)){
    int j;
    bool flag;
    for ( j = n / 2 ; j >= 2 ; j-- )
        if ( n % j == 0 )
            flag = true;
    for ( i = n / 2 ; i >= 2 ; j-- )
        if ( n % j == 0 )
            flag = false ;
    return flag ;
}
