/* PROGRAM eightqueen4 */
int A[ 9 ];
bool check( int m );        /*检查某格局是否合格的函数*/
void out(void);             /* 输出函数 */
void queen( int r ) {       /* 试探法 */
    int i;
    for ( i=1; i<=8; i++ ) {
        A[8-r+1] = i;
        if ( check(8-r+1) )
            if ( r>0 ) queen(r-1);
            else out();
    }
}
int main(void){             /* 主程序 */
    queen(8)
}
