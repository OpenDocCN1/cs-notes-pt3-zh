#define n 8
int A[ n+1 ];
bool check( int m );      /*检查某格局是否合格的函数*/
void out(void);          /*输出函数*/
void queen( int r ) {    /*试探法*/
    int i;
    for ( i=1; i<=n; i++ ) {
        A[n-r+1] = i;
        if ( check(n-r+1) )
            if ( r>1 ) queen(r-1); //☆
            else out();
    }
}
int main(void){          /*主程序*/
    queen(n) //◇
}
