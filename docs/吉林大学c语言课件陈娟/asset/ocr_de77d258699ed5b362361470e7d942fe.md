# 运行结果演示

```
#include "stdio.h"
void main( ) {
    int u,v,w,k;
    int flag;
    u=0;
    v=1;
    printf ( "%5d\n%5d\n",u,v);
    k=3;
    flag = -1;
    do {
        w = v + flag*u;
        flag = -flag;
        printf ( "%5d\n",w);
        u=v;
        v=w;
        k=k+1;
    }while ( k<=50 );
}
```
