/* 生成序列 */
for ( i=2; i<=n; i++ )
    for (j=1; j<i; j++)
        if ( gcd(i,j) == 1 ) { /* 查 j/i 位置 r0,r */
            r = fn ;
            while ( j*r->denominator > i*r->numerator ) {
                r0 = r ;
                r = r->next ;
            }
            /* j/i 插入到 r0,r 之间 */
            p = (farleipointer)malloc(sizeof(struct farlei_item));
            p->numerator = j ;
            p->denominator = i ;
            r0->next = p ;
            p->next = r ;
        }
    return fn ;
}
