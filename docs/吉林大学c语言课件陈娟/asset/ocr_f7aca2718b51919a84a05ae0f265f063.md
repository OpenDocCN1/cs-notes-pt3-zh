int str_cmp( char *str1 , char *str2){
    while(*str1!='\0' && *str2!='\0'){
        if ( *str1 > *str2 )            // 比较当前字符
            return 1;
        else
            if ( *str1 < *str2 )
                return -1;
        str1++;     str2++;
    }; // 注意此处"++"运算的作用
    if ( *str1=='\0' && *str2=='\0' ) // 两串全部结束，长度相同
        return 0;
    else
        if ( *str1 != '\0' )            // str1长
            return 1;
        else
            return -1;                  // str2长
}
