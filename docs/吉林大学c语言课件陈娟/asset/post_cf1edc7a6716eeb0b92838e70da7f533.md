```c
int H(int x,int n){
    int h0,h1,h2;
    if(n==0)
        return 0;
    if(n==1)
        return 2*x;
    h0=1;
    h1=2*x;
    for(int i=2;i<=n;i++){
        h2=2*x*h1-2*(i-1)*h0;
        h0=h1;
        h1=h2;
    }
    return h2;
}
```

```c
#include "stdio.h"
#define SUM 100
#define B 3
#define M 2
#define S 0.5
void main(){
    int big,mid,sm;
    for(big=0;big<=SUM/B;big++)
        for(mid=0;mid<=SUM/M;mid++){
            sm=SUM-big-mid;
            if(big*B+mid*M+sm*S==SUM)
                printf("big=%d,mid=%d,small=%d\n", big,mid,sm);
        }
}
```
