```c
/* PROGRAM writeprime */
#include "stdio.h"
bool prime( int m ) {
    int j ;
    for ( j = m / 2 ; j >= 2 ; j--)
        if ( m%j == 0 )
            return false ;
    return true ;
}
int u,v,n;
void main() {
    n=0;  u=3;   v=u+2;
    while(n<10){
        if ( prime(u)&&prime(v)){
            printf("%5d %5d\n", u, v );
            n++;
        }
        u++; v++;
    }
}
```

## 判断字符是否是十六进制数字对应的字符

```
bool isdigit(char ch){
    return (ch>='0')&&(ch<='9') ||
           (ch>='A')&&(ch<='F') ||
           (ch>='a')&&(ch<='f');
}
```

## 十六进制数字对应的十进制值

```
int value(char ch){
    if((ch>='0')&&(ch<='9'))
        return (int)ch-'0';
    else
        if((ch>='A')&&(ch<='F'))
            return (int)ch-'A'+10;
        else
            return (int)ch-'a'+10;
}
```
