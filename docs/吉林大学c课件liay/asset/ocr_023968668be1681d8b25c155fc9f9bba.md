# 14. 避免虚函数误用

```
class Base{
public:
    virtual void fn(int x){
        cout<<"in base class,int x=" <<x<<"\n"; 
    }
};

class Sub:public Base{
public:
    virtual void fn(double x){
        cout<<"in sub class,int x=" <<x<<"\n";
    }
};

void test(Base &b){
    b.fn(2);
    b.fn(3.5);
}

void main(){
    test (Base());
    test(Sub());
    Sub f; f.fn(2);
    f.Base::fn(2.5);
}
```

in base class ,int x=2
in base class ,int x=3
in base class ,int x=2
in base class ,int x=3
in sub class,int x=2
in base class,int x=2
