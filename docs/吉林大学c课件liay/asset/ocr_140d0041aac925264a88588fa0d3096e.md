# 构造、析构函数

```
//拷贝构造函数
class Student{
public:
    Student(int k):i(k){}
    Student (){}
    Student(Student const &m)
    {i = m.i*(-1);}
    void p(){cout<<i<<endl;}
protected:
    int i;
};
```

```
int main()
{
    Student s(9818);
    s.p();
    Student t(s);
    t.p();
    Student k=s;
    k.p();
    Student *p=new Student(s);
    //也调用拷贝构造
    p->p();
    Student m;
    m=s;
    //调用赋值运算
    m.p();
    return 0;
}
```
