```cpp
//栈类
class NODE{
public:
int age;
NODE* next;
};
class Stack{
public:
Stack(){head=0;}
void push( NODE& );
NODE pop( );
private:
NODE *head;
};
void Stack::push(NODE& i){
NODE *p=new NODE;
*p = i;
p->next=head;
head=p;
}
```

```cpp
NODE Stack::pop()
{ NODE p=*head;
NODE *q=head;
head=head->next;
delete q;
return(p);
}
