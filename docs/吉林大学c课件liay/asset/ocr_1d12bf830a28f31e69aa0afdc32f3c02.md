# 构造、析构函数

```
class Student{
public:
    void Student(); //错误
    void ~Student();//错误
protected:
    char *name;
    int age;
};
void Student::Student()//错误
{ age=5;
}
void Student::~Student()//错误
{ cout<< "对象结束了" ;
}
Student a;
```

```
class Student{
public:
    Student();
    ~Student();
protected:
    char *name;
    int age;
};
Student::Student(){ age=5;}
Student::~Student()
{ cout<<"对象结束了";}

Student a;
a.Student(); //错误
a.~Student(); //不提倡
```
