# 继承
抽象类

```
class Student{
public:
    Student(int i){id=i;} virtual void p()=0; int id;
};
class Master :public Student{
public:
    Master(char * p,int k):Student(k){dept =p;}
    virtual void p(){cout<<"Master";}
    string dept;
};
class Mba :public Student{
public:
    Mba(char * p,int k):Student(k){position =p;}
    virtual void p(){cout<<"Mba";}
    string position;
};
void main()
{Master a("Software",123);
Mba b("商学院",456);
Student *m=&a; m->p();
m=&b; m->p();
}
```

纯虚函数

如果没有纯虚函数则错误.因为 c++必须确认该类拥有该同名称函数才可以访问对应对象
