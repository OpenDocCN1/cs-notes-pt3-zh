# 9. 屏蔽类的内部实现

```
```cpp
//Point.h （公用接口）
class Point{
public:
    void Set(double ix,double iy);
    double xOffset();
    double yOffset();
    double angle();
    double radius();
protected:
    double x;
    double y;
    // double a;
    // double r;
};
```
```

```
```cpp
//Point.cpp
#include "Point.h"
#include <math.h>

void point::Set(double ix,double iy)
{ x=ix;y=iy; }

double point::xOffset()
{ return x; }

double point::yOffset()
{ return y; }

double point::angle() //计算极坐标
{ return (180/3.14159)*atan2(y,x); }

double point::radius() //取极坐标半径
{ return sqrt(x*x+y*y); }
