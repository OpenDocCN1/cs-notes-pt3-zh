# 运算符重载

```
class aa{
    float a;  float b;
    public:
        float &aaa(){return(a);}
        float &bbb(){return(b);}
        friend aa & operator ++(aa &);
        friend aa operator ++(aa &,int);
};
aa & operator ++(aa & a)
{
    a.a=a.a+1; a.b=a.b+1; return(a);
}
aa operator ++(aa & a,int)
{aa b(a);
    a.a=a.a+1;
    a.b=a.b+1;
    return(b);
}
```
