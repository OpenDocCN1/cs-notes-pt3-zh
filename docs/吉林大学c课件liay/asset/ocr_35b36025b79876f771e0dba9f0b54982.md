# 异常处理

```
class A{
    char net[20];
    public:
        virtual void x(){cout<<"网络错误";}
};
class B:public A{
    long card;
    public:
        void x(){cout<<"网卡错误";}
};
class C:public B{
    int port;
    public:
        void x(){cout<<"端口错误";}
};
void net()
{
    throw B();
}
void main()
{
    try{ net(); }
    catch(A & s){s.x();}
}//异常处理的多态
```
