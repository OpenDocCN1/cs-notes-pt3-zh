# 继承

```
class Student{
    int m3;
    public:
        int m1; void x(){m1=1;m2=2;m3=3;}
    protected:
        int m2;
};
class Master :public Student{
    int n3;
    public:
        void y(){m1=1;m2=2;m3=3;n1=1;n2=2;n3=3;}  //对内
        void z(){x();}  //间接访问了父类的private成员
        int n1;
    protected:
        int n2;
};
void main()
{ Master a;
    a.m1=1;a.m2=2;a.m3=3;a.n1=1;a.n2=2;a.n3=3;  //对外
    a.x();
    a.z();
}
```
