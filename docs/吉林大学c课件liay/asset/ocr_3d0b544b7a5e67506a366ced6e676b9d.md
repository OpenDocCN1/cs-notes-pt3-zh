# 第六章：运算符重载

```
class aa{
public:
    friend aa sum(aa,aa);
    friend void main();
    float &aaa(){return(a);}
    float &bbb(){return(b);}
private:
    float a;  float b;
};
aa sum(aa s,aa t)
{ aa c; c.a=s.a+t.a; c.b=s.b+t.b; return(c);}
void main()
{ aa a,b;
    a.a=1; a.b=2;
    b.a=3; b.b=4;
    aa c=sum(a,b);
    cout<<c.a<<endl<<c.b;
    aa d=sum(b,a);
    cout<<d.a<<endl<<d.b; }
```
