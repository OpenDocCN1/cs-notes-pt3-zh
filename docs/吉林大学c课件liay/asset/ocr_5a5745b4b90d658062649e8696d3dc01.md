# 模板&类

//定做（Specialization）定制、特化、专门化
//例子（上）

```
template<typename T>
class List {
public:
    void Set(T);
    T Get();
private:
    T i;
};
template<typename T>
void List<T>::Set(T k){i = k;}
template<typename T>
T List<T>::Get(){return(i);}
```

```
template<>
class List<double> {
public:
    void Set(double);
    double Get();
    void print();
private:
    double i;
    int m;
};//一个普通类
```

```
void List<double>::Set(double k){i = k+1;}
double List<double>::Get(){return(-1*i);}
void List<double>::print()
{cout<<i<<endl;}
```
