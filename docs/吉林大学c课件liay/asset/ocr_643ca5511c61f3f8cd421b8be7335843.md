# 8.this 指针

# Student.h

```
class Student{
public:
    void p(int);
    float score;
protected:
    char name;
    int age;
};
```

成员函数不属于对象

# Student.cpp

```
#include "student.h"
void Student::p(int a){
    int age; //this的必要性
    age=a;
    //this->age=a;(*this).age=a;等价
    cout<<this->age;
    this = new Student;//错误
}//this是个指针常量
```

# a.cpp

```
#include "Student.h"
int main()
{
    Student a,b;
    a.p(5);b.p(7);
    this->score = 5; //错误
    return 0;
}
```
