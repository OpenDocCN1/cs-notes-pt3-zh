# 异常处理

```
class StudentDept{
public:
static int Max;
StudentDept(string p){
Name=new char[Max];
strcpy(Name,p);
}
char p(int i){if(i>=0&&i<Max)
return(Name[i])
else throw "cccc";}
private: char * Name;
};
int StudentDept::Max=2000;
class Student{
int id; StudentDept Dept;
public:
Student(int i,string j):Dept(j){id=i;}
StudentDept & q(){return(Dept);};
}
```

```
void main(){
try{
Student a(1,"Computer"); }
catch(exception s){....;}
try{cout<<a.q().p(5);}
catch(string m){
cout<<m<<endl;
}
}
```

exception:为系统异常的类型
