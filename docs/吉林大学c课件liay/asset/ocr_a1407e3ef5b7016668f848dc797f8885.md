//头文件在一个文件中多次被包含

# 文件名Student.h

```
class Student{
public:
    void p(){
    cout<<score;
}
    float score;
protected:
    char name;
    int age;
};
```

# 文件名ss.h

```
#include "student.h"
class ss{
public:
    Student *a;
    float f;
};
```

# 文件名bb.cpp

```
#include <iostream.h>
#include "student.h"
#include "ss.h"
void main()
{
    Student a;
    ss s;
    a.score=80;
    ss.f=50;
}
```

//提示 Student类被重复定义
