# 运算符重载

```
class aa{
    float a;  float b;
    public:
    float &aaa(){return(a);}
    float &bbb(){return(b);}
    friend aa operator +(aa&,aa&);
};
aa operator +(aa& a,aa& b)//不提倡改变运算符的含义
{aa c;
    c.a=a.a-b.a;
    c.b=a.b-b.b;
    return(c);  }
void main()
{  aa a,b;
    a.aaa()=1; a.bbb()=2;
    b.aaa()=3; b.bbb()=4;
    aa c=a+b;        //c=operator +(a,b);
cout<<c.aaa()<<endl<<c.bbb();}
```
