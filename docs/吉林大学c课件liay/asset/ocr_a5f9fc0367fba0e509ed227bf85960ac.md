# 文件流类

```
#include <iostream>
#include<fstream>
using namespace std;
void fn()
{
  ofstream myf("e:\\myname"); //ios::out|ios::trunc(存在，清除)
  myf<<"Department\n"<<"of Computer Science\n";
  myf.close(); }

void fi()
{ char ch;
ifstream myf("e:\\myname",ios::in);
//如果文件不存在，则返回错误
if(myf.fail()) {cerr<<"文件打开错误";return;}
while(myf.eof()==0){ //文件结束返回值为1;
myf>>ch;   //从myf流中提取一个字符
cout<<ch;}
myf.close();
}
void main() { fn(); fi();}
```
