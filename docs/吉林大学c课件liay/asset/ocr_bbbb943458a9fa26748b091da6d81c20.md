```
template <typename T>
void swap1(T& a,T& b)
{
T temp=a;
a=b;
b=temp;
}
void main()
{
int a=1; int b=2;
swap1<int>(a,b);//显式
swap1(a,b);//隐式
cout<<a<<b<<endl;
}
```
