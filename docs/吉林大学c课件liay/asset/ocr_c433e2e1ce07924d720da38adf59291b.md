# 运算符重载

```
class aa{ //运算符重载的友元形式
float a; float b;
public:
float &aaa(){return(a);}
float &bbb(){return(b);}
friend aa operator +(aa&,aa&);
};
aa operator +(aa & a,aa& b)
{aa c;
c.a=a.a+b.a;
c.b=a.b+b.b;
return(c); }
void main()
{
aa a,b;
a.aaa()=1; a.bbb()=2;
b.aaa()=3; b.bbb()=4;
aa c=a+b;//等价aa c=operator +(a,b);
cout<<c.aaa()<<endl<<c.bbb();
}
```
