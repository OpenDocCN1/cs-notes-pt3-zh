# 构造
析构函数

```
class aa {
public:
    aa(int a=1){id=a;cout<< "构造" <<endl;}
    aa (aa & s){id = s.id;cout<< "拷贝" ;}
    int & Getid(){return(id);}
private:
    int id;
};
int main()
{ aa m;      // "构造"
    aa n(m);  // "拷贝"
    aa o = m; // "拷贝"  和aa o(m)等价
    aa s = 9; // "构造"  和aa s(9)等价
    aa t;     // "构造"
    t = 9;    // "构造"、 赋值运算
    return 0; }
```
