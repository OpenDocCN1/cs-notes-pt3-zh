# 6. 同化效应

```
class Automobile{
public:
//...
void Run(){
    cout<< "机械变速" <<endl;
}
};

class Car:public Automobile{
public:
//...
void Run(){
    cout<< "自动变速 "<<endl;
}
};
```

```
void Fn(Automobile *p){
    p->Run();
    delete p;
}

void main(){
    Fn(new Automobile);
    Fn(new Car);
}
```
