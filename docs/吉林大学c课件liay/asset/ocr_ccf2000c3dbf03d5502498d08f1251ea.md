#include <iostream.h>

void main()
{
    int* p;       int& rp=p;
    int x=100;    int &rx=x;
    p=&x;
    cout<<"*p="<<*p<<endl;
    cout<<"x="<<x<<endl;
    cout<<"rx="<<rx<<endl;
    cout<<"*rp="<<*rp<<endl;
    x++;    (*p)++;
    rx++;   (*rp)++;
    cout<<"*p="<<*p<<endl;
    cout<<"x="<<x<<endl;
    cout<<"rx="<<rx<<endl;
    cout<<"*rp="<<*rp<<endl;
}
