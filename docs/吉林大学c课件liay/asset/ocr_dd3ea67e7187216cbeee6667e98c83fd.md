```
template <typename T>
void swap1(T& a,T& b)
{ T temp=a;
a=b;
b=temp;}
```

```
void main()
{
    int a=1,b=2,c=3;
    swap1(a,b);
    cout<<a<<b<<endl;
    swap1(a,b,c);
    cout<<a<<b<<c<<endl;
    float i=5.1,j=5.2;
    swap1(i,j);
    cout<<i<<j<<endl;
    int *p=&a,*q=&b;
    swap1(p,q);
    cout<<*p<<*q<<endl;
}
```

```
template <typename T>
void swap1(T& a,T& b,T& c)
{ T temp=a;
a=b;
b=c;
c=temp;}
```

```
void swap1(int* a,int* b)
{ int temp=*a;
*a=*b;
*b=temp;}
```
