# 3. 公有&保护&私有成员

```
class Student{
    public:
        float score;
        void p(){ age=20;}
        void q() {add();}
        void r(int a){cout<<a;}
    private: //和protected只有继承区别
        int age;
        void add(){score=0;p();age++;}
};
```

```
int main()
{Student a;
a.p();
a.q();
a.score=80;
a.age=20;    //错误
a.add();     //错误
a.r(5);
a.r((int)(a.score));
a.r(a.age);  //错误
return 0;
}
```
