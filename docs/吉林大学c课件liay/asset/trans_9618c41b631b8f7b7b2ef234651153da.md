## 构造、析构函数

```
class Student{
public:
    Student();    //构造函数的实现;系统约定与类同名; 自动调用
    int p();
protected:
    char *name;
    int age;
};

Student::Student()    //没有返回类型; 特殊的成员函数
{
    age=5;    name="zhang"; //可以有参数,可以重载
}

Int student::p()
{
    return(age);
}
Student a;  cout<<a.p();
```
