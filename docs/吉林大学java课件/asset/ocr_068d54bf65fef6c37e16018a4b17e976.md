# 从order.txt中读取数据

```
String name = null ;
float price = 0.0f ;
int num = 0 ;
char temp[] = null ;                     // 接收字符串数据
char c = '\u0000' ;                    // 声明字符变量，内容为'
int len = 0 ;
while(true){
    temp = new char[200] ;
    len = 0 ;
    while((c=dis.readChar())!='\t'){   // 读取字符
        temp[len] = c;
        len++ ;
    }
    name = new String(temp,0,len) ;
    price = dis.readFloat() ;           // 读取float
    dis.readChar() ;                    // 读出\t
    num = dis.readInt() ;               // 读取int
    dis.readChar() ;                    // 读出\n
    System.out.printf("名称: %s; 价格: %5.2f; 数量: %d\n",name,price,num) ;
}
```
