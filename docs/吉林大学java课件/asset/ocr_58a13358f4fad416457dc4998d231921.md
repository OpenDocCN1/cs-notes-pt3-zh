# 单选钮事件操作 —— 性别选择

```
class MyRadio implements ItemListener{
    private String right = "E:" + File.separator + "right.png";        // 定义图片路径
    private String wrong = "E:" + File.separator + "wrong.png" ;        // 定义图片路径
    private JFrame frame = new JFrame("Welcome To MLDN");                // 得到窗体
    private Container cont = frame.getContentPane();                       // 得到窗体容器
    private JRadioButton jrb1 = new JRadioButton("男",new ImageIcon(right),true);
    private JRadioButton jrb2 = new JRadioButton("女",new ImageIcon(wrong),false);
    private JPanel pan = new JPanel();                                      // 定义一个面板
    public MyRadio() {
        ...
        jrb1.addItemListener(this);                                         // 加入事件监听
        jrb2.addItemListener(this);                                         // 加入事件监听
        ...
    }
    public void itemStateChanged(ItemEvent e) {
        if (e.getSource() == jrb1) {
            jrb1.setIcon(new ImageIcon(right));
            jrb2.setIcon(new ImageIcon(wrong));
        } else {
            jrb1.setIcon(new ImageIcon(wrong));
            jrb2.setIcon(new ImageIcon(right));
        }
    }
}
```
