```
static Mails getMails(){
    if (currentMails == 0)
    {
        System.out.println("No mail in the mail box...");
        return null;
    }
    currentMails--;
    return showList[currentMails];
}
static void showAll(){
    int i;
    for (i=0; i<currentMails; i++)
    {
        showList[i].showMe();
    }
    return;
}
```
