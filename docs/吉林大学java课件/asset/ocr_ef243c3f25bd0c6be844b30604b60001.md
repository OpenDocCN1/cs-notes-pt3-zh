```
public class Door {
    private boolean _isOpen=false;
    public boolean isOpen(){
        return _isOpen;
    }
    public void open(){
        _isOpen = true;
    }
    public void close(){
        _isOpen = false;
    }
}
```

```
public class SmartTest {
    public static void main(String[] args) {
        Hand myHand = new Hand();
        myHand.door = new Door();
        myHand.do();
    }
}
```

```
public class Hand {
    public Door door;
    void do() {
        if (door.isOpen())
            door.close();
        else
            door.open();
    }
}
```
