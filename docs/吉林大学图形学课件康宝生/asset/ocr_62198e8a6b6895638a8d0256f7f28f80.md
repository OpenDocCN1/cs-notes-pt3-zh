```
#include<graphics.h>
#include<math.h>
void polygonc();
void block();
main()
{
int i,j,a,length, n, theta, x=150,y=350;
int gdriver=DETECT, gmode;
printf("input length, n, theta: ");
scanf("%d, %d", &length, &n);
scanf("%d", &theta);
initgraph(&gdriver,&gmode,"c:\\tc");
```

```
cleardevice();
setbkcolor(9);setcolor(4);
a=length/4;
for(i=1;i<=4;i++) {
for(j=1;j<=4;j++) {
block(x,y,a,n,theta);
theta=-theta;x=x+a;}
x=150; y=y-a;

}
```
