# include <stdio.h>
# include <graphics.h>
void AntiliasingPolygon(int xs, int ys, int xe, int ye, int c)
{
    int dx = xe - xs, dy = ye - ys, i, x = xs, y = ys;
    float m = dy/dx, a =1- m, d = 0.5;
    for(i = 0;i <= dx;i++) {
        Drawpixel(x, y, d*c); /*用于设定像素的亮度等级*/
        if (d >= a) {
            y++; d -= a;
        }
        else
            d += m; x++;
    }
}
