void boundaryFill4 (int x, int y, int F_Color, int B_Color)
{
    int current_color;
    current_color = GetPixel (x, y);
    if ((current_color != B_Color) && (current_color != F_Color)){
        DrawPixel (x, y, F_Color);
        boundaryFill4 (x + 1, y, F_Color, B_Color);
    }
}
