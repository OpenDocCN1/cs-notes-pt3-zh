# 程序 3.1.1 绘制“金刚石”图案

```
#include <graphics.h>
#include <math.h>

main()
{
float t;
int x0 = 320, y0 = 240;
int x0 = 320, y0 = 240;
int n, i, j, r, x[50], y[50];
int gdriver = DETECT,gmode;
printf("input n(20-30) and r(100-200)\n");
scanf("%d, %d", &n, &r);
initgraph(&gdriver, &gmode, "c:\\tc");
cleardevice();
```

```
setbkcolor(MAGENTA);
setcolor(WHITE);
t = 6.28318/n;
for(i = 0; i < n; i++)
{
x[i] = r*cos(i*t) + x0;
y[i] = r*sin(i*t) + y0;
}
for(i = 0; i <= n-2;i++)
for(j = i+1; j <= n-1;j++)
line(x[i], y[i], x[j], y[j]);
getch();
closegraph();
}
```
