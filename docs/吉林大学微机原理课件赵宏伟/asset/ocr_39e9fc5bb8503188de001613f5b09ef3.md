# 程序段

```
START:  MOV   AL, 90H          ; 设置方式控制字，口 A 输入，口 B 输出
        OUT   0C7H, AL
X1:     IN    AL, 0C4H          ; 输入按键状态
        AND   AL, 0FH           ; 屏蔽掉不用的高 4 位
        MOV   BX, OFFSET LEDTAB  ; 设置段码表指针
        XLAT                     ; 读取段码
        OUT   0C5H, AL           ; 输出段码到端口 B
        MOV   AX, 200H           ; 延时
X2:     DEC   AX
        JNZ   X2
        JMP   X1
        HLT
```
