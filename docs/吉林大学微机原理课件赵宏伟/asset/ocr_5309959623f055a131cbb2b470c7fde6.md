YY1:    IN      AL, 64H      ; 检查甲机的 OBF=0? (PC4=0?)
        AND     AL, 10H      ; 即检查甲机是否有数据发来
        JNZ     YY1          ; 若无数据发来，则等待
        IN      AL, 60H      ; 从 A 口读入数据
        MOV     [BX], AL     ; 存入内存
        MOV     AL, 00H
        OUT     66H, AL      ; PC0 置 0 产生 ACK 信号，并发回给甲机
        NOP                  ; 同步
        NOP
        MOV     AL, 01H      ; PC0 置 1
        OUT     66H, AL
        INC     BX           ; 内存地址加 1
        LOOP    YY1          ; 字节未完，则继续
        MOV     AH, 4CH
        INT     21H          ; 返回 DOS
CODE    ENDS
        END      START
