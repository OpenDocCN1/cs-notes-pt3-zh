# 程序设计

```
CODE    SEGMENT
        ASSUME CS:CODE

KEY     PROC    FAR
START:  PUSH    DS
        MOV     AX, 0
        PUSH    AX
        MOV     AL, 90H
        OUT     63H, AL     ; 设置 8255 的 A 口为方式 0，输入
X1:     IN      AL, 60H     ; 输入 A 口键盘状态
        AND     AL, 0FH     ; 析取 K3~K0 信号线
        CMP     AL, 0FH
        JZ      X1          ; 没有键按下，继续查询
        CALL    D20ms       ; 有键按下，延时消抖
        IN      AL, 60H     ; 输入 A 口键盘状态
        AND     AL, 0FH     ; 析取 K3~K0 信号线
        CMP     AL, 0FH
        JZ      X1          ; 此时，说明延时消抖前的按键判断是源于干扰，
                            ; 或者，延时消抖时间不足，重新查询
```
