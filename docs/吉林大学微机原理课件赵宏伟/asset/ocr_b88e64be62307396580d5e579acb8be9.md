# 程序设计

```
X1:    IN    AL, ADCS      ; 读取状态口
      TEST  AL, 80H       ; 析取 EOC
      JNZ   X1            ; EOC≠0，ADC 0809 未开始转换，等待
X2:    IN    AL, ADCS      
      TEST  AL, 80H       
      JZ    X2            ; EOC≠1，ADC 0809 未转换完成，等待
      IN    AL, ADCC      ; 读数据
      MOV   [BX], AL      
      INC   BX            ; 指向下一个数据缓冲单元
      INC   DL            ; 指向下一个通道
      INC   CL            ; 采样次数加 1
      CMP   CL, 08H       
      JNZ   X3            
      MOV   AX, 4C00H     
      INT   21H           
CODE  ENDS                 
END    START
```
