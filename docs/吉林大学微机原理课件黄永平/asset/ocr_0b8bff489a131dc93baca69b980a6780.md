BTASC          PROC
                 PUSH  AX      ; 保存AX
                 MOV   AL, 0
                 CMP   CX, 8   ; 比较8位数
                 JNE   L1      ; 直接转换16位数
                 MOV   DH, DL  ; 8位数转换送DH
L1:              ROL   DX,, 1   ; DX最高位移入CF
                 RCL   AL, 1   ; CF移入AL最低位
                 ADD   AL, 30H
                 MOV   [DI], AL
