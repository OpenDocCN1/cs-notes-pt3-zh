# 4.源程序结构4

MASM 32环境, exe

```
;显示字符
;Author: hyp@jlu.edu.cn , QQ:hyper/790516, http://www.cnblogs.com/hyper99
;---模式设置----------------------------------------
.MODEL SMALL ;SMALL为EXE文件格式
.8086
;---堆栈段------------------------------------------
.stack
;---数据段------------------------------------------
.data
    msg db "Hello World!$"
;---代码段------------------------------------------
.code
.startup
    mov dl, 'A'
    mov ah, 2
    INT 21H
    call disp

loc_end: JMP loc_end

disp proc near
    mov dl, 'B'
    mov ah, 2
```
