MASM 5.0、Emu8086环境

```
;-----0主程序开始----------
org 100h
LEA BX, BUF
MOV CX, 7
CALL display
RET ;返回DOS
;-----1 子过程开始----------
display PROC NEAR
next1: MOV DL,[BX]
MOV AH, 2
INT 21H
INC BX
LOOP next1
RET
display ENDP
;-----2 定义变量----------
BUF DB 0AH, 0DH, 'ABCDE'
```

```
CODE SEGMENT
;-----0主程序开始----------
ORG 100H
ASSUME CS:CODE,DS:CODE
START: MOV AX,CS
MOV DS, AX
LEA BX, BUF
MOV CX, 7
CALL display
RET ; 返回DOS
;-----1 子过程开始----------
display PROC NEAR
next1: MOV DL,[BX]
MOV AH, 2
INT 21H
INC BX
LOOP next1
RET
display ENDP
;----- 2定义变量----------
BUF DB 0AH, 0DH, 'ABCDE'
CODE ENDS
END START
```
