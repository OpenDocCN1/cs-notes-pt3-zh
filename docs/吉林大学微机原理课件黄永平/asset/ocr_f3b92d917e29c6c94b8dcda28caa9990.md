```
;-----0主程序开始----------
    org 100h
    ;主程序代码
    RET ;返回DOS
;-----1 子过程开始----------
P_Name PROC    NEAR
    ;子过程代码
    RET
P_Name ENDP
;-----2 定义变量----------
X DB ...
Y DW ...
```

```
CODE    SEGMENT
;-----0主程序开始----------
        ORG 100H
        ASSUME CS:CODE,DS:CODE
START:
        ;!主程序代码
        RET ; 返回DOS
;-----1 子过程开始----------
P_Name PROC    NEAR
        ;!子过程代码
        RET
P_Name ENDP
;----- 2定义变量----------
        X DB ...
        Y DW ...
CODE    ENDS
END START
```
