# 例9. 2009研究生入学考试题

```
main()
{
    process P1
    do{
        n=produce();
        P(empty);
        P(mutex);
        put();
        V(mutex);
        if(n%2==0)
            V(even);
        else
            V(odd);
    }while(true)

    process P2
    do{
        P(odd);
        P(mutex);
        getodd();
        V(mutex);
        V(empty);
        countodd();
    }while(true)

    process P3
    do{
        P(even);
        P(mutex);
        geteven();
        V(mutex);
        V(empty);
        counteven();
    }while(true)
}
```
