Bitree* Create_tree( )
{
scanf("%s",a);
Ht=(Bitree*)malloc(sizeof(Binode));
Ht->data=a[0];
p= Ht;
InitStack(s);
Push(&s, Ht);
i=1;
while( a[i]!='\0' )
{
if(a[i]!='#')
{
q=(Bitree*)malloc(sizeof(Binode));
q->data=a[i];
if(p==gettop(s))
 p->lchild=q;
 else p->rchild=q;
 p=q;
 push(&s,p);
 i++;
}
else
{
if(p==gettop(s))
 p->lchild=null;
 else p->rchild=null;
 pop(&s,p); i++;
}
return Ht;
}
