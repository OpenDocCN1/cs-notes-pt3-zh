# 算法如下：

```
int match(char *exp)
{
	 top=0; //注意：与前面top=-1相比较
	 i=0;
	nomatch=0;
	While(exp[i]!='\0'&&nomatch===0){
		switch(exp[i]){
		case '(': st[top]=exp[i]; top++; break;
		case '[': st[top]=exp[i]; top++; break;
		case '{': st[top]=exp[i]; top++; break;
		case ')': if (st[top]==='(') top--;
		else nomatch=1; break;
		case ']': if (st[top]==='[') top--;
		else nomatch=1; break;
		case '}': if (st[top]==='{' ) top--;
		else nomatch=1; break;
		}
	 i++;
	}
	if (nomatch==0&&top==0) return 1;
	else return 0;
}
```
