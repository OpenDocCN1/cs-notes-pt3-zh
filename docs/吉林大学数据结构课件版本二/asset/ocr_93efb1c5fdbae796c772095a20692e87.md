int Listdelete_L( LinkList L, int i ,elemtype &e)
{
    //将链表第 i 号元素删去
    p = LocateElem_Sq( L, i-1 ); //寻找第i-1个结点
    if ( p == NULL || p->next == NULL) return 0;
    q = p->next;
    p->next = q->next;          //删除结点
    e=q->data;      free( q );      //释放
    return OK;
}
