Void DFSTree(graph G,int v,cstree &T)
{
visited[v]=true;first=true;
For(w=FirstAdjVex(G,v);w>=0;w=NextAdjVex(G,v,w))
if(!visited[w])
{
p=(cstree)malloc(sizeof(csn...))
*p={getvex(G,w),null,null};
if(first){
T->lchild=p;first=false;}
else{q->nextsibling=p;}
q=p;
DFSTree(G,w,q);}
}
