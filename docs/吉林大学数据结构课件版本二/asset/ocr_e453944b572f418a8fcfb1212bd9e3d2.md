int bin_search (SSTable ST, keytype key)
{
low=1; high=ST.length;    //置初值查找范围的低、高端指针
while (low<=high)
{
mid=(low+high)/2;         //计算中间项位置
if (key==ST.elem[mid].key) return mid;   //找到，结束循环
else if (key< ST.elem [mid].key)
high=mid-1;               //给定值k小
else low=mid+1;          //给定值k大
}
return 0 ;               //查找失败
}
