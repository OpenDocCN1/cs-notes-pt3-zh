int ListInsert_L (LinkList &L, int i, elemtype e)
{
    //将新元素 x 插入在链表中第 i 号结点位置
    p = LocateElem_Sq ( L, i-1 );

    if ( p == NULL ) return ERROR;

    s =(LinkList) malloc (sizeof (LNode) );
    //创建新结点
    s->data = e;
    s->next = p->next;
    //链入
    p->next = s;
    return ok;
}
