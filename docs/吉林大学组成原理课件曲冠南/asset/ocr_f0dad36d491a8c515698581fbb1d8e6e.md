# ③ Booth 算法（被乘数、乘数符号任意） 6.3

设[x]补 = x₀·x₁x₂ ⋯ xₙ    [y]补 = y₀·y₁y₂ ⋯ yₙ
[x·y]补
= [x]补(0,y₁ ⋯ yₙ) − [x]补·y₀
= [x]补(y₁ 2⁻¹+y₂2⁻²+ ⋯ +yₙ2⁻ⁿ) − [x]补·y₀
= [x]补(−y₀+y₁ 2⁻¹+y₂2⁻²+ ⋯ +yₙ2⁻ⁿ)
= [x]补[−y₀+(y₁−y₁2⁻¹)+(y₂2⁻¹−y₂2⁻²)+ ⋯ +(yₙ2⁻⁽ⁿ⁻¹⁾−yₙ2⁻ⁿ)]
= [x]补[(y₁−y₀)+(y₂−y₁)2⁻¹+ ⋯ +(yₙ−yₙ₋₁)2⁻⁽ⁿ⁻¹⁾+(0−yₙ)2⁻ⁿ]
= [x]补[(y₁−y₀)+(y₂−y₁)2⁻¹+ ⋯ +(yₙ₊₁−yₙ)2⁻ⁿ]

注：−[x]补 = +[−x]补
2⁻¹ = 2⁰ − 2⁻¹
2⁻² = 2⁻¹ − 2⁻²
附加位 yₙ₊₁
y₁ 2⁻¹ + ⋯ + yₙ 2⁻ⁿ
