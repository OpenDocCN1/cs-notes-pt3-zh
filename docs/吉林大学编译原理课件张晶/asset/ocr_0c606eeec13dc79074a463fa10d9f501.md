# 带有分程序的C程序的例子

a = 3, b = 2

a = 1, b = 4

a = 1, b = 1

```
int main()
{
    int a = 1;
    int b = 1;
    B1
    {
        int b = 2;
        B2
        {
            int a = 3;
            printf("a = %d, b = %d\n",a, b);
            B3
        }
        {
            int b = 4;
            printf("a = %d, b = %d\n",a, b);
            B4
        }
    }
    printf("a = %d, b = %d\n", a, b);
}
```
