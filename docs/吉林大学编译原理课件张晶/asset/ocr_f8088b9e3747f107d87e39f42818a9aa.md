# 关于Pascal语言的符号表的实例（跳转法）

```
PROGRAM example;
1.    procedure P()
2.        var i, j, k: integer;  {t1}
3.        procedure Q()
4.            var a, b: real;
5.            procedure R()      {t2}
6.                var a, b: boolean;  {t3}
7.                begin
8.                R的过程体 (a, b, i)
9.                end;
10.   begin
11.       Q的过程体 (a, b, i)
12.   end;
13.   procedure S()
14.       var c, e: char;      {t4}
15.       begin
16.           S的过程体 (a, b, i)
17.       end;    {t5}
18. begin
19.   P的过程体 (a, b, i)
end
begin      .....      end.
```
