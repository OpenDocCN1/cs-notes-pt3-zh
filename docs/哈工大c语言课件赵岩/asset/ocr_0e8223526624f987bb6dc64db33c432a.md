# 二维数组作为函数实参

```
1 int a[2][3];
2 GetMax(int *p, int m,int n,int *iRow,int *iCol)
3 {
4   *(p+1*n+2) /*第1行，第2列*/
5 }
6 GetMax(*a,2,3,&row,&col); /*调用*/
7 
8 GetMax(int array[][3],int m,int n,int *iRow,int *iCol)
9 {
10  *(*(array+1)+2) /*第1行，第2列*/
11  array[1][2] /*第1行，第2列*/
12 }
13 GetMax(a,2,3,&row,&col); /*调用*/
```
