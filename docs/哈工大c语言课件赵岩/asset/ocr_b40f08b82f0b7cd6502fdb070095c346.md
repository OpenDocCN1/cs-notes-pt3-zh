/* include <stdio.h> */
int main()
{
    int farH, closins;
    int lower = 0;
    upper = 300;
    step = 20;
    for (farH = lower; farH <= upper; farH += step)
    {
        printf("%d %d\n", farH, closins);
    }
    return 0;
}
