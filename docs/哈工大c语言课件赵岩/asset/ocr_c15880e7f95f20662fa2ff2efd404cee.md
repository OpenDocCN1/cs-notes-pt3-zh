```
1 typedef struct
2 {
3     short level;                /* 缓冲区‘满或空’的程度 */

6     unsigned char hold;         /* 如无缓冲区不读字符 */

9     unsigned char *curp;        /* 指针当前的方向 */
10    unsigned ist emp;            /* 临时文件指示器 */
```
