```c
int CopyFile(const char* srcName,
    const char* dstName){
    FILE* fpSrc = NULL;
    FILE* fpDst = NULL;
    int ch, rval=1;
    fpSrc = fopen(srcName, "rb");
    if(fpSrc == NULL)
        goto ERROR;
    fpDst = fopen(dstName, "wb");
    if(fpDst == NULL)
        goto ERROR;
    while((ch=fgetc(fpSrc))!=EOF){
        if(fputc(ch, fpDst)==EOF)
            goto ERROR;
    }
    ```
