public void go(){
    Border etched=BorderFactory.createEtchedBorder();
    Border border=BorderFactory.createTitledBorder(etched,"JCheckBox Group");
    p.setBorder(border);//设置边框
    ButtonGroup group=new ButtonGroup();
    group.add(cb1);group.add(cb2);group.add(cb3);
    p.add(cb1);p.add(cb2);p.add(cb3);
    frame.getContentPane().add(ta,BorderLayout.CENTER);
    frame.getContentPane().add(p,BorderLayout.SOUTH);
    frame.setSize(400,200);
    frame.setVisible(true);
    ItemListener il=new ItemListener(){//事件监听器(匿名类)
        //事件处理器
        public void itemStateChanged(ItemEvent e){
            JToggleButton tb=(JToggleButton) e.getSource();//取得事件源
            if(tb==cb1)//点击JToggleButton1
                ta.append("JToggleButton1");
            else if(tb==cb2)//点击JToggle
                ta.append("\n JToggleButton");
            else if(tb==cb3)//点击JToggle
                ta.append("\n JToggleButt");
        }
    };
    cb1.addItemListener(il);//注册事件监听
    cb2.addItemListener(il);//注册事件监听
    cb3.addItemListener(il);//注册事件监听
}
