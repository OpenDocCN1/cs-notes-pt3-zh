# 按钮

```
public void go(){
    p.add(tb1);
    p.add(tb2);
    p.add(tb3);
    frame.getContentPane().add(ta,
    frame.getContentPane().add(p,
    frame.setSize(400,200);
    frame.setVisible(true);
    ItemListener il=new ItemListener(){//事件监听器(匿名类)
        //事件处理器
        public void itemStateChanged(ItemEvent e){
            JToggleButton tb=(JToggleButton) e.getSource();//取得事件源
            if(tb==tb1){//点击JToggleButton1
                ta.append("JToggleButton1"+tb1.isSelected());
            }else if(tb==tb2){//点击JToggleButton2
                ta.append("\n JToggleButton2"+tb2.isSelected());
            }else if(tb==tb3){//点击JToggleButton3
                ta.append("\n JToggleButton3"+tb3.isSelected());
            }
        }
    };
    tb1.addItemListener(il);//注册事件监听器
    tb2.addItemListener(il);//注册事件监听器
    tb3.addItemListener(il);//注册事件监听器
}
```
