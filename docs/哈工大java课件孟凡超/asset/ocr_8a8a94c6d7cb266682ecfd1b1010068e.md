# 5.9 抽象类

# 抽象类示例

```
abstract class ObjectStorage{//抽象类
    int objectnum=0;
    Object storage[]=new Object[100];
    abstract void put(Object o);//抽象方法
    abstract Object get();        //抽象方法
}
```

```
class Stack extends ObjectStorage{
    private int point=0;
    public void put(Object o){//实现方法
        storage[point++]=o;
        objectnum++;
    }
    public Object get(){//实现方法
        objectnum--;
        return storage[--point];
    }
}
```

```
class Queue extends ObjectStorage{
    private int top=0;
    private int bottom=0;
    public void put(Object o){//实现方法
        storage[top++]=o;
        objectnum++;
    }
    public Object get(){//实现方法
        objectnum--;
        return storage[bottom++];
    }
}
```
