public void go(){
    Border etched=BorderFactory.createEtchedBorder();
    Border border=BorderFactory.createTitledBorder(etched,"JRadioButton Group");
    p.setBorder(border);//设置边框
    ButtonGroup group=new ButtonGroup();
    group.add(rb1);group.add(rb2);group.add(rb3);
    p.add(rb1); p.add(rb2);p.add(rb3);
    frame.getContentPane().add(ta,BorderLayout.CENTER);
    frame.getContentPane().add(p,BorderLayout.SOUTH);
    frame.setSize(400,200);
    frame.setVisible(true);
    ItemListener il=new ItemListener(){//事件监听器(匿名类)
    //事件处理器
    public void itemStateChanged(ItemEvent e){
        JToggleButton tb=(JToggleButton) e.getSource();//取得事件源
        if(tb==rb1){//点击JToggleButton1
            ta.append("JToggleButton1"+rb1.isSelected());
        }else if(tb==rb2){//点击JToggleButton2
            ta.append("\n JToggleButton2"+rb2.isSelected());
        }else if(tb==rb3){//点击JToggleButton3
            ta.append("\n JToggleButton3"+rb3.isSelected());
        }
    }
    };
    rb1.addItemListener(il);//注册事件监听器
    rb2.addItemListener(il);//注册事件监听器
    rb3.addItemListener(il);//注册事件监听器
}
