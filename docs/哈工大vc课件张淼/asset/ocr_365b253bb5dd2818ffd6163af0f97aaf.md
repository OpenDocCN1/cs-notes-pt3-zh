class RMB
{
public:
    RMB(double value=0.0)
    {
        yuan =value;
        fen = (value-yuan)*100+0.5;
    }
    void ShowRMB()
    {
        cout<<yuan<<"元"
            <<fen<<"分"<<endl;
    }
    operator double ()
    {
        return yuan+fen/100.0;
    }
private:
    int yuan, fen;
};
