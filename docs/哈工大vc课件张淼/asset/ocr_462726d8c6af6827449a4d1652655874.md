# 例：向标准设备输出字符串

```
```cpp
#include <iostream>
using namespace std;
int main()
{
    char *word = "hello world!";
    double pi = 3.1415926;
    int dayOfYear = 365;
    cout<<"value of word is:"<<word<<endl;
    cout<<"PI = "<<pi<<endl;
    cout<<"days of year: "<<dayOfYear<<endl;
    cout.put('B').put('C').put('\n');
    return 0;
}
