# // 提取数据函数的实现

```
template <class T>
T Store<T>::GetElem()
{
    // 如果试图提取未初始化的数据，则终止程序
    if (haveValue == 0)
    {
        cout << "No item present!" << endl;
        exit(1);
    }
    return item;   // 返回item中存放的数据
}
```
