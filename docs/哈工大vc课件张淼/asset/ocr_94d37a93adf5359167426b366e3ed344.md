# 过程式

```
// 声明学生结构Student
struct Student
{
    char strStudentName[20];      // 学生姓名
    int  nStudentAge;             // 学生年龄
    char cStudentSex;             // 学生性别
    int  nStudentNumber;          // 学生学号
};
// 声明老师结构Teacher
struct Teacher
{
    char strTeacherName[20];      // 老师姓名
    int  nTeacherAge;             // 老师年龄
    char cTeacherSex;             // 老师性别
    int  nIsTeaching;             // 是否教书
};
```
