CWinThread::Run()
{
    // 捕获并派送消息直到收到WM_QUIT消息
    for (;;)
    {
        // phase1: 检查是否空闲状态，是则做空闲循环
        while (bIdle && !::PeekMessage(&m_msgCur, ...))
        {
            // call OnIdle while in bIdle state
        }
        // phase2:接收到消息，派送消息
        do
        {
            // 派送消息,如果是 WM_QUIT退出
            if (!PumpMessage())
                return ExitInstance();
            //...
        } while (::PeekMessage(&m_msgCur, ...));
    }
}
