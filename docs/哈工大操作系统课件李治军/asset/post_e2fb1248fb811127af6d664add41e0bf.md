## 多个进程共同使用磁盘

```
static void make_request()
{
 ... .req->sector=bh->b_blocknr<<1;
 add_request(major+blk_dev,req); }
```

```
static void add_request(struct blk_dev_struct *dev, struct request *req)
{
 struct request *tmp=dev->current_request;
 req->next=NULL; cli(); //关中断(互斥)
 for(;tmp->next;tmp=tmp->next)
 if ((IN_ORDER(tmp, req) || !IN_ORDER(tmp, tmp->next)) && IN_ORDER(req, tmp->next)) break;
 req->next=tmp->next; tmp->next=req; sti(); }
```

```
#define IN_ORDER(s1, s2) \
 ((s1)->dev<(s2)->dev || ((s1)->dev == (s2)->dev\
 && (s1)->sector<(s2)->sector))
```

sector = Cx(Heads×Sectors) + HxSectors + S
