```
cellptr Search(keytype k, HASH F)
{
    cellptr bptr;
    bptr=F[h(k)];
    while(bptr!=NULL)
        if(bptr->data.key==k)
            return bptr;
        else
            bptr=bptr->next;
    return bptr;//没找到
}/*Search*/
```
