# 结点结构

| lchild | ltag | data | rchild | rtag |

p->ltag = { TRUE   p->lchild 指向左孩子
FALSE   p->lchild 指向（中序）前驱 }

p->rtag = { TRUE   p->rchild 指向右孩子
FALSE   p->lchild 指向（中序）后继 }
