```
void Delete(keytype k, HASH F)
{ int bucket=h(k); celltype bptr, p;
if(F[bucket] != NULL)//可能在表中
if(F[bucket]->data.key==k)//首元素就是
bptr= F[bucket];
F[bucket]=F[bucket]->next;
free (bptr);
}else{//可能在中间或不存在
bptr=F[bucket];
while(bptr->next!=NULL)
if(bptr->next->data.key==k){
p=bptr->next;
bptr->next=p->next;
free( p);
}else
bptr=bptr->next;
}
}/* Delete */
```
