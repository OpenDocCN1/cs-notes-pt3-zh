哈夫曼树构造算法的实现

```
void CreartHT(HuffmanT T)//构造huffam树,T[2n-2]为其根
{ int i ,p1 ,p2;
    InitHT(T);                     //1.初始化
    InputW(T);                     //2.输入权值
    for ( i = n; i < 2n-1; i++ ) {  //3. n-1次合并*/
        SelectMin(T, i-1, &p1, &p2); //3.1
        T[p1].parent =T[p2].parent = i; //3.2
        T[i].lchild= p1;
        T[i].rchild= p2;
        T[i].weight =T[p1].weight + T[p2].weight;
    }
}
```
