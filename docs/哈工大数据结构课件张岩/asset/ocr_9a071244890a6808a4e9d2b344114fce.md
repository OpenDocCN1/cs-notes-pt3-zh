# 6.9 基数排序---多关键字排序（cont.）

算法实现：

```
void RadixSort( int figure, QUEUE &A)
{
  QUEUE Q[10]; records data ;
  int pass, r, i ;
  for ( pass=1; pass<=figure ; pass++ ){
    for ( i=0 ; i<=9 ; i++ ) /*置空队列*/
      MAKENULL( Q[i] );
    while ( !EMPTY( A ) ) {/*分配*/
      data = FRONT ( A );
      DEQUEUE( A );
      r = Radix(data.key, pass);
      ENQUEUE( data , Q[r] ); }
    for ( i=0 ; i <=9 ; i++ ) /*收集*/
      while ( !EMPTY( Q[i] ) ) {
        data = FRONT ( Q[i] );
        DEQUEUE( Q[i] );
        ENQUEUE( data, A );}
}
```

```
/*求整数 k 的第 p 位*/
int Radix ( int k, int p )
{ int power=1 ;
  for ( int i=1; i<=p-1 ; i++ )
    power = power * 10 ;
  return
    (( k%(power*10))/power );
}
```
