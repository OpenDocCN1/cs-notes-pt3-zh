# Dijkstra算法的实现

```
void Dijkstra(GRAPH C, costtype D[n+1], int P[n+1], bool S[n+1])
{
  for ( i=1 ; i<=n; i++ )
  {
    D[i]=C[1][i]; S[i]=FALSE ;}
  S [1]= TRUE ;
  for( i=1; i<n; i++)
  { w=MinCost ( D, S );
    S[w]=TRUE ;
    for ( v=2 ; v<= n ; n++ )
      if ( S[v]!=TRUE )
      {  sum=D[w] + C[w][v] ;
        if (sum < D[v]){D[v] = sum ; P[v]=w;}}
  }
}// 时间复杂度： O (n²)
```

costtype MinCost (D, S)
{
temp = INFINITY ;
w = 2 ;
for ( i=2 ; i<=n ; i++ )
  if (!S[i]&&D[i]<temp)
  { temp = D[i] ;
    w = i ;
  }
return w ;
}
