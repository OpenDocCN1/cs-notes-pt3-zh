# 普里姆（Prim）算法的实现

```
void Prim(Costtype C[n+1][n+1])
{ costtype LOWCOST[n+1]; int CLOSSET[n+1]; int i,j,k; costtype min;
    for( i=2; i<=n; i++ )
    { LOWCOST[i] = C[1][i]; CLOSSET[i] = 1; }
    for( i = 2; i <= n; i++ )
    { min = LOWCOST[i];
      k = i;
      for( j = 2; j <= n; j++ )
        if ( LOWCOST[j] < min )
        { min = LOWCOST[j]; k=j; }
      cout << "(" << k << "," << CLOSSET[k] << ")" << endl;
      LOWCOST[k] = infinity ;
      for ( j = 2; j <= n; j++ )
        if ( C[k][j] < LOWCOST[j] && LOWCOST[j] < infinity )
        { LOWCOST[j]=C[k][j]; CLOSSET[j]=k; }
    }
} /* 时间复杂度： O(|V|^2) */
```
