从一个顶点出发的一次深度优先遍历算法：

```cpp
void DFS2(MTGraph *G, int i)
//以vi为出发点对邻接矩阵表示的图G进行深度优先搜索
{
 int j;
 cout<<G->vexlist[i]; //访问定点vi
 visit[i]=TRUE; //标记vi已访问
 dfn[i]=count; //对vi进行编号
 count ++; //下一个顶点的编号
 for( j=0; j<G->n; j++ ) //依次搜索vi的邻接点
 if ( (G->edge[i][j] == 1)&&! visited[j] ) //若vj尚未访问
 DFS2( G, j );
}//DFS2
