# 8.4.1 程序块结构的符号表

变量的作用域满足最近嵌套原则

```
(1) int main()
(2) {
(3)     int a=0;
(4)     int b=0;
(5)     {
(6)         int b=1;
(7)         {
(8)             int a=2;
(9)             printf("%d  %d\n", a, b);
(10)        }
(11)        {
(12)            int b=3;
(13)            printf("%d  %d\n", a, b);
(14)        }
(15)        printf("%d  %d\n", a, b);
(16)    }
(17)    printf("%d  %d\n", a, b);
(18) }
```
