procedure SORT(CH);
{  case CH of ‘字母’ :
    ‘字母’ :  call RECOGID(CH, TOKEN);
    ‘/’ :      call HANDLECOM(CH, TOKEN);
    ‘数字’ :  call RECOGDIG(CH, TOKEN);
    ‘,’ ‘  call RECOGSTR(CH, TOKEN);
    otherwise call RECOGDEL(CH, TOKEN);
end case;
write TOKEN into TOKEN文件;
Return  }
