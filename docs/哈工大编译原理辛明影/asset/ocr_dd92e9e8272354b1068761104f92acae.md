# 用数值表示布尔值的翻译

B → ( B₁ ) {B.addr:= B₁.addr}

B → E₁ relop E₂ {B.addr := newtemp;
gencode('if'E₁.addr relop.op E₂.addr'goto'nextquad+3);
gencode(B.addr ':=' '0');
gencode('goto'nextquad+2);
gencode(B.addr ':=' '1')}

B → true  {B.addr:= newtemp; gencode(B.addr ':=' '1')}

B → false  {B.addr:= newtemp; gencode(B.addr ':=' '0')}
