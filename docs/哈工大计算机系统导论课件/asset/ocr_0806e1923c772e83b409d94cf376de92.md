# 共享公用子表达式

+   - 重用表达式的一部分
+   - GCC 使用 -O1 做到这个

3 乘法: i*n, (i-1)*n, (i+1)*n

1 乘法: i*n

```
/* Sum neighbors of i,j */
up =     val[(i-1)*n + j  ];
down =   val[(i+1)*n + j  ];
left =   val[i*n      + j-1];
right =  val[i*n      + j+1];
sum = up + down + left + right;
```

```
long inj = i*n + j;
up =     val[inj - n];
down =   val[inj + n];
left =   val[inj - 1];
right =  val[inj + 1];
sum = up + down + left + right;
```

```
leaq  1(%rsi), %rax  # i+1
leaq  -1(%rsi), %r8  # i-1
imulq %rcx, %rsi     # i*n
imulq %rcx, %rax     # (i+1)*n
imulq %rcx, %r8      # (i-1)*n
addq  %rdx, %rsi     # i*n+j
addq  %rdx, %rax     # (i+1)*n+j
addq  %rdx, %r8      # (i-1)*n+j
```

```
imulq  %rcx, %rsi  # i*n
addq   %rdx, %rsi  # i*n+j
movq   %rsi, %rax  # i*n+j
subq   %rcx, %rax  # i*n+j-n
leaq   (%rsi,%rcx), %rcx # i*n+j+n
```
