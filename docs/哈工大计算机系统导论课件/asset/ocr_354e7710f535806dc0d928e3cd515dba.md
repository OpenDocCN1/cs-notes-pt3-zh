# 数组和循环的例子

```
```c
void zincr(zip_dig z) {
    size_t i;
    for (i = 0; i < ZLEN; i++)
        z[i]++;
}
```
```

```
```asm
# %rdi = z
movl    $0, %eax       # i = 0
jmp     .L3            # goto middle
.L4:                    # loop:
    addl    $1, (%rdi,%rax,4) # z[i]++
    addq    $1, %rax          # i++
.L3:                    # middle
    cmpq    $4, %rax           # i:4
    jbe     .L4                # if <=, goto loop
rep; ret
```
