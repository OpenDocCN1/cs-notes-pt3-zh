# 算术表达式例子

```
long arith
(long x, long y, long z)
{
long t1 = x+y;
long t2 = z+t1;
long t3 = x+4;
long t4 = y * 48;
long t5 = t3 + t4;
long rval = t2 * t5;
return rval;
}
```

```
arith:
leaq (%rdi,%rsi), %rax  # t1
addq %rdx, %rax      # t2
leaq (%rsi,%rsi,2), %rdx
salq $4, %rdx        # t4
leaq 4(%rdi,%rdx), %rcx # t5
imulq %rcx, %rax      # rval
ret
```

| 寄存器 | 用途 |
|---|---|
| %rdi | 参数x |
| %rsi | 参数y |
| %rdx | 参数z |
| %rax | t1, t2, rval |
| %rdx | t4 |
| %rcx | t5 |
