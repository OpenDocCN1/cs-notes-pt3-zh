# 流程控制语句 – 示例4

| // C for
for(i=0; i<8; i++)
  sum += i; | ## ASM for
    move t0, $0
    li   t1, 8
1: add  t2, t0, t2
    addi t0, t0, 1
    bne  t0, t1, 1b # label1 backward
    nop |
| // C while
i = 0;
while (i < 8)
  i++; | ## ASM while
    move t0, $0
    li   t1, 8
1: bge  t0, t1, 2f
    nop
    b    1b
    addi t0, t0, 1  # delay slot
2: | 
| // C do ~ while
i = 0;
do {
  i++;
}while (i < 8) | ## ASM do ~ while
    move t0, $0
    li   t1, 8
1: blt  t0, t1, 1b
    addi t0, t0, 1  # delay slot |
