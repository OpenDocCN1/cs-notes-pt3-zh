```cpp
Point::Point(Point &p){
    x = p.x; y = p.y;
    name = p.name;
    flag=0;   //flag为0时代表拷贝构造生成对象
    cout<<name<<" copy constructing"<<endl;
}
```
```

```
```cpp
Point::~Point(){
    cout<<name<<" destructing"<<endl;
    if(flag)
        delete [] name;
}
