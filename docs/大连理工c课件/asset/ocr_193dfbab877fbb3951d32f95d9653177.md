# 7.3 类模板

template<typename T>
Array<T>::Array(int s)
{ size = (s > 1) ? s : 1 ;
element = new T [ size ] ; }
template < typename T > Array < T > :: ~Array()
{ delete [ ] element ; }
template < typename T >
const T& Array < T > :: Entry ( int index ) const
{ return element [ index ] ; }
template < typename T >
void Array < T > :: Enter(int index, const T& value)
{ element [ index ] = value ; }
