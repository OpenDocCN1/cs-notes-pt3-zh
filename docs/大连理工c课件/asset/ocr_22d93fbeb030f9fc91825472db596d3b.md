# # 第8章 异常处理

```
class ArrayIndexOutOfBounds{
public:
    ArrayIndexOutOfBounds(int index):index(index){}
    void show(){cout<<"数组下标越界异常 index="<<index<<endl;}
private: int index;
};
class MyArray{
private: int *p; int sz;
public: MyArray(int s):sz(s) {p=new int[sz]; }
    ~MyArray( ) { delete [ ] p ; }
    int size( ) { return sz; }
    int& operator[ ] (int i){
    if(i<0 || i>=sz) throw ArrayIndexOutOfBounds(i); return p[i];
    };
};
```
