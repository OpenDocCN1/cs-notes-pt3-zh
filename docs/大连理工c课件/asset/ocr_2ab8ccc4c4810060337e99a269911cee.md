# # 公有继承的测试

void main()
{
 A objA ;
 B objB ;
 C objC ;
 cout << "It is object_A :\n" ;
 objA.get_XY() ;
 objA.put_XY() ;
 cout << "It is object_B :\n" ;
 objB.get_XY() ;
 objB.make_S() ;
 cout << "S = " << objB.get_S() << endl ;
 cout << "It is object_C :\n" ;
 objC.get_XY() ;
 objC.get_H();
 objC.make_V() ;
 cout << "V = " << objC.get_V() << endl ;
}

|    | x    | y    |
|---|---|---|
| objA | objA.x | objA.y |

|    |    |    | s    |
|---|---|---|---|
| objB | objB.x | objB.y | objB.s |

|    |    |    | h    | v    |
|---|---|---|---|---|
| objC | objC.x | objC.y | objC.s | objC.h | objC.v |
