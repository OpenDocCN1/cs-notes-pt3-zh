```cpp
class Point
{
private:
    double x, y;    //x,y 坐标
public:
    Point(double x=0.0, double y=0.0);
    void disp();
    friend class ComputeTools; //声明为Point的友元类
};
Point::Point(double x, double y){
    this->x = x; this->y = y;
}
void Point::disp() { cout<<"点（"<<x<<","<<y<<")"; }
