IntArray& IntArray::operator= (IntArray& a)
{
	if(pArr!=NULL)
		delete [] pArr ; //释放原有空间
	size = a.size;
	if(size==0)
		pArr = NULL;
	else
		pArr = new int[size+1]; //重新分配空间
	for(int i = 1; i<= size; i++) //逐个元素复制
		pArr[i] = a.pArr[i];
	return *this;
}
