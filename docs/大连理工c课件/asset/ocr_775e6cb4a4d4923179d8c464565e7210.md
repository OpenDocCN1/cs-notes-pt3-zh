class Clock
{
private :
    int hour, minute, second;    //关于时间的数据
public :
    void setClock (int h, int m, int s){
        hour= ( h >=0 && h <=23)? h : 0 ;
        minute= ( m >=0 && m <=59)? m : 0 ;
        second= ( s >=0 && s <=59)? s : 0 ;
    }
    void showClock ( ){
        cout<<hour<<minute<<second;
    }
};
