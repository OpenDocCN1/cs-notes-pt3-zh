```
int main(){
    Point p1(1.0,2.0),p2(3.0,4.0);
    Line l1(3.0,4.0,5.0);
    p1.disp(); p2.disp();
    cout<<"距离为："<<ComputeTools::distance(p1,p2)<<endl;
    p1.disp(); l1.disp();
    cout<<"距离为："<<ComputeTools::distance(p1,l1)<<endl;
    return 0;
}
```
