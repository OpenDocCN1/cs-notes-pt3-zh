# # 7.2 函数模板

#include <iostream.h>

template < typename T >
T min ( T a , T b )
{ return a < b ? a : b ; }

void main ( )
{ cout << " min ( 3 , 5 ) is " << min ( 3 , 5 ) << endl ;
cout << " min ( 'y' , 'e' ) is " << min ( 'y' , 'e' ) << endl ;
cout << " min ( 9.3 , 0.5 ) is " << min ( 9.3 , 0.5 ) << endl ;
}

int min ( int a , int b )
{ return a < b ? a : b ; }

char min ( char a , char b )
{ return a < b ? a : b ; }

double min ( double a , double b )
{ return a < b ? a : b ; }
