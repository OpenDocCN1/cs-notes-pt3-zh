```cpp
#include <iostream.h>
class Circle
{
    double radius;
public:
    void setRadius(double r)
    {radius=r; }
    double getRadius()
    {return radius; }
    double getGirth()
    {return radius*2*3.14;}
    double getArea()
    {return 3.14*radius*radius;}
};
```

}
