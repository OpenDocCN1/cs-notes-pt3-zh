# ## 7.2 函数模板

template <class T>
T min(T a, T b){
    return (a<b)? a: b;  }

template<> const char * min( const char * s1, const
char * s2 ){
return (strcmp(s1,s2)<0 ? s1 :s2); }

int main(){
    int m_iv = 5, m_iu = 10;
    double m_dv = 5.5, m_du = 10.5;
    cout <<< min(m_iv,m_iu) <<< endl;
    cout <<< min(m_dv,m_du) <<< endl;
    cout <<< min("hello","world") <<< endl;
    return 0;}
