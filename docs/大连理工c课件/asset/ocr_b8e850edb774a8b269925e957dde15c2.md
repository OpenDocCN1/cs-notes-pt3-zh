IntArray::IntArray(int s){
	size = s;
	if(size == 0) //空数组
	pArr = NULL;
	else //分配空间，0号元素不用
	pArr = new int[size+1];
	for(int i = 1; i<= size;i++)
		pArr[i] = 0; // 数组元素置0
}
IntArray::~IntArray(){
	if(pArr!=NULL)
		delete []pArr ; //释放数组空间
}
