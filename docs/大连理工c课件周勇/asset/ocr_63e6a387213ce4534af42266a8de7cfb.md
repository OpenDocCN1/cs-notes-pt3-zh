#include<iostream.h>
class Counter
{
    static int num;
public:
    void setnum(int i);
    void shownum();
};
int Counter::num = 0;
void main()
{
    Counter a, b;
    a.shownum();       b.shownum();
    a.setnum(10);
    a.shownum();       b.shownum();
}
