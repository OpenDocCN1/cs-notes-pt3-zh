# 例 使用公有静态数据成员

```
```cpp
#include<iostream.h>
class counter
{
public:
        void setnum ( int i ) { num = i ; }
        void shownum ( ) { cout << num << '\t' ; }
        static int num ;
} ;
int counter :: num = 1 ;        // 初始值为1
void main()
{
  int i ;
  for ( i = 0 ; i < 5 ; i ++ )
  {
    counter :: num += i ;
    cout << counter :: num << '\t' ;
  }
  cout << endl ;
}
