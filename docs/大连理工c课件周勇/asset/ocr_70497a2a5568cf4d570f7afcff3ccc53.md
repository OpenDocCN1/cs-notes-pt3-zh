# # 代码分析

//学生平均分
#include <iostream.h>
class Student
{ public:
    Student(int n,int a,float s):num(n),score(s)
    { count++; }
    void total() { sum+=score; }
    static float average() { return sum/count; }
private:
    int num;
    float score;
    static float sum;
    static int count;
};

> 最好只用静态成员函数
> 引用静态数据成员，
> 而不引用非静态数据成员。
