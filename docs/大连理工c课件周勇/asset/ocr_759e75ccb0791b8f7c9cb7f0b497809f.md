# # 默认复制构造函数

Date d1(3, 30, 2010);
Date d2 = d1;   Date d3(d2);

class Date
{
    int year, month, day;
    public:
    Date(int m, int d, int m);
    Date(const Date& d);
    set(int m, int d, int m);
    get() ;
};

Date :: Date(const Date& d)
{
    year = d.year;
    month = d.month;
    day = d.day;
}

将d对象中的数据成员的值赋给新定义的对象中的对应的数据成员
