# # 1. 结构一类

```
struct Clock
{
    int hour;
    int minute;
    int sec;
};
```

```
#include <iostream.h>
int main()
{
    Clock t1;
    setTime(t1);
    showTime( t1 );
    Clock t2;
    setTime(t2);
    showTime( t2 );
    return 0;
}
```

```
void setTime( Clock& t );
void showTime (Clock &t);
```
