# 代码模拟一统计对象的数目

```
```cpp
#include<iostream.h>
class Counter
{
	static int num ;
	public :
		Counter() { num++; }
		~Counter() { num--; }
		void setnum ( int i ) { num = i ; }
		void shownum() { cout << num << '\t' ; }
		static int get() { return num ; }
} ;
int Counter :: num = 0 ;
void main ()
{
	Counter a ; a.shownum()
	Counter b ; b.shownum()
	cout<<Counter::get() <<endl;
}```
```
