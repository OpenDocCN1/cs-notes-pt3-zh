if (Flag) { //X方向单位增量
for (n=0; n<= Length; n++) { //X方向插补过程
WritePixel(ix, Round(y), value);
ix+=idx;
y+=dy;
} //End of for
} //End of if
else { //Y方向斜率增量
for (n=0; n<= Length; n++) { //Y方向插补过程
WritePixel (Round(x), iy, value);
iy+=idy;
x+=dx;
} //End of for
} //End of else
} //Finish
