```

```
```sql
CREATE TABLE
Course(
    Cno CHAR(9) PRIMARY KEY,
    Cname CHAR(20) UNIQUE,
    Cpno CHAR(2),
    Credit NUMERIC(2,0),
    FOREIGN KEY (Cpno) REFERENCES Course(Cno)
);
```
