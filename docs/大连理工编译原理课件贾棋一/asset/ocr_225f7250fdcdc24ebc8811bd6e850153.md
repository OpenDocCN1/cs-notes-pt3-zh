procedure type;
begin
    if lookahead in {integer, char, num} then
        simple()
    else if lookahead = '^' then begin
        match('^'); match(id)
    end
    else if lookahead = array then begin
        match(array); match('['); simple();
        match(']'); match(of); type()
    end
    else error()
end;
