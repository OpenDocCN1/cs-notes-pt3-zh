# ## 3.7 分析器的生成器

lines : lines expr '\n' {printf ( "%g \n", $2 ) }
| lines '\n'
| /* ε */
;

expr : expr '+' expr {$$ = $1 + $3; }
| expr '-' expr {$$ = $1 - $3; }
| expr '*' expr {$$ = $1 * $3; }
| expr '/' expr {$$ = $1 / $3; }
| '(' expr ')' {$$ = $2; }
| '-' expr %prec UMINUS {$$ = -$2; }
| NUMBER
;

lines→ lines expr '\n' | lines '\n' | e
E→ E + E | E − E | E ∗ E | E / E | (E) | −E |
