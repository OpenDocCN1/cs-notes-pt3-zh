# 构造抽象语法树的属性文法定义转换成翻译模式

E → T    {R.i:=T.nptr}\
       {E.nptr:=R.s}

R → + T R₁    {R₁.i:=mknode('+',R.i,T.nptr)}\
              {R.s:=R₁.s}

R → − T R₁    {R₁.i:=mknode('−',R.i,T.nptr)}\
              {R.s:=R₁.s}

R → ε    {R.s:=R.i}

T → ( E )    {T.nptr:=E.nptr}

T → id    {T.nptr:=mkleaf(id,id.entry)}

T → num    {T.nptr:=mkleaf(num,num.val)}
