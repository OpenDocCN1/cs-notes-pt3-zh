# # 6.5.1 McCabe方法(McCabe’s Metrics)

PDL
procedure:sort
1: do while records remain
2: read record;
if record field 1=0
3: then process record;
store in buffer;
increment counter;
4: elseif record field 2=0
5: then reset counter;
6: else process record;
store in file;
7a: endif
endif
7b: enddo
8: end
