void sort (int *data,int n)
{
    int i,j,temp;
    for (i = n;i > 0;i--)
    {
        for (j = 0;j < i - 1;j++)
        {
            if (data[j] > data[j+1])
            {
                temp = data[j];
                data[j] = data[j+1];
                data[j+1] = temp;
            }
        }
    }
}

void display(int *data,int n)

        printf("%d ",data[i]);
