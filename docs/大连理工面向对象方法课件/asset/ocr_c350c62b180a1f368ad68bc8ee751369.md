```cpp
class Singleton
{
public:
    static Singleton* instance();
protected:
    Singleton();
private:
    static Singleton * instance_;
};

#include "stdafx.h"
#include "Singleton.h"
Singleton * Singleton::instance_ = 0;
Singleton::Singleton()
{
}
Singleton* Singleton::instance()
{
    if(instance_ == 0)
        instance_ = new Singleton;
    return instance_;
}

#include "stdafx.h"
#include "Singleton.h"
#include <assert.h>

int main(int argc, char* argv[])
{
    Singleton * ins = Singleton::instance();

    Singleton * ins4 = Singleton::instance();

    assert(ins == ins4);
