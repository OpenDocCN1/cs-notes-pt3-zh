

## Singleton 模式例子 1

```cpp
class Singleton
{
public:
    static Singleton* instance();
protected:
    Singleton();
private:
    static Singleton * instance_;
};

#include "stdafx.h"
#include "Singleton.h"
Singleton * Singleton::instance_ = 0;
Singleton::Singleton()
{
}
Singleton* Singleton::instance()
{
    if(instance_ == 0)
        instance_ = new Singleton;
    return instance_;
}

#include "stdafx.h"
#include "Singleton.h"
#include <assert.h>

int main(int argc, char* argv[])
{
    Singleton * ins = Singleton::instance();

    Singleton * ins4 = Singleton::instance();

    assert(ins == ins4);

    //Singleton ins2;

    //Singleton *ins3 = new Singleton;

    return 0;
}
```

解释：
Singleton * ins = Singleton::instance();
Singleton * ins4 = Singleton::instance();//两个代码返回的是同一个指针；不能建立Singleton的对象，因为要调用构造函数，如果建立则不能保证建立一个，有可能建立多个。
如果不是static 的 instance()，则没有办法调用该 instance()方法，因为只能用对象。instance()去掉用，这意味着要先创建Singleton的实例，就与初衷矛盾了。

## Singleton 模式例子 3

```cpp
#include “stdio.h”
#include “iostream.h”
```

# class Singleton

```cpp
{
public:
    static Singleton * instance();
protected:
    Singleton();
private:
    static Singleton * instance;
    static int number; //静态变量，使用时不用创建对象，“类名::number”即可
    static int num;
};
//--------------------------------
Singleton * Singleton::instance=0; //初始化
int Singleton::num=0; //初始化
```

```cpp
int Singleton::number=0; //初始化
Singleton::Singleton() //构造函数
{
    num++;
    cout<<"这是"<<num<<"个实例"<<endl;
}
Singleton * Singleton::instance()
{
    if(_instance==0)
    {
        _instance=new Singleton;
    }
    ++number;
    cout<<"这是第"<<number<<"次调用"<<endl;
    cout<<"产生了"<<num<<"个实例"<<endl<<endl;
    return _instance;
}
int main(int argc, char* argv[]) {
    Singleton * newsingleton1, * newsingleton2, * newsingleton3;
    newsingleton1=Singleton::instance(); //newsingleton1, newsingleton2, newsingleton3 的值都是一个
    newsingleton2=Singleton::instance();
    newsingleton3=Singleton::instance();
    return 0;
}
```

（**略讲**）创建 Singleton 类的子类的一个更灵活的方法是使用一个单件注册表。可能的 Singleton 类的集合不是由 Instance 定义的，Singleton 类可以根据名字在一个众所周知的注册表中注册它们的单件实例。

```cpp
class Singleton
{
public:
    static void Register(const string& name, Singleton*);
    static Singleton* Instance();
protected:
    static Singleton* Lookup(const string& name);
private:
    static Singleton* _instance;
    static map<string, Singleton*> _registry;
};
```
