while ( input >= 0 ) {
    cout << "#" << count << ":";
    cin >> input;
    total = total + input;
    count = count + 1;
}

count = count - 1;

if ( count > 0 ) {
    cout << "The total of your " << count;
    cout << " numbers is: " << total << endl;
    cout << "The average of your " << count
    cout << " number is: " << ( total / count ) << endl;
