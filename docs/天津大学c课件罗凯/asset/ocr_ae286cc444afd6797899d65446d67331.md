int min( int n1, int n2 );        // funct a
int min( int n1, float f1);       // funct b

int main() {
    cout << min( 2, 1 )     << endl; // calls a
    cout << min( 2, 3.2f )  << endl; // calls b
    cout << min( 2.5f, 4 )  << endl; // calls c
}
