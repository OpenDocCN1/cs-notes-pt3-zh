int i = 1;                 // creates global

int main() {
    cout << i << endl;     // uses global
    {
        char i = 'a';      // creates local
        cout << i << endl; // prints local
        ::i = 2;           // refer to global
        cout << i << endl; // prints local
        cout << ::i << endl; // print global
    }
    cout << i << endl;     // uses ????
    return 0;
}
