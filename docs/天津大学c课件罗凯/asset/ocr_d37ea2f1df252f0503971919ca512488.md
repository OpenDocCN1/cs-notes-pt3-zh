# However, Don’t Do This At Home

```
```cpp
void foo() { // #1
    { // #2
        int i = 1;
        cout << i << endl;        // prints 1
        { // #3
            cout << i << endl;    // prints 1
            char i = 'a';
            cout << i << endl;    // prints a
        } // closes #3
        cout << i << endl;        // prints 1
    } // closes #2
    cout << i << endl;            // unknown variable
} // closes #1
