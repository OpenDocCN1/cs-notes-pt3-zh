## Fixing our Problem

```cpp
// above main
const int END = -99;     // signal for no more scores

// below main
int averageScore ( int scores[][ NUM_STDS ],
                   int numClasses ) {
    int total = 0, stdCount = 0;
    for ( int i = 0; i < numClasses; i++ ) {
        for ( int j = 0; j < NUM_STDS &&
             scores[ i ][ j ] != END ; j++ ) {
            total += scores [ i ] [ j ];
            stdCount++;
        }
    }
    return total / stdCount ;
}
```
