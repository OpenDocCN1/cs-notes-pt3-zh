class Producer implements Runnable {
Q q;
Producer(Q q) {
this.q = q; new Thread(this, "Producer").start();
}
public void run() {
int i = 0;
while(true) {
q.put(i++);
}
}
}
class Consumer implements Runnable {
Q q;
Consumer(Q q) {
this.q = q; new Thread(this, "Consumer").start();
}
public void run() {
while(true) {
q.get();
}
}
}
