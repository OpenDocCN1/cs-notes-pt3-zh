// Demonstrate thread priorities.  
class clicker implements Runnable {  
int click = 0;  
Thread t;  
private volatile boolean running = true; // 每次值都得到验证  
public clicker(int p) {  
t = new Thread(this);  
t.setPriority(p);  
}  
public void run() {  
while (running) {  
click++;  
}  
}  
public void stop() {  
running = false;  
}  
public void start() {  
t.start();  
}  
}
