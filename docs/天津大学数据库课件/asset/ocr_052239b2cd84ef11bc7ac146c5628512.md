# # 文件管理阶段数据管理示例

```
```c
/* 程序 3：
求文件中 10 个数之和*/
#include <stdio.h>
main()
{
    int i,x,s=0,
    FILE * fp,
    /*打开文件*/
    fp=fopen("c:\\data.dat", "r");
    for(i=0;i<10;i++)
    {   /*文件中读数据*/
        fscanf(fp, "%d",&x);
        s=s+x;
    }
    printf("%d",s),
    fclose(fp), /*关闭文件*/
}

main()

        fscanf(fp, "%d",&x);
        if (s<x) s=x;
    }

}
