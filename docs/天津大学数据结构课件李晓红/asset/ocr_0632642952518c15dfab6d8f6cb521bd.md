int Push (SeqStack *S, StackData x) {
    //插入元素x为新的栈顶元素
    if ( StackFull (S) ){
        S->base =( StackData *)malloc(S->base，
                      (S->stacksize+ STACKINCREMENT) *
                      sizeof(StackData));
        if(! S->base)exit(overflow);
        S->top= S->base + S->stacksize;
        S->stacksize+= STACKINCREMENT; //追加存储空间
    }
    *(S->top)=x;
    (S->top) ++;
    Return ok
}
