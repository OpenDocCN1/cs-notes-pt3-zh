Status Push(SqStack &S,SElemType e)
{
    // 插入元素e为新的栈顶元素
    if(S.top-S.base>=S.stacksize) // 栈满，追加存储空间
    {
        S.base=(SElemType
        *)realloc(S.base,(S.stacksize+STACKINCREMENT)*size
of(SElemType));
        if(!S.base)
            exit(OVERFLOW); // 存储分配失败
        S.top=S.base+S.stacksize;
        S.stacksize+=STACKINCREMENT;
    }
    *(S.top)++=e;
    return OK;
}
