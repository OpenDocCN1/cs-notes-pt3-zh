# ## 入队

```
int EnQueue ( LinkQueue *Q, QueueData x ) {
    QueueNode *p = ( QueueNode * ) malloc
    ( sizeof ( QueueNode ) );
    p->data = x; p->link = NULL;
    if ( Q->front == NULL )
    //空,创建第一个结点
    Q->front = Q->rear = p;
    else Q->rear->link = p; //入队
    Q->rear =p;
    return 1;
}
```
