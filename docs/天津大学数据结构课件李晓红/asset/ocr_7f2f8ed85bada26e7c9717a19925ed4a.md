int Delete ( LinkList& first, int i ) {
    //在链表中删除第 i 个结点
    ListNode *p, *q;
    if ( i == 1 )              //删除表中第 1 个结点
    { q = first; first = first->link; }
    else {
        p = first; int k = 0;  //找第 i-1 个结点
        while ( p != NULL && k < i-1 )
        { p = p->link; k++; }
    }
}
