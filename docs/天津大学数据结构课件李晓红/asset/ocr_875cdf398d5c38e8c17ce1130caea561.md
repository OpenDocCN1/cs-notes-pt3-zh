void BFS (AdjGraph G, int v, int visited[ ]) {
    cout << GetValue (v) << ' ';  visited[v] = 1;
    Queue<int> q;  InitQueue(&q);
    EnQueue (&q, v);  //进队列
    while ( ! QueueEmpty (&q) ) {  //队空搜索结束
        DeQueue (&q, v);
        int w = GetFirstNeighbor (G, v);
        while ( w != −1) {  //若邻接顶点 w 存在
            if ( !visited[w] ) {  //未访问过
                cout << GetValue (w) << ‘ ’;
            }
        }
    }
}
