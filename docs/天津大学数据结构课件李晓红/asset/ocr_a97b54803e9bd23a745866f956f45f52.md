void selectSort ( int a[ ], int n ) {
    //对n个整数a[0],a[1],...,a[n-1]按递增顺序排序
    for ( int i = 0; i < n-1; i++ ) {
        int k = i;
        //从a[i]查到a[n-1], 找最小整数， 在a[k]
        for ( int j = i+1; j < n; j++ )
            if ( a[j] < a[k] ) k = j;
        int temp = a[i]; a[i] = a[k]; a[k] = temp;
    }
}
