# # 查找算法

Status SearchHash (HashTable H, KeyType K, int &p, int &c) {
// 在开放定址哈希表H中查找关键码为K的记录
p = Hash(K);    // 求得哈希地址
while ( H.elem[p].key != NULLKEY &&
!EQ(K, H.elem[p].key))
collision(p, ++c);    // 求得下一探查地址 p
if (EQ(K, H.elem[p].key)) return SUCCESS;
// 查找成功， 返回待查数据元素位置 p
else return UNSUCCESS; // 查找不成功
} // SearchHash
