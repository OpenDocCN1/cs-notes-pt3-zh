# 起泡排序的算法

typedef int SortData;

void BubbleSort ( SortData V[ ], int n ) {
    int i = 1;
    int exchange = 1;
    while ( i < n && exchange ){
        exchange = 0;  //标志置为0,假定未交换
        for ( int j = n-1; j >= i; j— )
            if ( V[j-1] > V[j] ) {  //逆序
                Swap ( V[j-1], V[j] ); //交换
                exchange = 1;  //标志置为1,有交换
            }
        i++;
    }
}
