void Queen( int i ) {
for ( int j = 0; j < n; j++ ) {
if ( !col[j] && !md[n+i-j-1] && !sd[i+j] )
{ /*第 i 行第 j 列没有攻击 */
col[j] = md[n+i-j-1] = sd[i+j] = 1;
q[i] = j; /*在第 i 行第 j 列安放皇后*/
if ( i == n-1 ) { /*输出一个布局*/
for ( int k = 0; k < n; k++ )
cout << q[k] << ' ';
cout << endl;
}
else Queen ( i+1 );
col[j] = md[n+i-j-1] = sd[i+j] = 0;
q[i] = 0; /*撤消第 i 行第 j 列的皇后*/
}
}
}
