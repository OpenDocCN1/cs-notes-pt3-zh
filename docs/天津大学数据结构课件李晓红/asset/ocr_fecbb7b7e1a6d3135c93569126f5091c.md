void PreOrder( BinTree T ){
    stack S;   InitStack(&S); //递归工作栈
    BinTreeNode * p = T; Push (&S, NULL);
    while ( p != NULL ) {
        cout << p->data << endl;
        if ( p->rightChild != NULL )
            Push ( &S, p->rightChild );
        if ( p->leftChild != NULL )
            p = p->leftChild; //进左子树
        else Pop( &S, p);
    }
}
