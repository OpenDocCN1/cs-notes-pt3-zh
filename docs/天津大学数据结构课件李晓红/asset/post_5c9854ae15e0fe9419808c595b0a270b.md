
## 入栈

```c
int Push(SeqStack *S, StackData x) {
    // 插入元素x为新的栈顶元素
    if (StackFull(S)) {
        S->base = (StackData *)realloc(S->base,
                      (S->stacksize + STACKINCREMENT) *
                      sizeof(StackData));
        if (!S->base) exit(OVERFLOW);
        S->top = S->base + S->stacksize;
        S->stacksize += STACKINCREMENT; // 追加存储空间
    }
    *(S->top) = x;
    (S->top)++;
    return OK;
}
```

## 取栈顶元素

```cpp
int GetTop(SeqStack *S, StackData &x) {
    // 若栈空返回0, 否则栈顶元素读到x并返回1
    if (StackEmpty(S)) return 0;
    (S->top)--;
    x = *(S->top);
    return 1;
}
```

## 出栈

```cpp
int Pop(SeqStack *S, StackData &x) {
    // 若栈空返回0, 否则栈顶元素退出到x并返回1
    if (StackEmpty(S)) return 0;
    --(S->top);
    x = *(S->top);
    return 1;
}
```

```cpp
Status ClearStack(SqStack &S)
{
    // 把S置为空栈

```

```cpp
Status StackEmpty(SqStack S)
{
    // 若栈S为空栈，则返回TRUE，否则返回FALSE
    if (S.top == S.base)
        return TRUE;
    else
        return FALSE;
}
```

```cpp
int StackLength(SqStack S)
{
    // 返回S的元素个数，即栈的长度
    return S.top - S.base;
}
```

```cpp
Status GetTop(SqStack S, SElemType &e)
{
    // 若栈不空，则用e返回S的栈顶元素，并返回OK；否则返回ERROR
    if (S.top > S.base)
    {
        e = *(S.top - 1);
        return OK;
    }
    else
        return ERROR;
}
```

```cpp
Status Push(SqStack &S, SElemType e)
{
    // 插入元素e为新的栈顶元素
    if (S.top - S.base >= S.stacksize) // 栈满，追加存储空间
    {
        S.base = (SElemType *)realloc(S.base, (S.stacksize + STACKINCREMENT) * sizeof(SElemType));
        if (!S.base)
            exit(OVERFLOW); // 存储分配失败
        S.top = S.base + S.stacksize;
        S.stacksize += STACKINCREMENT;
    }
    *(S.top)++ = e;
    return OK;
}
```

## 链式栈：栈的链接表示

- 链式栈无栈满问题，空间可扩充
- 插入与删除仅在栈顶处执行
- 链式栈的栈顶在链头
- 适合于多栈操作

![](img/04e8ddadedb998b3013c751f1cd77a21_127_0.png)

### 链式栈（LinkStack）的定义

```c
typedef int StackData;
typedef struct node {
    StackData data;        // 结点
    struct node *link;     // 链指针
} StackNode;
typedef struct {
    StackNode *top;        // 栈顶指针
} LinkStack;
```

### 链式栈操作实现

**初始化**

```c
void InitStack(LinkStack *S) {
    S->top = NULL;
}
```

**入栈**

```c
int Push(LinkStack *S, StackData x) {
    StackNode *p = (StackNode *)malloc(sizeof(StackNode));
    p->data = x;
    p->link = S->top;
    S->top = p;
    return 1;
}
```

**判栈空**

```c
int StackEmpty(LinkStack *S) {
    return S->top == NULL;
}
```

**出栈**

```cpp
int Pop(LinkStack *S, StackData &x) {
    if (StackEmpty(S)) return 0;
    StackNode *p = S->top;
    S->top = p->link;
    x = p->data;
    free(p);
    return 1;
}
```

**取栈顶**

```cpp
int GetTop(LinkStack *S, StackData &x) {
    if (StackEmpty(S)) return 0;
    x = S->top->data;
    return 1;
}
```

**置栈空**

```c
int MakeEmpty(LinkStack *S) {
    while (S->top != NULL) {
        StackNode *p = S->top;
        S->top = S->top->link;
        free(p);
    }
}
```

### 数制转换

N = (N div d) × d + N mod d

例如：`(1348)₁₀ = (2504)₈`，其运算过程如下：

| N | N div 8 | N mod 8 |
|---:|---:|---:|
| 1348 | 168 | 4 |
| 168 | 21 | 0 |
| 21 | 2 | 5 |
| 2 | 0 | 2 |

计算顺序 ↓

输出顺序 ↑

```c
void conversion() {
    InitStack(S);
    scanf("%d", &N);
    while (N) {
        Push(S, N % 8);
        N = N / 8;
    }
    while (!StackEmpty(S)) {
        Pop(S, e);
        printf("%d", e);
    }
} // conversion
```

### 行编辑程序

在用户输入一行的过程中，允许用户输入出差错，并在发现有误时可以及时更正。

设立一个输入缓冲区，用以接受用户输入的一行字符，然后逐行存入用户数据区；并假设“#”为退格符，“@”为退行符。

假设从终端接受两行字符：

```text
whli##ilr#e (s#*s)
```

```text
outcha@putchar(*s=#++);
```

实际有效行为：

```c
while (*s)
```

```c
putchar(*s++);
```
