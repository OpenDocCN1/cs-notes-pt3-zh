template<class T>
void Knapsack(T p[], int w[], int c, int n, T** f)
{
// Compute f[i][y] for all i and y.
    // initialize f[i][0]
    int ymax = min(w[0],c+1);
    for (int y = 0; y < ymax; y++)
        f[0][y] = 0;
    for (int y = w[0]; y <= c; y++)
        f[0][y] = p[0];
    // compute remaining f's
    for (int i = 1; i < n; i++) {
        ymax = min(w[i],c+1);
        for (int y = 0; y < ymax; y++)
            f[i][y] = f[i-1][y];
        for (int y = w[i]; y <= c; y++)
            f[i][y] = max(f[i-1][y], f[i-1][y-w[i]] + p[i]);
    }
    f[n][c] = f[n-1][c];
    if (c >= w[n])
        f[n][c] = max(f[n][c], f[n-1][c-w[n]] + p[n]);
}
