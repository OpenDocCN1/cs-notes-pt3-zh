template<class T>
void AdjacencyWDigraph<T>::AllPairs(T **c, int n)
{
    // All pairs shortest paths.
    // Compute c(i][j] and hay[i][j] for all i
    // and j.
    // initialize c(i][j] = c[i][j]
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            c[i][j] = a[i][j];
    for (int i = 1; i <= n; i++)
        c[i][i] = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            if (c[i][j] < MaxInt)
                hay[i][j] = j;
            else hay[i][j] = 0;
    // compute c(i][j] = c[i][k] + c[k][j]
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            if (c[i][k] != MaxInt)
                for (int j = 1; j <= n; j++)
                    if (c[k][j] != MaxInt &&
                        t2 = c[i][k] + c[k][j])
                        c[i][j] = t2;
                        hay[i][j] = hay[i][k];
}

    // Actual code to output i to j path.
