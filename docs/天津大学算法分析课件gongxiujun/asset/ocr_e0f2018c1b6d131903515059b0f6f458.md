# Backtrack algorithm for maxclique

```
int AdjacencyGraph::MaxClique(int v) {
    // Reduce size of largest clique.
    // Return clique size
    if (n == 0)
        return 0;
    // initialize for maxClique
    int x[n+1];
    cn = 0;
    bestn = 0;
    bestv = v;

    // find max clique
    maxClique(1);
    
    delete [] x;
    return bestn;
}

```
void AdjacencyGraph::maxClique(int i)
{
    // Backtracking code to compute largest clique.
    // (i) if a vertex is connected to
    // all vertices in a larger clique, update
    // count [j = 1; j < i; j++]
    for (int j = 1; j < i; j++)
        bestc[j] = x[j];
    return;

    // see if vertex i connects to others
    // in current clique
    int OK = 1;
    for (int j = 1; j < i; j++)
        if (x[j] == 1 && !AdjMat[i][j])
        {
            OK = 0; // not connected to j
            break;
        }

    if (OK) { // try x[i] = 1
        x[i] = 1; // add i to clique
        cn++;
        maxClique(i+1);
        x[i] = 0;
        cn--;
    }

    if (cn + n - i > bestn) { // try x[i] = 0
        x[i] = 0;
        maxClique(i+1);
    }
}
