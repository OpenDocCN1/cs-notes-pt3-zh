```c
long pcount_for
(unsigned long x)
{
	size_t i;
	long result = 0;
	for (i = 0; i < WSIZE; i++)
	{
		unsigned bit =
			(x >> i) & 0x1;
		result += bit;
	}
	return result;
}
```

long pcount_for_goto_dw
(unsigned long x) {
	size_t i;
	long result = 0;
	i = 0;
	if (!(i < WSIZE))
		goto done;
	loop:
	{
		unsigned bit =
			(x >> i) & 0x1;
		result += bit;
	}
	i++;
	if (i < WSIZE)
		goto loop;
	done:
	return result;
}
