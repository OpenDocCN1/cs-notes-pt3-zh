# # C Assessment: Question 1

All allocated memory must be freed!

```
```c
int main(int argc, char** argv) {
    int *a = (int*) calloc(213 * sizeof(int));
    if (a == NULL) return 0;
    for (int i=0; i<213; i++) {
        if (a[i] == 0) a[i]=i;
        else a[i]=-i;
    }
    free(a);
    return 0;
}
