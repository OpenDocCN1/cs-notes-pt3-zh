/\* Multiply n x n matrices a and b \*/
void mmm(double *a, double *b, double *c, int n) {
    int i, j, k;
    for (i = 0; i < n; i+=B)
        for (j = 0; j < n; j+=B)
            for (k = 0; k < n; k+=B)
                /\* B x B mini matrix multiplications \*/
                for (il = i; il < i+B; i++)
                    for (jl = j; jl < j+B; j++)
                        for (kl = k; kl < k+B; k++)
                            c[il*n+jl] += a[il*n + kl]*b[kl*n + jl];
}
