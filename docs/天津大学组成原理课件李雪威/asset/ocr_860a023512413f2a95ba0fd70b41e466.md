# ## Jump table

```
```asm
.section .rodata
.align 8
.L4:
	.quad .L8 # x = 0
	.quad .L3 # x = 1
	.quad .L5 # x = 2
	.quad .L9 # x = 3
	.quad .L8 # x = 4
	.quad .L7 # x = 5
	.quad .L7 # x = 6
```

case 6: // .L7
