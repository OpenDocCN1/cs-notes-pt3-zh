# # Understanding Swap()

```
```c
void swap
(long *xp, long *yp)
{
    long t0 = *xp;
    long t1 = *yp;
    *xp = t1;
    *yp = t0;
}
```

```asm
swap:
    movq  (%rdi), %rax    # t0 = *xp
    movq  (%rsi), %rdx    # t1 = *yp
    movq  %rdx, (%rdi)    # *xp = t1
    movq  %rax, (%rsi)    # *yp = t0
    ret
