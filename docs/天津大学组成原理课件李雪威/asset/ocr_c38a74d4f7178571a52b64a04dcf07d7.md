# “Do-While” Loop Compilation

# Goto Version

```
long pcount_goto
  (unsigned long x) {
  long result = 0;
loop:
  result += x & 0x1;
  x >>= 1;
  if(x) goto loop;
  return result;
}
```

| Register | Use(s) |
|---|---|
| %rdi | Argument x |
| %rax | result |

```
        movl    $0, %eax     #  result = 0
.L2:                         # loop:
        movq    %rdi, %rdx
        andl    $1, %edx      #  t = x & 0x1
        addq    %rdx, %rax    #  result += t
        shrq    %rdi          #  x >>= 1
        jne     .L2           #  if (x) goto loop
        rep; ret
```
