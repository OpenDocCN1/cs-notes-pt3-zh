# ## Setup:

switch_eg:
movq	%rdx, %rcx
cmpq	$6, %rdi	# x:6
ja	.L8
jmp	*.L4(,%rdi,8)

# ## What range of values takes default?

| Register | Use(s) |
|---|---|
| %rdi | Argument x |
| %rsi | Argument y |
| %rdx | Argument z |
| %rax | Return value |

Note that w not initialized here
