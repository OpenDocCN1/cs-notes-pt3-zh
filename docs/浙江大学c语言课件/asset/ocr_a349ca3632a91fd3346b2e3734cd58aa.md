+   【Example 2】用递归方法求n!
+   【递归公式】
+   1 n = 0,1
+   n! = (n-1)!·n n > 1
+   【Solution】
+   #include <stdio.h>
+   long fac(int n)
+   { long f;
+   if (n < 0) printf("Error!\n");
+   else if (n == 0 || n == 1)
+   f = 1;
+   else f = fac(n-1) * n;
+   return(f);}
+   main()
+   { int n; long y;
+   printf("Please input an integer number: ");
+   scanf("%d", &n);
+   y = fac(n);
+   printf("%d! = %ld\n", n, y);
+   }
