## 3.3 Else-if

```c
if (expression)
    statement
else if (expression)
    statement
else if (expression)
    statement
else
    statement
```

```c
/* binsearch: find x in v[0]<=v[1]<=...<=v[n-1] */
int binsearch(int x, int v[], int n)
{
    int low, high, mid;
    low = 0;
    high = n - 1;
    while (low <= high) {
        mid = (low + high) / 2;
        if (x < v[mid])
            high = mid - 1;
        else if (x > v[mid])
            low = mid + 1;
        else
            return mid;
    }
    return -1;
}
```

#include <stdio.h>
main() /* count digits, white space, others */

    int c, i, nwhite, nother, ndigit[10];

        ndigit[i] = 0;

## 3.5 Loops — While and For

- while:
  - `while (expression)`
    `statement`
- For

![](img/1879308d3bf2ec0664445e4bc0146293_57_0.png)

```c
for (expr1; expr2; expr3)
    statement
```

```c
expr1;
while (expr2) {
    statement
    expr3;
}
```

infinite loop:

```c
for (;;) { ... }
```

processing the first n elements of an array:

```c
for (i = 0; i < n; i++)
    ...
```

### Example 1:

```c
#include <ctype.h>
/* atoi: convert s to integer; version 2 */
int atoi(char s[])
{
    int i, n, sign;
    for (i = 0; isspace(s[i]); i++)
        ; /* skip white space */
    sign = (s[i] == '-') ? -1 : 1;
    if (s[i] == '+' || s[i] == '-')
        i++;
    for (n = 0; isdigit(s[i]); i++)
        n = 10 * n + (s[i] - '0');
    return sign * n;
}
```

/* shellsort: sort v[0]...v[n-1] into increasing order */
void shellsort(int v[], int n)
{
    int gap, i, j, temp;
    for (gap = n / 2; gap > 0; gap /= 2)
        for (i = gap; i < n; i++)
            for (j = i - gap; j >= 0 && v[j] > v[j + gap]; j -= gap) {
                temp = v[j];
                v[j] = v[j + gap];
                v[j + gap] = temp;
            }
}

### Example 3:

```c
#include <string.h>
/* reverse: reverse string s in place */
void reverse(char s[])
{
    int c, i, j;
    for (i = 0, j = strlen(s) - 1; i < j; i++, j--) {
        c = s[i];
        s[i] = s[j];
        s[j] = c;
    }
}
```

## 3.6 Loops — Do-while

syntax:

```c
do
    statement
while (expression);
```

### Example:

```c
/* itoa: convert n to characters in s */
void itoa(int n, char s[])
{
    int i, sign;
    if ((sign = n) < 0) /* record sign */
        n = -n;
    i = 0;
    do {
        s[i++] = n % 10 + '0'; /* get next digit */
    } while ((n /= 10) > 0); /* delete it */
    if (sign < 0)
        s[i++] = '-';
    s[i] = '\0';
    reverse(s);
}
```
