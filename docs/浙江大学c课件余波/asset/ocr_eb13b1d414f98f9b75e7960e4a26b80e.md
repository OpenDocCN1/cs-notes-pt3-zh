```cpp
#include<iostream.h>
void main()
{
    int sum=0; int inval; int num=0;
    while(1)
    {
        cin>>inval;
        if(inval==0)break;
        if(inval<0)continue;
        sum=sum+inval; num++;
    }
    cout<<"sum/num="<<(float)sum/num;
}
