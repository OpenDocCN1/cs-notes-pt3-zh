        System.out.println("(m+n>k)?(m++):(--m+n++)的运算结果："+temp);

        float x=3.14f;
        System.out.println("float型变量强制转换为int型的结果："+(int)x);

        m=1;
        x=(m+2*n)/k;
        System.out.println("int型变量运算：(m+2*n)/k的结果是："+((m+2*n)/k));
        System.out.println("(m+2*n)/k的结果是赋值给float型变量x="+x);
    }
