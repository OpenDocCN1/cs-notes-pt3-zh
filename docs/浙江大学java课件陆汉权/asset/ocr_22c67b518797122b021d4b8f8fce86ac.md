# 以字节流的方式读文件

```
import java.io.*;
public class E10_5{
public static void main(String []args)throws IOException{
try{
FileInputStream fr=new FileInputStream("d:\td\E10_4.txt");
int i;
while(((i=fr.read())!= -1))
// -1为文件结束符
System.out.print((char)i);
fr.close();
}
catch(FileNotFoundException e){
System.out.println(e);
}
catch(IOException e){
System.out.println(e);
}
}
}
```
