public class Chap4DemoMathod2 {
    static int func(int n, int m) { // 定义Java方法
        int sum=0;
        for( ; n<=m; n++ ) //Σ (n, n+1, ... m)
        sum=sum+n;
        return sum;
    }
    public static void main(String[] args) {
        int f;
        f = func(2, 100)+func(200, 298)+ // 调用方法
        func(-99, -1)+func(51, 199);
        System.out.printf("The Result: %d", f);
    }
}
