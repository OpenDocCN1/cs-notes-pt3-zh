public static void main(String args[]) {

    Student1 p2=new Student1("张敏", "上海市闵行区");

    System.out.println(p2.name+" "+p2.address+" "+p2.count);
    p1.count=p1.count-1; // count为2
    System.out.println(p2.name+" "+p2.address+" "+p2.count);
}
} // 类结束
