    }
    public void maintain() {
    }
    public String forward() {
        return "Little_car forward";
    }
    public String reverse() {
        return "Little_car reverse";
    }
}
interface Automobile { // 接口
    int i = 5;
    void accelent();
    void maintain();
    String forward();
    String reverse();
}
