public class no7 {
    static int gcd(int x, int y){
        int r;
        r=y % x;
        while (r!=0) {
        y=x;
        x=r;
        r=y % x; }
        return x;
    }
    public static void main(String[] args) {
        int a=Integer.parseInt(args[0]);
        int b=Integer.parseInt(args[1]);
        int n=gcd(a,b);
        System.out.println(n);
    }
} // 实验结果 若输入4 与6， 所得结果为2
