# Examples

```
GLfloat list[6][2] ;

glColor3f(0.0, 1.0, 0.0);
glBegin(GL_TRIANGLES)
    for (int i = 0 ; i < 6 ; i++)
        glVertex2v(list[i]);
glEnd() ;

glBegin(GL_TRIANGLES)
    glColor3f(1.0, 0.0, 0.0);
    for ( i = 0 ; i < 3 ; i++)
        glVertex2v(list[i]);
    glColor3f(1.0, 1.0, 1.0);
    for ( i = 3 ; i < 6 ; i++)
        glVertex2v(list[i]);
glEnd() ;
```
