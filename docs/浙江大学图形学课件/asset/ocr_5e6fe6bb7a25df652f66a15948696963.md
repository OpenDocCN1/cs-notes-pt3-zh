# Examples

```
GLfloat list[6][2] ;

glBegin(GL_LINES)
    for (int i = 0 ; i < 6 ;i++)
        glVertex2v(list[i]);
glEnd() ;

glBegin(GL_LINE_STRIP)
    for (int i = 0 ; i < 6 ;i++)
        glVertex2v(list[i]);
glEnd() ;

glBegin(GL_LINE_LOOP)
    for (int i = 0 ; i < 6 ;i++)
        glVertex2v(list[i]);
glEnd() ;
```
