# # {2)Insertion

+   - Add a new tuple to _account:_

INSERT INTO _account_  
VALUES ('A_9732', 'Perryridge', 1200);  
or equivalently  
INSERT INTO _account_  
_(branch_name, balance, account_number)_  
VALUES ('Perryridge', 1200, 'A_9732');

Format:

INSERT INTO <table/view>[(c1,c2,...)]  
VALUES (e1,e2,...);

INSERT INTO <table/view>[(c1,c2,...)]  
select e1,e2,... from ...;
