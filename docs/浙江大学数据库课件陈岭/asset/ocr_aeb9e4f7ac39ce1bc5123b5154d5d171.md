# # Example of BCNF Decomposition

R = (branch-name, branch-city, assets,
customer-name, loan-number, amount)
F = { branch-name → branch-city assets
loan-number → amount branch-name }

Key = { loan-number, customer-name },
∴ (loan-number, customer-name ) +=R

# ## Decomposition

R1 = (branch-name, branch-city, assets),
R2 = (branch-name, customer-name, loan-number, amount),
R3 = (loan-number, branch-name, amount)
R4 = (customer-name, loan-number)

# ## Final decomposition :

R1, R3, R4 , all in BCNF.
