Example1: Find out the students who have enrolled more than 10 courses.
Select sno from enroll
group by sno
having count(cno) > 10

Example2: Find out the student names who have enrolled more than 10 courses.
Select sno, sname, Count( cno) course_num ??
from student where sno in
(select sno from enroll  group by sno
having count(cno) > 10 )
