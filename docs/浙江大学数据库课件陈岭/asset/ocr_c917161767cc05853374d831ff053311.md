Every loan has at least one borrower who maintains an account with a minimum balance of $1000. (对每一笔借款，至少有一个借款人还有存款 $1000以上。)

create assertion balance-constraint check (not exists ( select * from loan L where not exists ( select * from borrower B, depositor D, account A where L.loan-number = B.loan-number and B.customer-name = D.customer-name and D.account-number = A.account-number and A.balance >= 1000)))
