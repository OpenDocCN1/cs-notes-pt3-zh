# # DFS算法

```
void DFS ( Vertex V )
{ visited[V] = true;
    for ( V 的每个邻接点 W )
        if ( !visited[W] )
            DFS(W);
}
```

```
int DFS ( Vertex V )
{ visited[V] = true;
    if ( IsSafe(V) ) answer = YES;
    else {
        for ( each W in G )
            if ( !visited[W] && Jump(V,W) ) {
                answer = DFS(W);
                if (answer==YES) break;
            }
    }
    return answer;
}
```
