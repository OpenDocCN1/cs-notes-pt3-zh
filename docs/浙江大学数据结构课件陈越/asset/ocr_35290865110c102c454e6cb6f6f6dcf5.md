int main()
{
    SetType S;
    int n;
    char in;
    scanf("%d\n", &n);
    Initialization( S, n );
    do {
        scanf("%c", &in);
        switch (in) {
        case 'I': Input_connection( S ); break;
        case 'C': Check_connection( S ); break;
        case 'S': Check_network( S, n ); break;
        }
    } while ( in != 'S' );
    return 0;
}
