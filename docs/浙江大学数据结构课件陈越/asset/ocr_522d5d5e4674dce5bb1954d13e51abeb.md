# # 如何建搜索树

```
Tree MakeTree( int N )
{
        Tree T;
        int i, V;

        scanf("%d", &V);
        T = NewNode(V);
        for (i=1; i<=N; i++) {
                scanf("%d", &V);
                T = Insert(T, V);
        }
        return T;
}
```

```
Tree Insert( Tree T, int V )
{
        if ( !T ) T = NewNode(V);
        else {
                if ( V>T->v )
                        T->Right = Insert( T->Right, V );
                else
                        T->Left = Insert( T->Left, V );
        }
        return T;
}
```

```
Tree NewNode( int V )
{
        Tree T = (Tree)malloc(sizeof(struct TreeNode));
        T->V = V;
        T->Left = T->Right = NULL;
        T->flag = 0;
        return T;
}
```
