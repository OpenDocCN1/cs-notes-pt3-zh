```cpp
void TopSort()
{
    for ( cnt = 0; cnt < |V|; cnt++ ) {
        v = 未输出的入度为0的顶点;  /* O(|V|) */
        if ( 这样的v不存在 ) {
            Error ( “图中有回路” );
            break;
        }
        输出v，或者记录v的输出序号;
        for ( v 的每个邻接点 w )
            Indegree[w]--;
    }
}
```
