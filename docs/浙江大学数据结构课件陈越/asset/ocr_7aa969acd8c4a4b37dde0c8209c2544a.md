```
int G[MAXN][MAXN], Nv, Ne;
void BuildGraph()
{   int i, j, v1, v2, w;

    scanf("%d", &Nv);
    /* Create Graph */
    for (i=0; i<Nv; i++)
        for (j=0; j<Nv; j++)
            G[i][j] = 0;  /* 或INFINITY */
    scanf("%d", &Ne);
    for (i=0; i<Ne; i++) {
        scanf("%d %d %d", &v1, &v2, &w);
        /* InsertEdge */
        G[v1][v2] = w;
        G[v2][v1] = w;
    }
}
```
