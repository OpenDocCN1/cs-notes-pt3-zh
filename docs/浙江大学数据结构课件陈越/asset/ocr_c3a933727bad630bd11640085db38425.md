# 如何判别

```
int Judge( Tree T, int N )
{
        int i, V, flag = 0;
        /* flag: 0代表目前还一致，1代表已经不一致*/

        scanf("%d", &V);
        if ( V!=T->v) flag = 1;
        else T->flag = 1;
        for (i=1; i<N; i++) {
                scanf("%d", &V);
                if ( (!flag) && (!check(T, V)) ) flag = 1;
        }
        if (flag) return 0;
        else return 1;
}
```
