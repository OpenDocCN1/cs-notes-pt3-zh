Position Find( ElementType Key, HashTable H )
{
    Position P;
    int Pos;

    Pos = Hash( Key, H->TableSize );  /*初始散列位置*/
    P = H->TheLists[Pos].Next;        /*获得链表头*/
    while( P != NULL && strcmp(P->Element, Key) )
        P = P->Next;
    return P;
}
