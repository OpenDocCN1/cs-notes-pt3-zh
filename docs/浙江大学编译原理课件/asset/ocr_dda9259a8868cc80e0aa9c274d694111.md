# # Example: A FORTRAN77 sample program

```
SUBROUTINE   QUADMEAN(A, SIZE,QMEAN)
COMMON   MAXSIZE
INTEGER   MAXSIZE,SIZE
REAL A(SIZE),QMEAN,TEMP
INTEGER K
TEMP=0.0
IF ((SIZE .GT. MAXSIZE) .OR. (SIZE .LT. 1) GOTO 99
DO 10 K=1,SIZE
   TEMP=TEMP+A(K)*A(K)
10   CONTINUE
99   QMEAN = SQRT(TEMP/SIZE)
RETURN
END
```
