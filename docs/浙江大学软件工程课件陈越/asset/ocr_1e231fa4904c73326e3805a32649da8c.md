```c
#include <stdio.h>
#include <ctype.h>
#include <string.h>
# define MAXWORD 100

int getword(char *, int);
struct key *binsearch(char *, struct key *, int);

main()
{
        struct key *p;
        char word[MAXWORD];
        while (getword(word, MAXWORD) != EOF)
                if (isalpha(word[0]))
                        if ((n = binsearch(word, keytab, NKEYS)) >= 0)
                                p->count++;
        for (p=keytab; p<keytab+NKEYS; p++)
                if (p->count > 0) printf("%4d %s\n", p->count, p->word);
        return 0;
}
