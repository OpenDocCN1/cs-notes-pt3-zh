# # 6.3 Arrays of Structures

```
/* binsearch: find word in tab[0]...tab[n-1] */
int binsearch(char *word, struct key tab[], int n)
{
    int cond, low, high,mid;
    low=0;
    while(low<=high) {
        mid=(low+high)/2;
        if ((cond=strcmp(word, tab[mid].word))<0)
            high = mid-1;
        else if (cond >0) low = mid +1;
        else return mid;
    }
    return -1;
}
```
