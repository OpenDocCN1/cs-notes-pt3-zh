# 6.2 Structures and Functions

struct pointer:
struct point *pp;
(*pp).x and (*pp).y
pp->x and pp->y
are members

e.g: struct point origin, *pp;
pp = &origin;
printf("origin is (%d, %d)\n", (*pp).x, (*pp).y);
printf("origin is (%d, %d)\n", pp->x, pp->y);

Notice: . and -> associate from left to right:
struct rect r. *rp=r;
r.pt1.x <=> rp->pt1.x <=> (r.pt1).x <=> (rp->pt1).x
