# 7.5 File Access

```
eg2: concatenate files
#include <stdio.h>

main(int argc, char *argv[])
{
    FILE *fp;
    if(argc==1)
        filecopy(stdin,stdout);
    else
        while(--argc>0)
            if((fp=fopen(*++argv,"r")) == NULL) {
                printf("cat:can’t open %s\n",*argv);
                return 1;
            } else {
                filecopy(fp,stdout);
                fclose(fp);
            }
    return 0;
}
```
