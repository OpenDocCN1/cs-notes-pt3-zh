#include <stdio.h>
int factors(int i, int fac[]); /* 函数原型说明 */
main()
{ int fac[1000];
int i, j, n, s;
for (i = 0; i < 1000; i++) {
n=factors(i, fac); /*调函数factors求i的所有因子*/
s = 0; /*fac 为存放因子的数组， n为个数*/
for (j = 0; j < n; j++)
s += fac[j]; /* 将所有的因子累加起来 */
if (i == s) { /* 若该数i等于其所有因子之和 */
printf("%3d its factors are ", i);
for (j = 0; j < n; j++) {
printf("%d", fac[j]);
if (j != n-1) printf(","); }
}
}
}
